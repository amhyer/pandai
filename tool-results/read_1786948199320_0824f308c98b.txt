   372→function PerformanceBar({ value, max = 100, height = 'h-2.5' }: { value: number; max?: number; height?: string }) {
   373→  const pct = Math.min((value / max) * 100, 100);
   374→  const color = pct >= 75 ? 'bg-emerald-500' : pct >= 60 ? 'bg-amber-500' : 'bg-red-500';
   375→  return (
   376→    <div className={cn('w-full overflow-hidden rounded-full bg-muted', height)}>
   377→      <div
   378→        className={cn('h-full rounded-full transition-all duration-500', color)}
   379→        style={{ width: pct + '%' }}
   380→      />
   381→    </div>
   382→  );
   383→}

  1200→// 4. GURU NILAI VIEW  (UPGRADED WITH API)
  1201→// ═══════════════════════════════════════════════════════════════════
  1202→
  1203→export function GuruNilaiView() {
  1204→  const user = useAppStore((s) => s.user);
  1205→  const [exams, setExams] = useState<ExamPackageData[]>([]);
  1206→  const [attempts, setAttempts] = useState<AttemptData[]>([]);
  1207→  const [students, setStudents] = useState<StudentData[]>([]);
  1208→  const [selectedExam, setSelectedExam] = useState('');
  1209→  const [loading, setLoading] = useState(true);
  1210→  const [saving, setSaving] = useState(false);
  1211→  const [nilaiMap, setNilaiMap] = useState<Record<string, string>>({});
  1212→  const [learningObjective, setLearningObjective] = useState('');
  1213→
  1214→  const fetchExams = useCallback(async () => {
  1215→    try {
  1216→      const params = new URLSearchParams();
  1217→      if (user?.schoolId) params.set('schoolId', user.schoolId);
  1218→      const res = await fetch(`/api/exams?${params.toString()}`);
  1219→      if (!res.ok) throw new Error('API Error');
  1220→      const data = await res.json();
  1221→      setExams(Array.isArray(data) ? data : []);
  1222→    } catch {
  1223→      // silent fallback
  1224→    }
  1225→  }, [user?.schoolId]);
  1226→
  1227→  const fetchAttempts = useCallback(async (examPackageId?: string) => {
  1228→    setLoading(true);
  1229→    try {
  1230→      const params = new URLSearchParams();
  1231→      if (user?.schoolId) params.set('schoolId', user.schoolId);
  1232→      const res = await fetch(`/api/attempts?${params.toString()}`);
  1233→      if (!res.ok) throw new Error('API Error');
  1234→      const data = await res.json();
  1235→      let all = Array.isArray(data) ? data : [];
  1236→      if (examPackageId) {
  1237→        all = all.filter((a: AttemptData) => a.examPackageId === examPackageId);
  1238→      }
  1239→      setAttempts(all);
  1240→    } catch {
  1241→      setAttempts([]);
  1242→    } finally {
  1243→      setLoading(false);
  1244→    }
  1245→  }, [user?.schoolId]);
  1246→
  1247→  const fetchStudents = useCallback(async () => {
  1248→    try {
  1249→      const params = new URLSearchParams();
  1250→      params.set('role', 'SISWA');
  1251→      if (user?.schoolId) params.set('schoolId', user.schoolId);
  1252→      const res = await fetch(`/api/users?${params.toString()}`);
  1253→      if (!res.ok) throw new Error('API Error');
  1254→      const data = await res.json();
  1255→      setStudents(Array.isArray(data) ? data : []);
  1256→    } catch {
  1257→      setStudents([]);
  1258→    }
  1259→  }, [user?.schoolId]);
  1260→
  1261→  useEffect(() => {
  1262→    fetchExams();
  1263→    fetchStudents();
  1264→    fetchAttempts();
  1265→  }, [fetchExams, fetchStudents, fetchAttempts]);
  1266→

  1281→      setNilaiMap((prev) => ({ ...prev, [id]: value }));

  1305→            userId: attempt.userId,
  1306→            examPackageId: attempt.examPackageId,
  1307→            schoolId: attempt.schoolId,

  1311→              answer: i < benar ? 'correct' : 'wrong',
  1312→              timeSpent: 0,

  1315→            learningObjective: learningObjective.trim() || undefined,
  1316→          }),
  1317→        });

  1320→      fetchAttempts(selectedExam || undefined);

  1322→    } catch {

  1330→
  1331→  return (

  1335→          className="bg-[#1F3864] transition-all duration-200 hover:bg-[#152850] hover:shadow-sm active:scale-[0.98]"
  1336→          onClick={handleSave}
  1337→          disabled={saving}
  1338→        >
  1339→          {saving ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Save className="mr-2 h-4 w-4" />}
  1340→          {saving ? 'Menyimpan...' : 'Simpan Nilai'}
  1341→        </Button>
  1342→      </PageHeader>
  1343→
  1344→      {/* Selectors */}
  1345→      <Card className="rounded-xl shadow-sm">
  1346→        <CardContent className="flex flex-col gap-4 p-4 sm:flex-row sm:items-end">
  1347→          <div className="flex-1 space-y-1.5">
  1348→            <Label className="text-sm font-medium">Pilih Tryout</Label>
  1349→            <Select value={selectedExam} onValueChange={handleExamChange}>
  1350→              <SelectTrigger className="rounded-lg focus:ring-[#1F3864]/30">
  1351→                <GraduationCap className="mr-2 h-4 w-4" />
  1352→                <SelectValue placeholder="Pilih tryout..." />
  1353→              </SelectTrigger>
  1354→              <SelectContent>
  1355→                {exams.map((e) => (
  1356→                  <SelectItem key={e.id} value={e.id}>{e.title}</SelectItem>
