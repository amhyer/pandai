   101→function getInitials(name: string) {
   102→  return name
   103→    .split(' ')
   104→    .map((w) => w[0])
   105→    .slice(0, 2)
   106→    .join('')
   107→    .toUpperCase();
   108→}

   110→function getAvatarColor(name: string) {
   111→  const colors = [
   112→    'bg-gradient-to-br from-rose-400 to-rose-600',
   113→    'bg-gradient-to-br from-sky-400 to-sky-600',
   114→    'bg-gradient-to-br from-emerald-400 to-emerald-600',
   115→    'bg-gradient-to-br from-amber-400 to-amber-600',
   116→    'bg-gradient-to-br from-violet-400 to-violet-600',
   117→  ];
   118→  let hash = 0;
   119→  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
   120→  return colors[Math.abs(hash) % colors.length];
   121→}

   136→function getScoreColor(score: number) {
   137→  if (score >= 4.5) return { text: 'text-emerald-600', bg: 'bg-emerald-50', label: 'Luar Biasa' };
   138→  if (score >= 3.5) return { text: 'text-green-600', bg: 'bg-green-50', label: 'Baik' };
   139→  if (score >= 2.5) return { text: 'text-amber-600', bg: 'bg-amber-50', label: 'Cukup' };
   140→  return { text: 'text-red-500', bg: 'bg-red-50', label: 'Perlu Perhatian' };
   141→}

   144→  if (score >= 4) return 'bg-emerald-400';
   145→  if (score >= 3) return 'bg-amber-400';
   146→  if (score >= 2) return 'bg-orange-400';

   153→
   154→function HabitRatingSelector({
   155→  rating,
   156→  onChange,
   157→  size = 'md',
   158→  showLabel = false,
   159→}: {
   160→  rating: number;
   161→  onChange?: (r: number) => void;
   162→  size?: 'sm' | 'md' | 'lg';
   163→  showLabel?: boolean;
   164→}) {
   165→  const [hovered, setHovered] = useState(0);
   166→  const displayRating = hovered || rating;
   167→
   168→  const sizeClasses = { sm: 'text-xs px-2 py-1', md: 'text-sm px-3 py-1.5', lg: 'text-base px-4 py-2' };
   169→
   170→  return (
   171→    <div className="flex items-center gap-1.5">
   172→      <div className="flex gap-1">
   173→        {RATING_SCALE.map((level) => (
   174→          <button
   175→            key={level}
   176→            type="button"
   177→            onClick={() => onChange?.(level === rating ? 0 : level)}
   178→            onMouseEnter={() => onChange && setHovered(level)}
   179→            onMouseLeave={() => setHovered(0)}
   180→            className={cn(
   181→              sizeClasses[size],
   182→              'rounded-lg border font-semibold transition-all duration-200',
   183→              onChange
   184→                ? 'cursor-pointer hover:scale-105 active:scale-95'
   185→                : 'cursor-default',
   186→              level <= displayRating
   187→                ? level === 4 ? 'bg-blue-100 border-blue-300 text-blue-700'
   188→                  : level === 3 ? 'bg-emerald-100 border-emerald-300 text-emerald-700'
   189→                  : level === 2 ? 'bg-amber-100 border-amber-300 text-amber-700'
   190→                  : 'bg-red-100 border-red-300 text-red-700'
   191→                : 'bg-gray-50 border-gray-200 text-gray-400'
   192→            )}
   193→          >
   194→            {RATING_LABELS[level].label}
   195→          </button>
   196→        ))}
   197→      </div>
   198→      {showLabel && displayRating > 0 && (
   199→        <span className={cn('text-xs font-medium ml-1', RATING_LABELS[displayRating]?.color || 'text-muted-foreground')}>
   200→          {RATING_LABELS[displayRating].label}
   201→        </span>
   202→      )}
   203→    </div>
   204→  );
   205→}
