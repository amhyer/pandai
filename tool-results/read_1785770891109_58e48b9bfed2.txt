   111→function getScoreColorClass(percentage: number): {
   112→  bg: string;
   113→  text: string;
   114→  bar: string;
   115→  border: string;
   116→  label: string;
   117→} {
   118→  if (percentage >= 85) {
   119→    return {
   120→      bg: 'bg-emerald-50',
   121→      text: 'text-emerald-700',
   122→      bar: 'bg-gradient-to-r from-emerald-400 to-emerald-500',
   123→      border: 'border-emerald-200',
   124→      label: 'Sangat Baik',
   125→    };
   126→  }
   127→  if (percentage >= 70) {
   128→    return {
   129→      bg: 'bg-amber-50',
   130→      text: 'text-amber-700',
   131→      bar: 'bg-gradient-to-r from-amber-400 to-amber-500',
   132→      border: 'border-amber-200',
   133→      label: 'Baik',
   134→    };
   135→  }
   136→  if (percentage >= 55) {
   137→    return {
   138→      bg: 'bg-orange-50',
   139→      text: 'text-orange-700',
   140→      bar: 'bg-gradient-to-r from-orange-400 to-orange-500',
   141→      border: 'border-orange-200',
   142→      label: 'Cukup',
   143→    };
   144→  }
   145→  return {
   146→    bg: 'bg-red-50',
   147→    text: 'text-red-700',
   148→    bar: 'bg-gradient-to-r from-red-400 to-red-500',
   149→    border: 'border-red-200',
   150→    label: 'Perlu Perbaikan',
   151→  };
   152→}

   154→function getStatusBadge(status: string) {
   155→  switch (status) {
   156→    case 'submitted':
   157→      return <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-emerald-200">Selesai</Badge>;
   158→    case 'in_progress':
   159→      return <Badge className="bg-blue-100 text-blue-700 hover:bg-blue-100 border-blue-200">Berlangsung</Badge>;
   160→    case 'graded':
   161→      return <Badge className="bg-violet-100 text-violet-700 hover:bg-violet-100 border-violet-200">Dinilai</Badge>;
   162→    default:
   163→      return <Badge variant="outline">{status}</Badge>;
   164→  }
   165→}
   166→
   167→function formatDuration(seconds: number): string {
   168→  if (seconds <= 0) return '0m';
   169→  const h = Math.floor(seconds / 3600);
   170→  const m = Math.floor((seconds % 3600) / 60);
   171→  if (h > 0) return `${h}j ${m}m`;
   172→  return `${m}m`;
   173→}
   174→
   175→function formatDate(dateStr: string): string {
   176→  const d = new Date(dateStr);
   177→  return d.toLocaleDateString('id-ID', {
   178→    day: 'numeric',
   179→    month: 'short',
   180→    year: 'numeric',
   181→  });
   182→}
   183→
   184→function formatDateTime(dateStr: string): string {
   185→  const d = new Date(dateStr);
   186→  return d.toLocaleDateString('id-ID', {
   187→    day: 'numeric',
   188→    month: 'short',
   189→    year: 'numeric',
   190→    hour: '2-digit',
   191→    minute: '2-digit',
   192→  });
   193→}

   215→
   216→export function SiswaNilaiView() {
   217→  const user = useAppStore((s) => s.user);
   218→  const [loading, setLoading] = useState(true);
   219→  const [attempts, setAttempts] = useState<Attempt[]>([]);
   220→  const [exams, setExams] = useState<ExamPackage[]>([]);
   221→  const [expandedSubject, setExpandedSubject] = useState<string | null>(null);
   222→  const [searchQuery, setSearchQuery] = useState('');
   223→  const [error, setError] = useState<string | null>(null);
   224→
   225→  const fetchData = useCallback(async () => {
   226→    if (!user?.id) return;
   227→    setLoading(true);
   228→    setError(null);
   229→    try {
   230→      const [attemptsRes, examsRes] = await Promise.all([
   231→        fetch(`/api/attempts?userId=${user.id}`),
   232→        fetch(`/api/exams${user.schoolId ? `?schoolId=${user.schoolId}` : ''}`),
   233→      ]);
   234→      if (!attemptsRes.ok || !examsRes.ok) throw new Error('Gagal mengambil data');
   235→      const attemptsData: Attempt[] = await attemptsRes.json();
   236→      const examsData: ExamPackage[] = await examsRes.json();
   237→      setAttempts(attemptsData);
   238→      setExams(examsData);
   239→    } catch (err) {
   240→      console.error(err);
   241→      setError('Gagal memuat data nilai. Coba refresh halaman.');
   242→      toast.error('Gagal memuat data nilai');
   243→    } finally {
   244→      setLoading(false);
   245→    }
   246→  }, [user?.id, user?.schoolId]);
   247→
   248→  useEffect(() => {
   249→    fetchData();
   250→  }, [fetchData]);
   251→
   252→  // Build a map of examPackageId → title
   253→  const examTitleMap = useMemo(() => {
   254→    const map = new Map<string, string>();
   255→    exams.forEach((e) => map.set(e.id, e.title));
   256→    return map;
   257→  }, [exams]);
   258→
   259→  // Only consider submitted attempts for grading
   260→  const completedAttempts = useMemo(
   261→    () => attempts.filter((a) => a.status === 'submitted' || a.status === 'graded'),
   262→    [attempts]
   263→  );
   264→
   265→  // Group by subject from answers
   266→  const subjectScores = useMemo((): SubjectScore[] => {
   267→    const subjectMap = new Map<string, {
   268→      subjectId: string;
   269→      subjectName: string;
   270→      subjectCode: string;
   271→      attempts: Attempt[];
   272→    }>();
   273→
   274→    completedAttempts.forEach((attempt) => {
   275→      if (!attempt.answers || attempt.answers.length === 0) return;
   276→      // Use the first answer's subject as representative
   277→      const firstSubject = attempt.answers[0]?.question?.subject;
   278→      if (!firstSubject) return;
   279→      const sid = firstSubject.id;
   280→      if (!subjectMap.has(sid)) {
   281→        subjectMap.set(sid, {
   282→          subjectId: sid,
   283→          subjectName: firstSubject.name,
   284→          subjectCode: firstSubject.code,
   285→          attempts: [],
   286→        });
   287→      }
   288→      subjectMap.get(sid)!.attempts.push(attempt);
   289→    });
   290→
   291→    return Array.from(subjectMap.values()).map((group) => {
   292→      const pcts = group.attempts.map((a) => a.percentage).sort((a, b) => b - a);
   293→      const avg = pcts.reduce((s, v) => s + v, 0) / pcts.length;
   294→      const latest = group.attempts.sort(
   295→        (a, b) => new Date(b.submittedAt || b.startedAt).getTime() - new Date(a.submittedAt || a.startedAt).getTime()
   296→      )[0];
   297→      const prev = group.attempts.length > 1
   298→        ? group.attempts.sort(
   299→            (a, b) => new Date(b.submittedAt || b.startedAt).getTime() - new Date(a.submittedAt || a.startedAt).getTime()
   300→          )[1]
   301→        : null;
   302→
   303→      let trend: 'up' | 'down' | 'stable' | 'new' = 'new';
   304→      if (prev) {
   305→        const diff = latest.percentage - prev.percentage;
   306→        if (diff > 2) trend = 'up';
   307→        else if (diff < -2) trend = 'down';
   308→        else trend = 'stable';
   309→      }
   310→
   311→      return {
   312→        subjectId: group.subjectId,
   313→        subjectName: group.subjectName,
   314→        subjectCode: group.subjectCode,
   315→        attempts: group.attempts,
   316→        avgPercentage: Math.round(avg * 10) / 10,
   317→        bestPercentage: pcts[0],
   318→        latestPercentage: latest.percentage,
   319→        trend,
   320→        totalAttempts: group.attempts.length,
   321→      };
   322→    }).sort((a, b) => b.avgPercentage - a.avgPercentage);
   323→  }, [completedAttempts]);
   324→
   325→  // Filter by search
   326→  const filteredSubjects = useMemo(() => {
   327→    if (!searchQuery.trim()) return subjectScores;
   328→    const q = searchQuery.toLowerCase();
   329→    return subjectScores.filter(
   330→      (s) =>
   331→        s.subjectName.toLowerCase().includes(q) ||
   332→        s.subjectCode.toLowerCase().includes(q)
   333→    );
   334→  }, [subjectScores, searchQuery]);
   335→
   336→  // Overall stats
   337→  const overallStats = useMemo(() => {
   338→    const allPcts = completedAttempts.map((a) => a.percentage);
   339→    const avg = allPcts.length > 0 ? Math.round(allPcts.reduce((s, v) => s + v, 0) / allPcts.length) : 0;
   340→    const highest = allPcts.length > 0 ? Math.round(Math.max(...allPcts)) : 0;
   341→    const lowest = allPcts.length > 0 ? Math.round(Math.min(...allPcts)) : 0;
   342→    return { avg, highest, lowest, total: completedAttempts.length };
   343→  }, [completedAttempts]);
   344→
   345→  return (
   346→    <div className="space-y-6">
   347→      {/* Header */}
   348→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   349→        <div className="flex items-center gap-3">
   350→          <div className="h-10 w-10 rounded-xl bg-gradient-to-br from-[#1F3864] to-[#2d4a7a] flex items-center justify-center shadow-sm">
   351→            <BarChart3 className="h-5 w-5 text-white" />
   352→          </div>
   353→          <div>
   354→            <h1 className="text-2xl font-bold tracking-tight">Nilai Saya</h1>
   355→            <p className="text-sm text-muted-foreground">
   356→              Pantau perkembangan skor dan lihat kemajuanmu
   357→            </p>
   358→          </div>
   359→        </div>
   360→        <Button
   361→          variant="outline"
   362→          size="sm"
   363→          className="transition-all duration-200 hover:shadow-sm active:scale-[0.98]"
   364→          onClick={() => {
   365→            fetchData();
   366→            toast.success('Data berhasil diperbarui');
   367→          }}
   368→        >
   369→          <Target className="mr-1.5 h-4 w-4" />
   370→          Refresh Data
   371→        </Button>
   372→      </div>
   373→
   374→      {/* Error State */}
   375→      {error && (
   376→        <Card className={`${cardBase} border-red-200 bg-red-50`}>
   377→          <CardContent className="p-4 flex items-center gap-3">
   378→            <AlertCircle className="h-5 w-5 text-red-500 shrink-0" />
   379→            <p className="text-sm text-red-700">{error}</p>
   380→            <Button
   381→              variant="outline"
   382→              size="sm"
   383→              className="ml-auto shrink-0 transition-all duration-200 hover:shadow-sm active:scale-[0.98]"
   384→              onClick={fetchData}
   385→            >
   386→              Coba Lagi
   387→            </Button>
   388→          </CardContent>
   389→        </Card>
   390→      )}
   391→
   392→      {/* Overview Stats */}
   393→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   394→        {loading ? (
   395→          Array.from({ length: 4 }).map((_, i) => (
   396→            <Card key={i} className={cardBase}>
   397→              <CardContent className="p-4 space-y-2">
   398→                <Skeleton className="h-4 w-24" />
   399→                <Skeleton className="h-8 w-16" />
   400→                <Skeleton className="h-3 w-20" />
   401→              </CardContent>
   402→            </Card>
   403→          ))
   404→        ) : (
   405→          <>
   406→            <Card className={cardBase}>
   407→              <CardContent className="p-4">
   408→                <div className="flex items-center gap-2 mb-1">
   409→                  <div className="h-7 w-7 rounded-lg bg-emerald-100 flex items-center justify-center">
   410→                    <TrendingUp className="h-3.5 w-3.5 text-emerald-600" />
   411→                  </div>
   412→                  <p className="text-xs text-muted-foreground font-medium">Rata-rata</p>
   413→                </div>
   414→                <p className="text-3xl font-bold text-[#1F3864]">{overallStats.avg}</p>
   415→                <p className="text-xs text-muted-foreground mt-0.5">dari 100</p>
   416→              </CardContent>
   417→            </Card>
   418→            <Card className={cardBase}>
   419→              <CardContent className="p-4">
   420→                <div className="flex items-center gap-2 mb-1">
   421→                  <div className="h-7 w-7 rounded-lg bg-amber-100 flex items-center justify-center">
   422→                    <Trophy className="h-3.5 w-3.5 text-amber-600" />
   423→                  </div>
   424→                  <p className="text-xs text-muted-foreground font-medium">Tertinggi</p>
   425→                </div>
   426→                <p className="text-3xl font-bold text-amber-600">{overallStats.highest}</p>
   427→                <p className="text-xs text-muted-foreground mt-0.5">skor terbaik</p>
   428→              </CardContent>
   429→            </Card>
   430→            <Card className={cardBase}>
   431→              <CardContent className="p-4">
   432→                <div className="flex items-center gap-2 mb-1">
   433→                  <div className="h-7 w-7 rounded-lg bg-red-100 flex items-center justify-center">
   434→                    <AlertCircle className="h-3.5 w-3.5 text-red-500" />
   435→                  </div>
   436→                  <p className="text-xs text-muted-foreground font-medium">Terendah</p>
   437→                </div>
   438→                <p className="text-3xl font-bold text-red-500">{overallStats.lowest}</p>
   439→                <p className="text-xs text-muted-foreground mt-0.5">perlu ditingkatkan</p>
   440→              </CardContent>
   441→            </Card>
   442→            <Card className={cardBase}>
   443→              <CardContent className="p-4">
   444→                <div className="flex items-center gap-2 mb-1">
   445→                  <div className="h-7 w-7 rounded-lg bg-[#1F3864]/10 flex items-center justify-center">
   446→                    <ClipboardList className="h-3.5 w-3.5 text-[#1F3864]" />
   447→                  </div>
   448→                  <p className="text-xs text-muted-foreground font-medium">Total Ujian</p>
   449→                </div>
   450→                <p className="text-3xl font-bold">{overallStats.total}</p>
   451→                <p className="text-xs text-muted-foreground mt-0.5">diselesaikan</p>
   452→              </CardContent>
   453→            </Card>
   454→          </>
   455→        )}
   456→      </div>
   457→
   458→      {/* Subject Cards */}
   459→      <Card className={cardBase}>
   460→        <CardHeader className="pb-4">
   461→          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   462→            <div>
   463→              <CardTitle className="text-lg">Nilai Per Mata Pelajaran</CardTitle>
   464→              <CardDescription className="text-sm mt-1">
   465→                Klik kartu untuk melihat detail riwayat ujian
   466→              </CardDescription>
   467→            </div>
   468→            <div className="relative w-full sm:w-64">
   469→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   470→              <Input
   471→                placeholder="Cari mata pelajaran..."
   472→                value={searchQuery}
   473→                onChange={(e) => setSearchQuery(e.target.value)}
   474→                className="pl-9 rounded-lg focus:ring-2 focus:ring-[#1F3864]/20 focus:border-[#1F3864]/40"
   475→              />
   476→            </div>
   477→          </div>
   478→        </CardHeader>
   479→        <CardContent>
   480→          {loading ? (
   481→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   482→              {Array.from({ length: 4 }).map((_, i) => (
   483→                <Skeleton key={i} className="h-32 w-full rounded-xl" />
   484→              ))}
   485→            </div>
   486→          ) : filteredSubjects.length > 0 ? (
   487→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   488→              {filteredSubjects.map((subject) => {
   489→                const colors = getScoreColorClass(subject.avgPercentage);
   490→                return (
   491→                  <React.Fragment key={subject.subjectId}>
   492→                    <Card
   493→                      className={`${cardBase} cursor-pointer border ${
   494→                        expandedSubject === subject.subjectId ? 'ring-2 ring-[#1F3864]/20' : ''
   495→                      }`}
   496→                      onClick={() =>
   497→                        setExpandedSubject(expandedSubject === subject.subjectId ? null : subject.subjectId)
   498→                      }
   499→                    >
   500→                      <CardContent className="p-4">
   501→                        <div className="flex items-start justify-between mb-3">
   502→                          <div className="flex items-center gap-2.5">
   503→                            <div className={`h-9 w-9 rounded-lg ${colors.bg} flex items-center justify-center`}>
   504→                              <BookOpen className={`h-4.5 w-4.5 ${colors.text}`} />
   505→                            </div>
   506→                            <div>
   507→                              <h3 className="font-semibold text-sm">{subject.subjectName}</h3>
   508→                              <p className="text-xs text-muted-foreground">{subject.subjectCode}</p>
   509→                            </div>
   510→                          </div>
   511→                          <div className="flex items-center gap-1.5">
   512→                            <TrendIcon trend={subject.trend} />
   513→                            <span className={`text-2xl font-bold ${colors.text}`}>
   514→                              {Math.round(subject.avgPercentage)}
   515→                            </span>
   516→                          </div>
   517→                        </div>
   518→
   519→                        {/* Progress Bar */}
   520→                        <div className="mb-2">
   521→                          <div className="h-2.5 w-full bg-muted rounded-full overflow-hidden">
   522→                            <div
   523→                              className={`h-full rounded-full transition-all duration-700 ease-out ${colors.bar}`}
   524→                              style={{ width: `${Math.max(subject.avgPercentage, 2)}%` }}
   525→                            />
   526→                          </div>
   527→                        </div>
   528→
   529→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   530→                          <div className="flex items-center gap-3">
   531→                            <span>{subject.totalAttempts}x dikerjakan</span>
   532→                            <span className={`${colors.text} font-medium`}>{colors.label}</span>
   533→                          </div>
   534→                          <span className="font-medium">
   535→                            Terbaik: <span className={colors.text}>{Math.round(subject.bestPercentage)}</span>
   536→                          </span>
   537→                        </div>
   538→                      </CardContent>
   539→                    </Card>
   540→
   541→                    {/* Expanded detail for this subject */}
   542→                    {expandedSubject === subject.subjectId && (
   543→                      <Card className={`${cardBase} border-0 shadow-none -mt-2 -mb-2`}>
