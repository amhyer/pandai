   421→   421→// Sidebar Navigation — grouped by section (SIMANTAP pattern)
   422→   422→// ═══════════════════════════════════════════════════════════════════════
   423→   423→
   424→   424→interface SidebarNavProps {
   425→   425→  sections: NavSection[];
   426→   426→  currentView: ViewType;
   427→   427→  onNavigate: (view: ViewType) => void;

   430→   430→
   431→   431→function SidebarNav({ sections, currentView, onNavigate, onLogout }: SidebarNavProps) {
   432→   432→  const [collapsed, setCollapsed] = React.useState<Record<string, boolean>>({});
   433→   433→
   434→   434→  const toggleSection = (section: string) => {
   435→   435→    setCollapsed((prev) => ({ ...prev, [section]: !prev[section] }));
   436→   436→  };
   437→   437→
   438→   438→  return (
   439→   439→    <div className="flex h-full flex-col min-h-0">
   440→   440→      {/* Scrollable nav area */}
   441→   441→      <ScrollArea className="flex-1 px-3 py-2 min-h-0">
   442→   442→        <nav className="flex flex-col gap-1 pb-2" role="navigation" aria-label="Main navigation">
   443→   443→          {sections.map((group) => (
   444→   444→            <React.Fragment key={group.section}>
   445→   445→              {/* Section label — clickable to collapse */}
   446→   446→              <button
   447→   447→                onClick={() => toggleSection(group.section)}
   448→   448→                className="mt-4 mb-1 flex w-full items-center justify-between px-3 first:mt-0 group/section"
   449→   449→                aria-expanded={!collapsed[group.section]}
   450→   450→              >
   451→   451→                <span className="text-[11px] font-semibold uppercase tracking-wider text-white/40">
   452→   452→                  {group.section}
   453→   453→                </span>
   454→   454→                <ChevronRight
   455→   455→                  className={cn(
   456→   456→                    'h-3 w-3 text-white/30 transition-transform duration-200',
   457→   457→                    !collapsed[group.section] && 'rotate-90'
   458→   458→                  )}
   459→   459→                />
   460→   460→              </button>
   461→   461→
   462→   462→              {/* Items — animated collapse */}
   463→   463→              <div
   464→   464→                className={cn(
   465→   465→                  'overflow-hidden transition-all duration-200 ease-in-out',
   466→   466→                  collapsed[group.section] ? 'max-h-0 opacity-0' : 'max-h-[600px] opacity-100'
   467→   467→                )}
   468→   468→              >
   469→   469→                {group.items.map((item) => {
   470→   470→                  const Icon = item.icon;
   471→   471→                  const isActive = currentView === item.view;
   472→   472→                  return (
   473→   473→                    <button
   474→   474→                      key={item.view}
   475→   475→                      onClick={() => onNavigate(item.view)}
   476→   476→                      className={cn(
   477→   477→                        'flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium transition-all duration-150 text-left',
   478→   478→                        'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/30',
   479→   479→                        'hover:scale-[1.02] active:scale-[0.98]',
