   224→   224→function generateMockAttendance(year: number, month: number): AttendanceDay[] {
   225→   225→  const days: AttendanceDay[] = [];
   226→   226→  const daysInMonth = new Date(year, month + 1, 0).getDate();
   227→   227→  const statuses: Array<'hadir' | 'izin' | 'sakit' | 'alpa'> = ['hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'izin', 'sakit', 'alpa'];
   228→   228→  let statusIdx = 0;
   229→   229→
   230→   230→  for (let d = 1; d <= daysInMonth; d++) {
   231→   231→    const date = new Date(year, month, d);
   232→   232→    const dow = date.getDay();
   233→   233→    if (dow === 0 || dow === 6) {
   234→   234→      days.push({ day: d, status: 'weekend' });
   235→   235→    } else {
   236→   236→      const s = statuses[statusIdx % statuses.length];
   237→   237→      days.push({ day: d, status: s });
   238→   238→      statusIdx++;
   239→   239→    }
   240→   240→  }
   241→   241→  return days;
   242→   242→}
