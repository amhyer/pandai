   184→function formatDate(d: string) {
   185→  if (!d) return '-';
   186→  const date = new Date(d);
   187→  return date.toLocaleDateString('id-ID', {
   188→    day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   189→  });
   190→}
   191→
   192→function toLocalDatetime(d: string) {
   193→  if (!d) return '';
   194→  const date = new Date(d);
   195→  const pad = (n: number) => String(n).padStart(2, '0');
   196→  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}T${pad(date.getHours())}:${pad(date.getMinutes())}`;
   197→}
