   395→function PerformanceBar({ value, max = 100 }: { value: number; max?: number }) {
   396→  const pct = Math.min((value / max) * 100, 100);
   397→  const color = pct >= 75 ? 'bg-emerald-500' : pct >= 60 ? 'bg-amber-500' : 'bg-red-500';
   398→  return (
   399→    <div className="h-2.5 w-full overflow-hidden rounded-full bg-muted">
   400→      <div
   401→        className={cn('h-full rounded-full transition-all', color)}
   402→        style={{ width: pct + '%' }}
   403→      />
   404→    </div>
   405→  );
   406→}
