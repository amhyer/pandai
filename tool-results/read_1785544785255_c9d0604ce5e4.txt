     1→'use client';
     2→
     3→import { useState, useEffect } from 'react';
     4→import { useAppStore } from '@/store/use-store';
     5→import { Button } from '@/components/ui/button';
     6→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
     7→import { Input } from '@/components/ui/input';
     8→import { Label } from '@/components/ui/label';
     9→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
    10→import { Badge } from '@/components/ui/badge';
    11→import { toast } from 'sonner';
    12→import {
    13→  Mail, Lock, User, School, Eye, EyeOff, GraduationCap,
    14→  Search, ShieldCheck, MapPin, Award, Calendar, BookOpen, Building2, Phone, Loader2, Sparkles
    15→} from 'lucide-react';
    16→
    17→type RegisterRole = 'SISWA' | 'GURU' | 'ADMIN_SCHOOL';
    18→
    19→interface DapodikSchool {
    20→  npsn: string;
    21→  name: string;
    22→  address: string;
    23→  province: string;
    24→  city: string;
    25→  district: string;
    26→  principalName: string;
    27→  accreditation: string;
    28→  schoolType: string;
    29→  established: string;
    30→  curriculum: string;
    31→  phone: string;
    32→  email: string;
    33→  emailDomain: string;
    34→  source: string;
    35→}
    36→
    37→export function RegisterForm() {
    38→  const { setUser, navigateTo } = useAppStore();
    39→  const [name, setName] = useState('');
    40→  const [email, setEmail] = useState('');
    41→  const [password, setPassword] = useState('');
    42→  const [confirmPassword, setConfirmPassword] = useState('');
    43→  const [role, setRole] = useState<RegisterRole>('SISWA');
    44→  const [schoolCode, setSchoolCode] = useState('');
    45→  const [showPassword, setShowPassword] = useState(false);
    46→  const [showConfirmPassword, setShowConfirmPassword] = useState(false);
    47→  const [isLoading, setIsLoading] = useState(false);
    48→
    49→  // Dapodik / Admin Sekolah state
    50→  const [npsnInput, setNpsnInput] = useState('');
    51→  const [isSearching, setIsSearching] = useState(false);
    52→  const [dapodikSchool, setDapodikSchool] = useState<DapodikSchool | null>(null);
    53→  const [dapodikVerified, setDapodikVerified] = useState(false);
    54→
    55→  // Track whether fields were auto-filled from Dapodik
    56→  const [nameAutoFilled, setNameAutoFilled] = useState(false);
    57→  const [emailAutoFilled, setEmailAutoFilled] = useState(false);
    58→
    59→  const isSchoolAdmin = role === 'ADMIN_SCHOOL';
    60→
    61→  // Auto-fill nama kepala sekolah & email sekolah when Dapodik data is verified
    62→  useEffect(() => {
    63→    if (dapodikVerified && dapodikSchool) {
    64→      // Auto-fill Nama Lengkap with principal name
    65→      if (dapodikSchool.principalName) {
    66→        setName(dapodikSchool.principalName);
    67→        setNameAutoFilled(true);
    68→      }
    69→      // Auto-fill Email: prefer direct email, otherwise construct from emailDomain
    70→      const schoolEmail = dapodikSchool.email || '';
    71→      if (schoolEmail) {
    72→        setEmail(schoolEmail);
    73→        setEmailAutoFilled(true);
    74→      } else if (dapodikSchool.emailDomain) {
    75→        // Construct email from email domain (e.g., info@sman1makassar.sch.id)
    76→        const constructedEmail = `info@${dapodikSchool.emailDomain}`;
    77→        setEmail(constructedEmail);
    78→        setEmailAutoFilled(true);
    79→      } else {
    80→        setEmail('');
    81→        setEmailAutoFilled(false);
    82→      }
    83→    } else {
    84→      // Reset auto-fill flags when verification is cleared
    85→      setNameAutoFilled(false);
    86→      setEmailAutoFilled(false);
    87→    }
    88→  }, [dapodikVerified, dapodikSchool]);
    89→
    90→  const validateForm = (): boolean => {
    91→    if (!name.trim()) {
    92→      toast.error('Harap isi nama lengkap');
    93→      return false;
    94→    }
    95→    if (!email.trim()) {
    96→      toast.error('Harap isi alamat email');
    97→      return false;
    98→    }
    99→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.trim())) {
   100→      toast.error('Format email tidak valid');
   101→      return false;
   102→    }
   103→    if (password.length < 6) {
   104→      toast.error('Password minimal 6 karakter');
   105→      return false;
   106→    }
   107→    if (password !== confirmPassword) {
   108→      toast.error('Konfirmasi password tidak cocok');
   109→      return false;
   110→    }
   111→    if (isSchoolAdmin) {
   112→      if (!dapodikVerified || !dapodikSchool) {
   113→        toast.error('Harap verifikasi data sekolah melalui Dapodik terlebih dahulu');
   114→        return false;
   115→      }
   116→    } else {
   117→      if (!schoolCode.trim()) {
   118→        toast.error('Harap isi kode sekolah');
   119→        return false;
   120→      }
   121→    }
   122→    return true;
   123→  };
   124→
   125→  const handleSearchNpsn = async () => {
   126→    const q = npsnInput.trim();
   127→    if (!q) {
   128→      toast.error('Masukkan NPSN atau nama sekolah');
   129→      return;
   130→    }
   131→
   132→    setIsSearching(true);
   133→    try {
   134→      const res = await fetch(`/api/schools/lookup?q=${encodeURIComponent(q)}`);
   135→      const data = await res.json();
   136→
   137→      if (!res.ok || !Array.isArray(data) || data.length === 0) {
   138→        toast.error('Sekolah tidak ditemukan dalam database Dapodik');
   139→        setDapodikSchool(null);
   140→        setDapodikVerified(false);
   141→        return;
   142→      }
   143→
   144→      // Pick the first result
   145→      const school = data[0];
   146→      setDapodikSchool(school);
   147→      setDapodikVerified(true);
   148→
   149→      if (school.source === 'dapodik-live') {
   150→        toast.success(`Data sekolah "${school.name}" ditemukan dari Dapodik`);
   151→      } else {
   152→        toast.success(`Data sekolah "${school.name}" ditemukan`);
   153→      }
   154→    } catch {
   155→      toast.error('Gagal mencari data sekolah. Coba lagi.');
   156→    } finally {
   157→      setIsSearching(false);
   158→    }
   159→  };
   160→
   161→  const handleSubmit = async (e: React.FormEvent) => {
   162→    e.preventDefault();
   163→    if (!validateForm()) return;
   164→
   165→    setIsLoading(true);
   166→    try {
   167→      if (isSchoolAdmin && dapodikSchool) {
   168→        // Admin Sekolah registration via /api/auth/register-school
   169→        const res = await fetch('/api/auth/register-school', {
   170→          method: 'POST',
   171→          headers: { 'Content-Type': 'application/json' },
   172→          body: JSON.stringify({
   173→            name: name.trim(),
   174→            email: email.trim(),
   175→            password,
   176→            schoolData: {
   177→              npsn: dapodikSchool.npsn,
   178→              name: dapodikSchool.name,
   179→              address: dapodikSchool.address,
   180→              province: dapodikSchool.province,
   181→              city: dapodikSchool.city,
   182→              district: dapodikSchool.district,
   183→              principalName: dapodikSchool.principalName,
   184→              accreditation: dapodikSchool.accreditation,
   185→              schoolType: dapodikSchool.schoolType,
   186→              phone: dapodikSchool.phone,
   187→              email: dapodikSchool.email || dapodikSchool.emailDomain || '',
   188→              curriculum: dapodikSchool.curriculum,
   189→            },
   190→          }),
   191→        });
   192→
   193→        const data = await res.json();
   194→
   195→        if (!res.ok) {
   196→          toast.error(data.error || 'Registrasi gagal. Silakan coba lagi.');
   197→          return;
   198→        }
   199→
   200→        setUser(data);
   201→        toast.success('Registrasi berhasil! Selamat datang, Admin Sekolah.');
   202→        navigateTo('dashboard');
   203→      } else {
   204→        // Siswa / Guru registration via /api/auth/register
   205→        const res = await fetch('/api/auth/register', {
   206→          method: 'POST',
   207→          headers: { 'Content-Type': 'application/json' },
   208→          body: JSON.stringify({
   209→            name: name.trim(),
   210→            email: email.trim(),
   211→            password,
   212→            role,
   213→            schoolCode: schoolCode.trim(),
   214→          }),
   215→        });
   216→
   217→        const data = await res.json();
   218→
   219→        if (!res.ok) {
   220→          toast.error(data.error || 'Registrasi gagal. Silakan coba lagi.');
   221→          return;
   222→        }
   223→
   224→        setUser(data);
   225→        toast.success('Registrasi berhasil! Selamat datang.');
   226→        navigateTo('dashboard');
   227→      }
   228→    } catch {
   229→      toast.error('Terjadi kesalahan jaringan. Silakan coba lagi.');
   230→    } finally {
   231→      setIsLoading(false);
   232→    }
   233→  };
   234→
   235→  const handleRoleChange = (val: string) => {
   236→    setRole(val as RegisterRole);
   237→    if (val === 'ADMIN_SCHOOL') {
   238→      setDapodikSchool(null);
   239→      setDapodikVerified(false);
   240→      setName('');
   241→      setEmail('');
   242→    }
   243→  };
   244→
   245→  return (
   246→    <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-slate-900 via-blue-950 to-slate-900 p-4">
   247→      <Card className="w-full max-w-md shadow-2xl border-0">
   248→        <CardHeader className="text-center space-y-2 pb-2">
   249→          <div className="flex items-center justify-center gap-2 mb-2">
   250→            <GraduationCap className="h-8 w-8 text-amber-500" />
   251→            <span className="text-3xl font-bold tracking-tight bg-gradient-to-r from-amber-400 to-amber-600 bg-clip-text text-transparent">
   252→              PANDAI
   253→            </span>
   254→          </div>
   255→          <CardTitle className="text-xl font-semibold text-slate-800">
   256→            Daftar Akun Baru
   257→          </CardTitle>
   258→          <CardDescription>
   259→            Bergabung dengan platform persiapan TKA
   260→          </CardDescription>
   261→        </CardHeader>
   262→
   263→        <CardContent className="space-y-5 pt-2">
   264→          <form onSubmit={handleSubmit} className="space-y-4">
   265→            {/* Role Selector */}
   266→            <div className="space-y-3">
   267→              <Label className="text-sm font-medium text-slate-700">
   268→                Daftar Sebagai
   269→              </Label>
   270→              <RadioGroup
   271→                value={role}
   272→                onValueChange={handleRoleChange}
   273→                className="grid grid-cols-3 gap-2"
   274→                disabled={isLoading}
   275→              >
   276→                <label
   277→                  htmlFor="role-siswa"
   278→                  className={`flex items-center justify-center gap-1.5 px-3 py-3 rounded-lg border-2 cursor-pointer transition-all duration-200 ${
   279→                    role === 'SISWA'
   280→                      ? 'border-amber-500 bg-amber-50 text-amber-700'
   281→                      : 'border-slate-200 hover:border-slate-300 text-slate-600'
   282→                  }`}
   283→                >
   284→                  <RadioGroupItem value="SISWA" id="role-siswa" className="sr-only" />
   285→                  <GraduationCap className="h-4 w-4 shrink-0" />
   286→                  <span className="text-sm font-medium">Siswa</span>
   287→                </label>
   288→                <label
   289→                  htmlFor="role-guru"
   290→                  className={`flex items-center justify-center gap-1.5 px-3 py-3 rounded-lg border-2 cursor-pointer transition-all duration-200 ${
   291→                    role === 'GURU'
   292→                      ? 'border-amber-500 bg-amber-50 text-amber-700'
   293→                      : 'border-slate-200 hover:border-slate-300 text-slate-600'
   294→                  }`}
   295→                >
   296→                  <RadioGroupItem value="GURU" id="role-guru" className="sr-only" />
   297→                  <BookOpen className="h-4 w-4 shrink-0" />
   298→                  <span className="text-sm font-medium">Guru</span>
   299→                </label>
   300→                <label
   301→                  htmlFor="role-admin"
   302→                  className={`flex items-center justify-center gap-1.5 px-3 py-3 rounded-lg border-2 cursor-pointer transition-all duration-200 ${
   303→                    role === 'ADMIN_SCHOOL'
   304→                      ? 'border-emerald-500 bg-emerald-50 text-emerald-700'
   305→                      : 'border-slate-200 hover:border-slate-300 text-slate-600'
   306→                  }`}
   307→                >
   308→                  <RadioGroupItem value="ADMIN_SCHOOL" id="role-admin" className="sr-only" />
   309→                  <Building2 className="h-4 w-4 shrink-0" />
   310→                  <span className="text-sm font-medium">Admin</span>
   311→                </label>
   312→              </RadioGroup>
   313→              {isSchoolAdmin && (
   314→                <p className="text-xs text-emerald-600 font-medium flex items-center gap-1">
   315→                  <ShieldCheck className="h-3.5 w-3.5" />
   316→                  Verifikasi data sekolah melalui database Dapodik
   317→                </p>
   318→              )}
   319→            </div>
   320→
   321→            {/* ===== ADMIN_SCHOOL: Dapodik Verification Section ===== */}
   322→            {isSchoolAdmin && (
   323→              <div className="space-y-3">
   324→                <Label className="text-sm font-medium text-slate-700">
   325→                  Verifikasi Data Sekolah
   326→                </Label>
   327→                <div className="flex gap-2">
   328→                  <div className="relative flex-1">
   329→                    <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
   330→                    <Input
   331→                      type="text"
   332→                      placeholder="Masukkan NPSN (8 digit)"
   333→                      value={npsnInput}
   334→                      onChange={(e) => {
   335→                        setNpsnInput(e.target.value);
   336→                        if (dapodikVerified) {
   337→                          setDapodikVerified(false);
   338→                          setDapodikSchool(null);
   339→                        }
   340→                      }}
   341→                      className="pl-10 h-11 border-slate-300 focus-visible:ring-emerald-500/30 focus-visible:border-emerald-500"
   342→                      disabled={isLoading || isSearching}
   343→                      onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); handleSearchNpsn(); } }}
   344→                    />
   345→                  </div>
   346→                  <Button
   347→                    type="button"
   348→                    onClick={handleSearchNpsn}
   349→                    disabled={isLoading || isSearching || !npsnInput.trim()}
   350→                    className="h-11 px-5 bg-emerald-600 hover:bg-emerald-700 text-white font-medium shrink-0 transition-colors"
   351→                  >
   352→                    {isSearching ? (
   353→                      <Loader2 className="h-4 w-4 animate-spin" />
   354→                    ) : (
   355→                      <Search className="h-4 w-4" />
   356→                    )}
   357→                    <span className="ml-1.5">Cari</span>
   358→                  </Button>
   359→                </div>
   360→
   361→                {/* Dapodik School Card */}
   362→                {dapodikVerified && dapodikSchool && (
   363→                  <div className="border-l-4 border-emerald-500 bg-emerald-50/60 rounded-r-lg p-4 space-y-3 animate-in fade-in slide-in-from-bottom-2 duration-300">
   364→                    <div className="flex items-center justify-between">
   365→                      <div className="flex items-center gap-2">
   366→                        <ShieldCheck className="h-5 w-5 text-emerald-600" />
   367→                        <span className="text-sm font-semibold text-emerald-700">Data Terverifikasi Dapodik</span>
   368→                      </div>
   369→                      <Badge className="bg-emerald-600 text-white hover:bg-emerald-700 text-xs font-medium">
   370→                        {dapodikSchool.npsn}
   371→                      </Badge>
   372→                    </div>
   373→
   374→                    <div className="space-y-2.5 text-sm">
   375→                      <div className="flex items-start gap-2">
   376→                        <School className="h-4 w-4 text-emerald-600 mt-0.5 shrink-0" />
   377→                        <div className="min-w-0">
   378→                          <p className="font-semibold text-slate-800">{dapodikSchool.name}</p>
   379→                          <p className="text-slate-500 text-xs">{dapodikSchool.address}</p>
   380→                        </div>
   381→                      </div>
   382→
   383→                      <div className="grid grid-cols-2 gap-x-4 gap-y-1.5 pl-6">
   384→                        <div className="flex items-center gap-1.5 text-slate-600">
   385→                          <MapPin className="h-3.5 w-3.5 text-emerald-500 shrink-0" />
   386→                          <span className="text-xs truncate">{dapodikSchool.district}, {dapodikSchool.city}</span>
   387→                        </div>
   388→                        <div className="flex items-center gap-1.5 text-slate-600">
   389→                          <MapPin className="h-3.5 w-3.5 text-emerald-500 shrink-0" />
   390→                          <span className="text-xs truncate">{dapodikSchool.province}</span>
   391→                        </div>
   392→                      </div>
   393→
   394→                      <div className="flex flex-wrap items-center gap-3 pl-6">
   395→                        <Badge variant="outline" className="border-emerald-300 text-emerald-700 text-xs font-medium">
   396→                          {dapodikSchool.schoolType}
   397→                        </Badge>
   398→                        <div className="flex items-center gap-1 text-xs text-slate-600">
   399→                          <Award className="h-3.5 w-3.5 text-emerald-500" />
   400→                          <span>Akreditasi <strong>{dapodikSchool.accreditation || '-'}</strong></span>
   401→                        </div>
   402→                        <div className="flex items-center gap-1 text-xs text-slate-600">
   403→                          <Calendar className="h-3.5 w-3.5 text-emerald-500" />
   404→                          <span>Berdiri {dapodikSchool.established || '-'}</span>
   405→                        </div>
   406→                      </div>
   407→
   408→                      <div className="flex flex-wrap items-center gap-4 pl-6 text-xs text-slate-500">
   409→                        <div className="flex items-center gap-1">
   410→                          <User className="h-3.5 w-3.5 text-emerald-500" />
   411→                          <span>KS: {dapodikSchool.principalName || '-'}</span>
   412→                        </div>
   413→                        <div className="flex items-center gap-1">
   414→                          <Phone className="h-3.5 w-3.5 text-emerald-500" />
   415→                          <span>{dapodikSchool.phone || '-'}</span>
   416→                        </div>
   417→                        {dapodikSchool.email && (
   418→                          <div className="flex items-center gap-1">
   419→                            <Mail className="h-3.5 w-3.5 text-emerald-500" />
   420→                            <span>{dapodikSchool.email}</span>
   421→                          </div>
   422→                        )}
   423→                      </div>
   424→                    </div>
   425→
   426→                    {/* Auto-fill indicator */}
   427→                    <div className="flex items-center gap-1.5 pt-1 border-t border-emerald-200/60">
   428→                      <Sparkles className="h-3.5 w-3.5 text-emerald-600" />
   429→                      <span className="text-xs text-emerald-600 font-medium">
   430→                        Nama & email telah diisi otomatis dari data Dapodik
   431→                      </span>
   432→                    </div>
   433→                  </div>
   434→                )}
   435→              </div>
   436→            )}
   437→
   438→            {/* Name */}
   439→            <div className="space-y-2">
   440→              <div className="flex items-center justify-between">
   441→                <Label htmlFor="name" className="text-sm font-medium text-slate-700">
   442→                  Nama Lengkap
   443→                </Label>
   444→                {nameAutoFilled && (
   445→                  <span className="text-[11px] text-emerald-600 font-medium flex items-center gap-0.5">
   446→                    <Sparkles className="h-3 w-3" />
   447→                    Auto-terisi (Kepala Sekolah)
   448→                  </span>
   449→                )}
   450→              </div>
   451→              <div className="relative">
   452→                <User className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
   453→                <Input
   454→                  id="name"
   455→                  type="text"
   456→                  placeholder={isSchoolAdmin ? 'Nama Kepala Sekolah / Admin' : 'Masukkan nama lengkap'}
   457→                  value={name}
   458→                  onChange={(e) => {
   459→                    setName(e.target.value);
   460→                    setNameAutoFilled(false); // Clear auto-fill flag on manual edit
   461→                  }}
   462→                  className={`pl-10 h-11 border-slate-300 focus-visible:ring-amber-500/30 focus-visible:border-amber-500 ${
   463→                    nameAutoFilled ? 'bg-emerald-50/50 border-emerald-300' : ''
   464→                  }`}
   465→                  disabled={isLoading}
   466→                />
   467→              </div>
   468→            </div>
   469→
   470→            {/* Email */}
   471→            <div className="space-y-2">
   472→              <div className="flex items-center justify-between">
   473→                <Label htmlFor="reg-email" className="text-sm font-medium text-slate-700">
   474→                  Email
   475→                </Label>
   476→                {emailAutoFilled && (
   477→                  <span className="text-[11px] text-emerald-600 font-medium flex items-center gap-0.5">
   478→                    <Sparkles className="h-3 w-3" />
   479→                    Auto-terisi (Email Sekolah)
   480→                  </span>
   481→                )}
   482→              </div>
   483→              <div className="relative">
   484→                <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
   485→                <Input
   486→                  id="reg-email"
   487→                  type="email"
   488→                  placeholder={isSchoolAdmin && dapodikSchool ? `nama@${dapodikSchool.emailDomain}` : 'nama@sekolah.sch.id'}
   489→                  value={email}
   490→                  onChange={(e) => {
   491→                    setEmail(e.target.value);
   492→                    setEmailAutoFilled(false); // Clear auto-fill flag on manual edit
   493→                  }}
   494→                  className={`pl-10 h-11 border-slate-300 focus-visible:ring-amber-500/30 focus-visible:border-amber-500 ${
   495→                    emailAutoFilled ? 'bg-emerald-50/50 border-emerald-300' : ''
   496→                  }`}
   497→                  disabled={isLoading}
   498→                />
   499→              </div>
   500→            </div>
   501→
   502→            {/* School Code (only for Siswa / Guru) */}
   503→            {!isSchoolAdmin && (
   504→              <div className="space-y-2">
   505→                <Label htmlFor="school-code" className="text-sm font-medium text-slate-700">
   506→                  Kode Sekolah
   507→                </Label>
   508→                <div className="relative">
   509→                  <School className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
   510→                  <Input
   511→                    id="school-code"
   512→                    type="text"
   513→                    placeholder="Masukkan kode sekolah"
