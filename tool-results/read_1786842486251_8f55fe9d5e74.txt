   197→   197→function generateMockAttendance(year: number, month: number): AttendanceDay[] {
   198→   198→  const days: AttendanceDay[] = [];
   199→   199→  const daysInMonth = new Date(year, month + 1, 0).getDate();
   200→   200→  const statuses: Array<'hadir' | 'izin' | 'sakit' | 'alpa'> = ['hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'izin', 'sakit', 'alpa'];
   201→   201→  const notes: Record<string, string> = { izin: 'Keperluan keluarga', sakit: 'Demam', alpa: 'Tanpa keterangan' };
   202→   202→  let statusIdx = 0;
   203→   203→  for (let d = 1; d <= daysInMonth; d++) {
   204→   204→    const date = new Date(year, month, d);
   205→   205→    const dow = date.getDay();
   206→   206→    if (dow === 0 || dow === 6) {
   207→   207→      days.push({ day: d, status: 'weekend', date: `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}` });
   208→   208→    } else {
   209→   209→      const s = statuses[statusIdx % statuses.length];
   210→   210→      days.push({ day: d, status: s, date: `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`, note: notes[s] || '' });
   211→   211→      statusIdx++;
   212→   212→    }
   213→   213→  }
   214→   214→  return days;
   215→   215→}

   661→   661→function getCountdown(dueDate: string): string {
   662→   662→  const now = new Date();
   663→   663→  const due = new Date(dueDate);
   664→   664→  const diffMs = due.getTime() - now.getTime();
   665→   665→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   666→   666→  if (diffDays < 0) return `Terlambat ${Math.abs(diffDays)} hari`;
   667→   667→  if (diffDays === 0) return 'Hari ini';
   668→   668→  if (diffDays === 1) return 'Besok';
   669→   669→  if (diffDays <= 7) return `${diffDays} hari lagi`;
   670→   670→  return `${Math.ceil(diffDays / 7)} minggu lagi`;
   671→   671→}
   672→   672→
   673→   673→function getDueUrgency(dueDate: string): 'overdue' | 'today' | 'soon' | 'future' {
   674→   674→  const now = new Date();
   675→   675→  const due = new Date(dueDate);
   676→   676→  const diffMs = due.getTime() - now.getTime();
   677→   677→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   678→   678→  if (diffDays < 0) return 'overdue';
   679→   679→  if (diffDays === 0) return 'today';
   680→   680→  if (diffDays <= 3) return 'soon';
   681→   681→  return 'future';
   682→   682→}

  1087→  1087→function PercentageRing({ percentage, size = 140, strokeWidth = 10, color = '#10B981' }: { percentage: number; size?: number; strokeWidth?: number; color?: string }) {
