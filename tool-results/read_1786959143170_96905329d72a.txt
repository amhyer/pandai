   554→// ═══════════════════════════════════════════════════════════════════════
   555→// Sidebar Navigation — grouped by section (SIMANTAP pattern)
   556→// ═══════════════════════════════════════════════════════════════════════
   557→
   558→interface SidebarNavProps {
   559→  sections: NavSection[];
   560→  currentView: ViewType;
   561→  onNavigate: (view: ViewType) => void;
   562→  onLogout: () => void;
   563→}
   564→
   565→function SidebarNav({ sections, currentView, onNavigate, onLogout }: SidebarNavProps) {
   566→  const [collapsed, setCollapsed] = React.useState<Record<string, boolean>>({});
   567→
   568→  const toggleSection = (section: string) => {
   569→    setCollapsed((prev) => ({ ...prev, [section]: !prev[section] }));
   570→  };
   571→
   572→  return (
   573→    <div className="flex h-full flex-col min-h-0">
   574→      {/* Scrollable nav area */}
   575→      <ScrollArea className="flex-1 px-3 py-2 min-h-0">
   576→        <nav className="flex flex-col gap-1 pb-2" role="navigation" aria-label="Main navigation">
   577→          {sections.map((group) => (
   578→            <React.Fragment key={group.section}>
   579→              {/* Section label — clickable to collapse */}
   580→              <button
   581→                onClick={() => toggleSection(group.section)}
   582→                className="mt-4 mb-1 flex w-full items-center justify-between px-3 first:mt-0 group/section"
   583→                aria-expanded={!collapsed[group.section]}
   584→              >
   585→                <span className="text-[11px] font-semibold uppercase tracking-wider text-white/40">
   586→                  {group.section}
   587→                </span>
   588→                <ChevronRight
   589→                  className={cn(
   590→                    'h-3 w-3 text-white/30 transition-transform duration-200',
   591→                    !collapsed[group.section] && 'rotate-90'
   592→                  )}
   593→                />
   594→              </button>
   595→
   596→              {/* Items — animated collapse */}
   597→              <div
   598→                className={cn(
   599→                  'overflow-hidden transition-all duration-200 ease-in-out',
   600→                  collapsed[group.section] ? 'max-h-0 opacity-0' : 'max-h-[600px] opacity-100'
   601→                )}
   602→              >
   603→                {group.items.map((item) => {
   604→                  const Icon = item.icon;
   605→                  const isActive = currentView === item.view;
   606→                  return (
   607→                    <button
   608→                      key={item.view}
   609→                      onClick={() => onNavigate(item.view)}
   610→                      className={cn(
   611→                        'flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium transition-all duration-150 text-left',
   612→                        'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/30',
   613→                        'hover:scale-[1.02] active:scale-[0.98]',
   614→                        isActive
   615→                          ? 'bg-white/10 text-amber-400 shadow-sm shadow-amber-400/10 border-l-[3px] border-amber-400'
   616→                          : 'text-white/70 hover:bg-white/5 hover:text-white border-l-[3px] border-transparent'
   617→                      )}
   618→                      aria-current={isActive ? 'page' : undefined}
   619→                    >
   620→                      <Icon className="h-4.5 w-4.5 shrink-0" />
   621→                      <span className="truncate">{item.label}</span>
   622→                    </button>
   623→                  );
   624→                })}
   625→              </div>
   626→            </React.Fragment>
   627→          ))}
   628→        </nav>
   629→      </ScrollArea>
   630→
   631→      {/* Logout at bottom — always visible */}
   632→      <div className="shrink-0 border-t border-white/10 p-3">
   633→        <button
   634→          onClick={onLogout}
   635→          className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium text-white/60 transition-all duration-150 hover:bg-red-500/10 hover:text-red-400 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/30 text-left hover:scale-[1.02] active:scale-[0.98]"
   636→        >
   637→          <LogOut className="h-4.5 w-4.5 shrink-0" />
   638→          <span>Keluar</span>
   639→        </button>
   640→      </div>
   641→    </div>
   642→  );
   643→}
