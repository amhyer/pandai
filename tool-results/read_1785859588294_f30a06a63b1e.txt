   420→// ═══════════════════════════════════════════════════════════════════════
   421→// Sidebar Navigation — grouped by section (SIMANTAP pattern)
   422→// ═══════════════════════════════════════════════════════════════════════
   423→
   424→interface SidebarNavProps {
   425→  sections: NavSection[];
   426→  currentView: ViewType;
   427→  onNavigate: (view: ViewType) => void;
   428→  onLogout: () => void;
   429→}
   430→
   431→function SidebarNav({ sections, currentView, onNavigate, onLogout }: SidebarNavProps) {
   432→  const [collapsed, setCollapsed] = React.useState<Record<string, boolean>>({});
   433→
   434→  const toggleSection = (section: string) => {
   435→    setCollapsed((prev) => ({ ...prev, [section]: !prev[section] }));
   436→  };
   437→
   438→  return (
   439→    <div className="flex h-full flex-col min-h-0">
   440→      {/* Scrollable nav area */}
   441→      <ScrollArea className="flex-1 px-3 py-2 min-h-0">
   442→        <nav className="flex flex-col gap-1 pb-2" role="navigation" aria-label="Main navigation">
   443→          {sections.map((group) => (
   444→            <React.Fragment key={group.section}>
   445→              {/* Section label — clickable to collapse */}
   446→              <button
   447→                onClick={() => toggleSection(group.section)}
   448→                className="mt-4 mb-1 flex w-full items-center justify-between px-3 first:mt-0 group/section"
   449→                aria-expanded={!collapsed[group.section]}
   450→              >
   451→                <span className="text-[11px] font-semibold uppercase tracking-wider text-white/40">
   452→                  {group.section}
   453→                </span>
   454→                <ChevronRight
   455→                  className={cn(
   456→                    'h-3 w-3 text-white/30 transition-transform duration-200',
   457→                    !collapsed[group.section] && 'rotate-90'
   458→                  )}
   459→                />
   460→              </button>
   461→
   462→              {/* Items — animated collapse */}
   463→              <div
   464→                className={cn(
   465→                  'overflow-hidden transition-all duration-200 ease-in-out',
   466→                  collapsed[group.section] ? 'max-h-0 opacity-0' : 'max-h-[600px] opacity-100'
   467→                )}
   468→              >
   469→                {group.items.map((item) => {
   470→                  const Icon = item.icon;
   471→                  const isActive = currentView === item.view;
   472→                  return (
   473→                    <button
   474→                      key={item.view}
   475→                      onClick={() => onNavigate(item.view)}
   476→                      className={cn(
   477→                        'flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium transition-all duration-150 text-left',
   478→                        'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/30',
   479→                        'hover:scale-[1.02] active:scale-[0.98]',
   480→                        isActive
   481→                          ? 'bg-white/10 text-amber-400 shadow-sm shadow-amber-400/10 border-l-[3px] border-amber-400'
   482→                          : 'text-white/70 hover:bg-white/5 hover:text-white border-l-[3px] border-transparent'
   483→                      )}
   484→                      aria-current={isActive ? 'page' : undefined}
   485→                    >
   486→                      <Icon className="h-4.5 w-4.5 shrink-0" />
   487→                      <span className="truncate">{item.label}</span>
   488→                    </button>
   489→                  );
   490→                })}
   491→              </div>
   492→            </React.Fragment>
   493→          ))}
   494→        </nav>
   495→      </ScrollArea>
   496→
   497→      {/* Logout at bottom — always visible */}
   498→      <div className="shrink-0 border-t border-white/10 p-3">
   499→        <button
   500→          onClick={onLogout}
   501→          className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium text-white/60 transition-all duration-150 hover:bg-red-500/10 hover:text-red-400 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/30 text-left hover:scale-[1.02] active:scale-[0.98]"
   502→        >
   503→          <LogOut className="h-4.5 w-4.5 shrink-0" />
   504→          <span>Keluar</span>
   505→        </button>
   506→      </div>
   507→    </div>
   508→  );
   509→}

   515→export default function AppLayout({ children }: { children: React.ReactNode }) {
   516→  const user = useAppStore((s) => s.user);
   517→  const currentView = useAppStore((s) => s.currentView);
   518→  const sidebarOpen = useAppStore((s) => s.sidebarOpen);
   519→  const navigateTo = useAppStore((s) => s.navigateTo);
   520→  const logout = useAppStore((s) => s.logout);
   521→  const toggleSidebar = useAppStore((s) => s.toggleSidebar);

   529→      .join('')
   530→      .toUpperCase()
   531→      .slice(0, 2) ?? 'U';
   532→
   533→  const handleLogout = () => {

   536→  };
   537→
   538→  // ── Sidebar content (shared between desktop & mobile) ──
   539→
   540→  const sidebarBranding = (

   542→      <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-amber-400">
   543→        <GraduationCap className="h-5 w-5 text-[#1F3864]" />
   544→      </div>
   545→      <span className="text-xl font-bold tracking-tight text-white">PANDAI</span>
   546→    </div>
   547→  );
   548→
   549→  const sidebarUser = user && (
   550→    <div className="px-4 py-3">
   551→      <p className="truncate text-sm font-semibold text-white">{user.name}</p>
   552→      <span className="mt-1 inline-block rounded-full bg-amber-400/20 px-2.5 py-0.5 text-xs font-medium text-amber-400">

   554→      </span>
   555→      {user.schoolName && (
   556→        <p className="mt-1.5 truncate text-xs text-white/50">{user.schoolName}</p>
   557→      )}
   558→    </div>
   559→  );
   560→
   561→  const sidebarNav = (
   562→    <SidebarNav
   563→      sections={navSections}
   564→      currentView={currentView}
   565→      onNavigate={navigateTo}
   566→      onLogout={handleLogout}
   567→    />
   568→  );
   569→
   570→  return (
   571→    <div className="flex h-screen overflow-hidden bg-gray-50">
   572→      {/* ── Desktop Sidebar ── */}
   573→      <aside
   574→        className={cn(
   575→          'hidden lg:flex lg:flex-col lg:w-64 bg-[#1F3864] shrink-0 transition-all duration-300 overflow-hidden',
   576→          !sidebarOpen && 'lg:w-0'
   577→        )}
   578→      >
   579→        {/* Branding + User info — fixed at top */}
   580→        <div className="shrink-0">
   581→          {sidebarBranding}
   582→          {sidebarUser && (
   583→            <>
   584→              <div className="mx-4">
   585→                <Separator className="bg-white/10" />
   586→              </div>
   587→              {sidebarUser}
   588→            </>
   589→          )}
   590→        </div>
   591→        {/* Nav takes remaining space with scroll */}
   592→        <div className="flex-1 min-h-0">
   593→          {sidebarNav}
   594→        </div>
   595→      </aside>
   596→
   597→      {/* ── Mobile Sidebar (Sheet) ── */}
   598→      <Sheet>
   599→        <SheetTrigger asChild>
   600→          <Button
   601→            variant="ghost"
   602→            size="icon"
   603→            className="fixed left-4 top-3 z-40 lg:hidden"
   604→            aria-label="Toggle menu"
   605→          >
   606→            <Menu className="h-5 w-5" />
   607→          </Button>
   608→        </SheetTrigger>
   609→        <SheetContent side="left" className="w-72 p-0 bg-[#1F3864] border-r-white/10 flex flex-col">
   610→          <SheetTitle className="sr-only">Menu Navigasi</SheetTitle>
   611→          {/* Branding + User info — fixed at top on mobile */}
   612→          <div className="shrink-0">
   613→            {sidebarBranding}
   614→            {sidebarUser && (
   615→              <>
   616→                <div className="mx-4">
   617→                  <Separator className="bg-white/10" />
   618→                </div>
   619→                {sidebarUser}
   620→              </>
   621→            )}
   622→          </div>
   623→          {/* Nav takes remaining space with scroll on mobile */}
   624→          <div className="flex-1 min-h-0">
   625→            {sidebarNav}
   626→          </div>
   627→        </SheetContent>
   628→      </Sheet>
   629→
   630→      {/* ── Right column: Header + Content ── */}
   631→      <div className="flex flex-1 flex-col overflow-hidden">
   632→        {/* ── Header ── */}
   633→        <header className="sticky top-0 z-30 flex h-14 shrink-0 items-center gap-4 border-b bg-white px-4 shadow-sm sm:px-6">

   641→            className="hidden lg:flex"
   642→            onClick={toggleSidebar}
   643→            aria-label="Toggle sidebar"
   644→          >
   645→            <Menu className="h-5 w-5" />
   646→          </Button>
   647→
   648→          {/* Breadcrumb */}
   649→          <Breadcrumb className="hidden sm:flex">
   650→            <BreadcrumbList>
   651→              {breadcrumbs.map((crumb, idx) => {
   652→                const isLast = idx === breadcrumbs.length - 1;
   653→                return (
   654→                  <React.Fragment key={crumb.label}>
   655→                    <BreadcrumbItem>
   656→                      {isLast ? (
   657→                        <BreadcrumbPage>{crumb.label}</BreadcrumbPage>
   658→                      ) : (
   659→                        <BreadcrumbLink
   660→                          className="cursor-pointer"
   661→                          onClick={() => crumb.view && navigateTo(crumb.view)}
   662→                        >
   663→                          {crumb.label}
   664→                        </BreadcrumbLink>
   665→                      )}
   666→                    </BreadcrumbItem>
   667→                    {!isLast && <BreadcrumbSeparator />}
   668→                  </React.Fragment>
   669→                );
   670→              })}
   671→            </BreadcrumbList>
   672→          </Breadcrumb>
   673→
   674→          {/* Mobile breadcrumb - just show current page */}
   675→          <span className="sm:hidden text-sm font-medium text-foreground">
   676→            {breadcrumbs[breadcrumbs.length - 1]?.label}
   677→          </span>
   678→
   679→          {/* Right side */}
   680→          <div className="ml-auto flex items-center gap-2">
   681→            {/* Notification bell */}
   682→            <Button
   683→              variant="ghost"
   684→              size="icon"
   685→              className="relative hover:bg-muted transition-colors duration-200"
   686→              aria-label="Notifikasi"

   693→            {/* User dropdown */}

   698→                    <AvatarFallback className="bg-[#1F3864] text-xs font-semibold text-white">
   699→                      {initials}
   700→                    </AvatarFallback>
   701→                  </Avatar>
   702→                </Button>
   703→              </DropdownMenuTrigger>
   704→              <DropdownMenuContent className="w-48" align="end" forceMount>
   705→                {user && (
   706→                  <>
   707→                    <div className="flex flex-col gap-1 p-2">
   708→                      <p className="text-sm font-medium leading-none">{user.name}</p>
   709→                      <p className="text-xs text-muted-foreground">{user.email}</p>
   710→                    </div>
   711→                    <DropdownMenuSeparator />

   718→                  Pengaturan
   719→                </DropdownMenuItem>
   720→                <DropdownMenuSeparator />
   721→                <DropdownMenuItem onClick={handleLogout} className="text-red-600 focus:text-red-600">
   722→                  Keluar
   723→                </DropdownMenuItem>
   724→              </DropdownMenuContent>
   725→            </DropdownMenu>
   726→          </div>

   729→        {/* ── Main Content ── */}
   730→        <main className="flex-1 overflow-y-auto bg-gradient-to-br from-slate-50 via-gray-50 to-slate-100 p-4 sm:p-6">
   731→          <div className="mx-auto max-w-7xl">
   732→            {children}
   733→          </div>
   734→        </main>
   735→      </div>
   736→    </div>
   737→  );
   738→}
