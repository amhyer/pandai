   114→// ---------------------------------------------------------------------------
   115→// Relative time helper
   116→// ---------------------------------------------------------------------------
   117→
   118→function timeAgo(dateStr: string): string {
   119→  const now = new Date();
   120→  const date = new Date(dateStr);
   121→  const diffMs = now.getTime() - date.getTime();
   122→  const diffMins = Math.floor(diffMs / 60000);
   123→  const diffHours = Math.floor(diffMins / 60);
   124→  const diffDays = Math.floor(diffHours / 24);
   125→
   126→  if (diffMins < 1) return 'Baru saja';
   127→  if (diffMins < 60) return `${diffMins} menit lalu`;
   128→  if (diffHours < 24) return `${diffHours} jam lalu`;
   129→  if (diffDays < 7) return `${diffDays} hari lalu`;
   130→  return date.toLocaleDateString('id-ID', {
   131→    day: 'numeric',
   132→    month: 'long',
   133→    year: 'numeric',
   134→  });
   135→}

   250→export function ProfileView() {
   251→  const { user, setUser } = useAppStore();
   252→
   253→  const [isEditing, setIsEditing] = useState(false);
   254→  const [showPassword, setShowPassword] = useState(false);
   255→  const [isSaving, setIsSaving] = useState(false);
   256→
   257→  const [formData, setFormData] = useState({
   258→    name: user?.name ?? '',
   259→    email: user?.email ?? '',
   260→    phone: user?.phone ?? '',
   261→    password: '',
   262→  });
   263→
   264→  // Sync form when user data changes
   265→  useEffect(() => {
   266→    if (user) {
   267→      setFormData({
   268→        name: user.name ?? '',
   269→        email: user.email ?? '',
   270→        phone: user.phone ?? '',
   271→        password: '',
   272→      });
   273→    }
   274→  }, [user]);
   275→
   276→  const handleSave = async () => {
   277→    if (!user) return;
   278→    setIsSaving(true);
   279→    try {
   280→      const res = await fetch('/api/users', {
   281→        method: 'PUT',
   282→        headers: { 'Content-Type': 'application/json' },
   283→        body: JSON.stringify({
   284→          id: user.id,
   285→          name: formData.name,
   286→          email: formData.email,
   287→          phone: formData.phone,
   288→        }),
   289→      });
   290→      if (!res.ok) {
   291→        const err = await res.json();
   292→        throw new Error(err.error || 'Gagal memperbarui profil');
   293→      }
   294→      const updatedUser = await res.json();
   295→      // Update store with fresh data
   296→      if (setUser) {
   297→        setUser({
   298→          ...user,
   299→          name: updatedUser.name ?? user.name,
   300→          email: updatedUser.email ?? user.email,
   301→          phone: updatedUser.phone ?? user.phone,
   302→        });
   303→      }
   304→      toast.success('Profil berhasil diperbarui');
   305→      setIsEditing(false);
   306→    } catch (error: any) {
   307→      toast.error(error.message || 'Terjadi kesalahan saat memperbarui profil');
   308→    } finally {
   309→      setIsSaving(false);
   310→    }
   311→  };
   312→

   343→  const getIdValue = () => {

   347→      case 'GURU':

   366→        </div>
   367→      </div>
   368→
