  1058→/**
  1059→ * Lookup schools by NPSN (exact match) or name/city/province/district (partial, case-insensitive)
  1060→ * Supports word-based matching: "SMA 1 Makassar" will match "SMA Negeri 1 Makassar"
  1061→ */
  1062→export function lookupSchool(query: string): (NpsnSchool & { source: string })[] {
  1063→  if (!query || !query.trim()) return [];
  1064→  const q = query.trim();
  1065→
  1066→  // Exact NPSN match first
  1067→  const npsnMatch = NPSN_DATABASE.filter((s) => s.npsn === q);
  1068→  if (npsnMatch.length > 0) {
  1069→    return npsnMatch.map((s) => ({ ...s, source: 'dapodik' }));
  1070→  }
  1071→
  1072→  const lower = q.toLowerCase();
  1073→
  1074→  // Partial match (case-insensitive) on name, city, province, and district
  1075→  const substringMatches = NPSN_DATABASE.filter(
  1076→    (s) =>
  1077→      s.name.toLowerCase().includes(lower) ||
  1078→      s.city.toLowerCase().includes(lower) ||
  1079→      s.province.toLowerCase().includes(lower) ||
  1080→      s.district.toLowerCase().includes(lower) ||
  1081→      s.address.toLowerCase().includes(lower)
  1082→  );
  1083→  if (substringMatches.length > 0) {
  1084→    return substringMatches.map((s) => ({ ...s, source: 'dapodik' }));
  1085→  }
  1086→
  1087→  // Word-based matching: all significant words (>= 2 chars) must appear in the name/city/province
  1088→  const keywords = lower.split(/\s+/).filter((w) => w.length >= 2);
  1089→  if (keywords.length >= 2) {
  1090→    const wordMatches = NPSN_DATABASE.filter((s) => {
  1091→      const searchPool = `${s.name} ${s.city} ${s.province} ${s.district} ${s.schoolType}`.toLowerCase();
  1092→      return keywords.every((kw) => searchPool.includes(kw));
  1093→    });
  1094→    if (wordMatches.length > 0) {
  1095→      return wordMatches.map((s) => ({ ...s, source: 'dapodik' }));
  1096→    }
  1097→  }
  1098→
  1099→  // Single word / short query: match against name, city, province
  1100→  const shortMatches = NPSN_DATABASE.filter((s) => {
  1101→    const searchPool = `${s.name} ${s.city} ${s.province} ${s.district} ${s.schoolType}`.toLowerCase();
  1102→    return keywords.some((kw) => searchPool.includes(kw));
  1103→  });
  1104→
  1105→  return shortMatches.map((s) => ({ ...s, source: 'dapodik' }));
  1106→}

  1111→export function getSchoolByNpsn(npsn: string): (NpsnSchool & { source: string }) | null {
