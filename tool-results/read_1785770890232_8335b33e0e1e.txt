   135→function scoreColor(score: number) {
   136→  if (score >= 85) return 'text-emerald-600';
   137→  if (score >= 70) return 'text-amber-600';
   138→  if (score >= 55) return 'text-orange-600';
   139→  return 'text-red-600';
   140→}
   141→
   142→function scoreBgColor(score: number) {
   143→  if (score >= 85) return 'bg-emerald-50 text-emerald-700 border-emerald-200';
   144→  if (score >= 70) return 'bg-amber-50 text-amber-700 border-amber-200';
   145→  if (score >= 55) return 'bg-orange-50 text-orange-700 border-orange-200';
   146→  return 'bg-red-50 text-red-700 border-red-200';
   147→}
   148→
   149→function progressBarColor(score: number) {
   150→  if (score >= 85) return 'bg-emerald-500';
   151→  if (score >= 70) return 'bg-amber-500';
   152→  if (score >= 55) return 'bg-orange-500';
   153→  return 'bg-red-500';
   154→}
   155→
   156→function scoreGradeLabel(score: number) {
   157→  if (score >= 85) return 'Sangat Baik';
   158→  if (score >= 70) return 'Baik';
   159→  if (score >= 55) return 'Cukup';
   160→  return 'Perlu Ditingkatkan';
   161→}

   163→function getInitials(name: string) {
   164→  return name
   165→    .split(' ')
   166→    .map((n) => n[0])
   167→    .join('')
   168→    .toUpperCase()
   169→    .slice(0, 2);
   170→}
