   292→   292→  async function handleSave() {
   293→   293→    if (!formCode.trim() || !formName.trim()) {
   294→   294→      toast.error('Kode dan Nama mata pelajaran wajib diisi');
   295→   295→      return;
   296→   296→    }
   297→   297→    setSaving(true);
   298→   298→    try {
   299→   299→      if (editingSubject) {
   300→   300→        const res = await fetch(`/api/subjects?id=${editingSubject.id}`, {
   301→   301→          method: 'PATCH',
   302→   302→          headers: { 'Content-Type': 'application/json' },
   303→   303→          body: JSON.stringify({
   304→   304→            code: formCode.trim(),
   305→   305→            name: formName.trim(),
   306→   306→            type: formType,
   307→   307→            sortOrder: formOrder,
   308→   308→          }),
   309→   309→        });
   310→   310→        if (res.ok) {
   311→   311→          toast.success('Mata pelajaran berhasil diperbarui');
   312→   312→          setDialogOpen(false);
   313→   313→          fetchSubjects();
   314→   314→          return;
   315→   315→        }
   316→   316→      } else {
   317→   317→        const res = await fetch('/api/subjects', {
   318→   318→          method: 'POST',
   319→   319→          headers: { 'Content-Type': 'application/json' },
   320→   320→          body: JSON.stringify({
   321→   321→            code: formCode.trim(),
   322→   322→            name: formName.trim(),
   323→   323→            type: formType,
   324→   324→            sortOrder: formOrder,
   325→   325→          }),
   326→   326→        });
   327→   327→        if (res.ok) {
   328→   328→          toast.success('Mata pelajaran berhasil ditambahkan');
   329→   329→          setDialogOpen(false);
   330→   330→          fetchSubjects();
   331→   331→          return;
   332→   332→        }
   333→   333→      }
   334→   334→    } catch {
   335→   335→      // fallback: update local state
   336→   336→    }
   337→   337→    // Mock fallback
   338→   338→    if (editingSubject) {
   339→   339→      setSubjects((prev) =>
   340→   340→        prev.map((s) =>
   341→   341→          s.id === editingSubject.id
   342→   342→            ? { ...s, code: formCode.trim(), name: formName.trim(), type: formType, sortOrder: formOrder }
   343→   343→            : s
   344→   344→        )
   345→   345→      );
   346→   346→      toast.success('Mata pelajaran berhasil diperbarui');
   347→   347→    } else {
   348→   348→      const newSubject: Subject = {
   349→   349→        id: `s${Date.now()}`,
   350→   350→        code: formCode.trim(),
   351→   351→        name: formName.trim(),
   352→   352→        type: formType,
   353→   353→        sortOrder: formOrder,
   354→   354→      };
   355→   355→      setSubjects((prev) => [...prev, newSubject]);
   356→   356→      toast.success('Mata pelajaran berhasil ditambahkan');
   357→   357→    }
   358→   358→    setDialogOpen(false);
   359→   359→    setSaving(false);
   360→   360→  }
   361→   361→
   362→   362→  async function handleDelete() {
