   224→function generateMockAttendance(year: number, month: number): AttendanceDay[] {
   225→  const days: AttendanceDay[] = [];
   226→  const daysInMonth = new Date(year, month + 1, 0).getDate();
   227→  const statuses: Array<'hadir' | 'izin' | 'sakit' | 'alpa'> = ['hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'izin', 'sakit', 'alpa'];
   228→  let statusIdx = 0;
   229→
   230→  for (let d = 1; d <= daysInMonth; d++) {
   231→    const date = new Date(year, month, d);
   232→    const dow = date.getDay();
   233→    if (dow === 0 || dow === 6) {
   234→      days.push({ day: d, status: 'weekend' });
   235→    } else {
   236→      const s = statuses[statusIdx % statuses.length];
   237→      days.push({ day: d, status: s });
   238→      statusIdx++;
   239→    }
   240→  }
   241→  return days;
   242→}

   260→        const params = new URLSearchParams({
   261→          schoolId: user?.schoolId || '',
   262→          classId: user?.classId || '',
   263→          type: 'materi',
   264→          status: 'published',
   265→        });

   267→        if (res.ok) {
   268→          const data = await res.json();
   269→          if (Array.isArray(data) && data.length > 0) {
   270→            // Map API data into our subject structure
   271→            const mapped = MOCK_SUBJECTS.map((subj) => {
   272→              const apiMaterials = (data as Material[]).filter(
   273→                (m) => m.subject === subj.name
   274→              );
   275→              return apiMaterials.length > 0
   276→                ? { ...subj, materials: apiMaterials }
   277→                : subj;
   278→            });

   281→            setSubjects(MOCK_SUBJECTS);

   284→          setSubjects(MOCK_SUBJECTS);
   285→        }
   286→      } catch {
   287→        setSubjects(MOCK_SUBJECTS);
   288→      } finally {
   289→        setIsLoading(false);

   292→    fetchMaterials();

   296→    setExpandedSubject((prev) => (prev === subjectId ? null : subjectId));

   303→      // Search filter
   304→      if (searchQuery.trim()) {
   305→        const q = searchQuery.toLowerCase();
   306→        mats = mats.filter(
   307→          (m) =>
   308→            m.title.toLowerCase().includes(q) ||
   309→            m.description.toLowerCase().includes(q)
   310→        );

   312→

   315→        mats = mats.filter((m) => !m.isRead);

   317→        mats = mats.filter((m) => m.isRead);
   318→      }

   321→    });

   325→    () => subjects.reduce((sum, s) => sum + s.materials.length, 0),

   336→      prev.map((subj) => {
   337→        if (subj.id !== subjectId) return subj;
   338→        return {
   339→          ...subj,
   340→          materials: subj.materials.map((m) =>
   341→            m.id === materialId ? { ...m, isRead: true } : m
   342→          ),
   343→        };
   344→      })
   345→    );
   346→    toast.success('Materi ditandai sudah dibaca!');
   347→  };
   348→
   349→  if (isLoading) {
   350→    return (
   351→      <div className="space-y-6">
   352→        <Skeleton className="h-44 w-full rounded-2xl" />
   353→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   354→          {Array.from({ length: 6 }).map((_, i) => (
   355→            <Skeleton key={i} className="h-36 rounded-xl" />
   356→          ))}
   357→        </div>
   358→      </div>
   359→    );
   360→  }
   361→
   362→  return (
   363→    <div className="space-y-6">
   364→      {/* ── Hero Banner ── */}
   365→      <div
   366→        className="relative rounded-2xl overflow-hidden p-6 md:p-8"

   368→      >

   371→        <div className="relative z-10">

   373→            <div className="p-2.5 rounded-xl bg-white/15 backdrop-blur-sm">
   374→              <BookMarked className="h-6 w-6 text-white" />
   375→            </div>

   377→          </div>

   379→            Kelas {user?.className || 'XII IPA'} — Semua materi tersedia untuk dipelajari

   384→              <span>{totalMaterials} Materi</span>
