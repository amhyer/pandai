    73→function doGet(e) {
    74→  if (e && e.parameter && e.parameter.action) {
    75→    return handleRequest({
    76→      action: e.parameter.action,
    77→      payload: e.parameter.payload ? JSON.parse(e.parameter.payload) : {},
    78→      token: e.parameter.token || ''
    79→    });
    80→  }
    81→  return ContentService
    82→    .createTextOutput('SIMANTAP API aktif. Gunakan aplikasi index.html untuk mengakses layanan ini.')
    83→    .setMimeType(ContentService.MimeType.TEXT);
    84→}
    85→
    86→function doPost(e) {
    87→  let body = {};
    88→  try {
    89→    body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    90→  } catch (err) {
    91→    return jsonOut({ ok: false, error: 'Format permintaan tidak valid.' });
    92→  }
    93→  return handleRequest(body);
    94→}
    95→
    96→function handleRequest(body) {
    97→  const action = body.action;
    98→  const payload = body.payload || {};
    99→  const token = body.token || '';
   100→  try {
   101→    const fn = ACTIONS[action];
   102→    if (!fn) throw new Error('Aksi tidak dikenal: ' + action);
   103→    const result = fn(payload, token);
   104→    return jsonOut({ ok: true, data: result });
   105→  } catch (err) {
   106→    return jsonOut({ ok: false, error: err.message || String(err) });
   107→  }
   108→}
   109→
   110→function jsonOut(obj) {
   111→  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
   112→}

   118→function sheet(name) {
   119→  const ss = SpreadsheetApp.getActiveSpreadsheet();
   120→  let sh = ss.getSheetByName(name);
   121→  if (!sh) sh = ss.insertSheet(name);
   122→  return sh;
   123→}

   125→function readSheet(name) {
   126→  const sh = sheet(name);
   127→  const values = sh.getDataRange().getValues();
   128→  if (values.length < 2) return [];
   129→  const header = values[0];
   130→  const rows = [];
   131→  for (let i = 1; i < values.length; i++) {
   132→    const row = values[i];
   133→    if (row.every(function (c) { return c === '' || c === null; })) continue;
   134→    const obj = {};
   135→    header.forEach(function (h, idx) { obj[h] = row[idx]; });
   136→    rows.push(obj);
   137→  }
   138→  return rows;
   139→}

   141→function getAll(name) { return readSheet(name); }

   143→function writeAll(name, objects) {
   144→  const sh = sheet(name);
   145→  const header = SCHEMA[name];
   146→  sh.clearContents();
   147→  sh.getRange(1, 1, 1, header.length).setValues([header]);
   148→  sh.setFrozenRows(1);
   149→  if (objects && objects.length) {
   150→    const rows = objects.map(function (o) {
   151→      return header.map(function (h) { return o[h] !== undefined ? o[h] : ''; });
   152→    });
   153→    sh.getRange(2, 1, rows.length, header.length).setValues(rows);
   154→  }
   155→}

   157→function nextId(name) {
   158→  const rows = getAll(name);
   159→  let max = 0;
   160→  rows.forEach(function (r) {
   161→    const n = parseInt(r.id, 10);
   162→    if (!isNaN(n) && n > max) max = n;
   163→  });
   164→  return max + 1;
   165→}

   167→function insert(name, obj) {
   168→  const header = SCHEMA[name];
   169→  if (header.indexOf('id') !== -1 && (obj.id === undefined || obj.id === '')) {
   170→    obj.id = nextId(name);
   171→  }
   172→  const sh = sheet(name);
   173→  const row = header.map(function (h) { return obj[h] !== undefined ? obj[h] : ''; });
   174→  sh.appendRow(row);
   175→  return obj;
   176→}

   178→function findRowIndexById(name, id) {
   179→  const sh = sheet(name);
   180→  const values = sh.getDataRange().getValues();
   181→  const header = values[0] || [];
   182→  const idCol = header.indexOf('id');
   183→  if (idCol === -1) return -1;
   184→  for (let i = 1; i < values.length; i++) {
   185→    if (String(values[i][idCol]) === String(id)) return i + 1;
   186→  }
   187→  return -1;
   188→}
   189→
   190→function updateById(name, id, patch) {
   191→  const sh = sheet(name);
   192→  const header = SCHEMA[name];
   193→  const rowIdx = findRowIndexById(name, id);
   194→  if (rowIdx === -1) throw new Error('Data dengan id ' + id + ' tidak ditemukan di ' + name + '.');
   195→  const rowValues = sh.getRange(rowIdx, 1, 1, header.length).getValues()[0];
   196→  const current = {};
   197→  header.forEach(function (h, i) { current[h] = rowValues[i]; });
   198→  const merged = Object.assign({}, current, patch, { id: id });
   199→  const newRow = header.map(function (h) { return merged[h] !== undefined ? merged[h] : ''; });
   200→  sh.getRange(rowIdx, 1, 1, header.length).setValues([newRow]);
   201→  return merged;
   202→}
   203→
   204→function deleteById(name, id) {
   205→  const rowIdx = findRowIndexById(name, id);
   206→  if (rowIdx === -1) throw new Error('Data tidak ditemukan.');
   207→  sheet(name).deleteRow(rowIdx);
   208→}

   210→function crudActions(sheetName, writeRoles, readRoles) {
   211→  return {
   212→    list: function (payload, token) {
   213→      requireAuth(token, readRoles);
   214→      return getAll(sheetName);
   215→    },
   216→    save: function (payload, token) {
   217→      requireAuth(token, writeRoles);
   218→      if (payload.id) return updateById(sheetName, payload.id, payload);
   219→      return insert(sheetName, payload);
   220→    },
   221→    remove: function (payload, token) {
   222→      requireAuth(token, writeRoles);
   223→      deleteById(sheetName, payload.id);
   224→      return { deleted: true };
   225→    }
   226→  };
   227→}

   229→function shuffleArray(arr) {
   230→  const a = arr.slice();
   231→  for (let i = a.length - 1; i > 0; i--) {
   232→    const j = Math.floor(Math.random() * (i + 1));
   233→    const tmp = a[i]; a[i] = a[j]; a[j] = tmp;
   234→  }
   235→  return a;
   236→}

   242→function getSettings() {

   244→  const obj = {};

   249→function saveSettings(patch) {
   250→  const sh = sheet(SHEETS.SETTINGS);
   251→  const values = sh.getDataRange().getValues();
   252→  const map = {};
   253→  for (let i = 1; i < values.length; i++) map[values[i][0]] = i + 1;
   254→  Object.keys(patch).forEach(function (k) {
   255→    if (k === 'id') return;
   256→    if (map[k]) sh.getRange(map[k], 2).setValue(patch[k]);
   257→    else sh.appendRow([k, patch[k]]);
   258→  });
   259→  return getSettings();
   260→}

   266→function hashPassword(pw) {
   267→  const raw = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, pw + SALT_PASSWORD);
   268→  return raw.map(function (b) { return (b < 0 ? b + 256 : b).toString(16).padStart(2, '0'); }).join('');
   269→}

   271→function cleanupSessions() {
   272→  const sh = sheet(SHEETS.SESSIONS);
   273→  const values = sh.getDataRange().getValues();
   274→  if (values.length < 2) return;
   275→  const now = new Date();
   276→  for (let i = values.length - 1; i >= 1; i--) {
   277→    const exp = new Date(values[i][7]);
   278→    if (isNaN(exp.getTime()) || exp < now) sh.deleteRow(i + 1);
   279→  }
   280→}

   282→function requireAuth(token, roles) {
   283→  if (!token) throw new Error('Anda belum login. Silakan login kembali.');
   284→  const sessions = getAll(SHEETS.SESSIONS);
   285→  const s = sessions.find(function (x) { return x.token === token; });
   286→  if (!s) throw new Error('Sesi tidak valid. Silakan login kembali.');
   287→  if (new Date(s.kedaluwarsa) < new Date()) throw new Error('Sesi telah berakhir. Silakan login kembali.');
   288→  if (roles && roles.length && roles.indexOf(s.role) === -1) {
   289→    throw new Error('Anda tidak memiliki akses untuk aksi ini.');
   290→  }
   291→  return s;
   292→}

   303→function actLogin(payload) {
   304→  const username = payload && payload.username;
   305→  const password = payload && payload.password;
   306→  if (!username || !password) throw new Error('Username dan password wajib diisi.');
   307→  cleanupSessions();
   308→  const users = getAll(SHEETS.USERS);
   309→  const u = users.find(function (x) { return x.username === username; });
   310→  if (!u) throw new Error('Username tidak ditemukan.');
   311→  if (String(u.aktif).toUpperCase() === 'FALSE') throw new Error('Akun tidak aktif. Hubungi admin sekolah.');
   312→  if (hashPassword(password) !== u.password) throw new Error('Password salah.');
   313→  const token = Utilities.getUuid();
   314→  const now = new Date();
   315→  const exp = new Date(now.getTime() + 12 * 60 * 60 * 1000);
   316→  insert(SHEETS.SESSIONS, {
   317→    token: token, user_id: u.id, username: u.username, role: u.role,
   318→    ref_id: u.ref_id, nama: u.nama, dibuat_pada: now.toISOString(), kedaluwarsa: exp.toISOString()
   319→  });
   320→  writeLog(u.username, u.role, 'login', '');
   321→  return { token: token, role: u.role, nama: u.nama, ref_id: u.ref_id, username: u.username, user_id: u.id };
   322→}

   376→function actAdminSaveUser(payload, token) {
   377→  const s = requireAuth(token, ['admin']);
   378→  if (!payload.username) throw new Error('Username wajib diisi.');
   379→  if (!payload.role) throw new Error('Peran (role) wajib dipilih.');
   380→  const users = getAll(SHEETS.USERS);
   381→  if (payload.id) {
   382→    const existing = users.find(function (u) { return String(u.id) === String(payload.id); });
   383→    if (!existing) throw new Error('Pengguna tidak ditemukan.');
   384→    const dup = users.find(function (u) { return u.username === payload.username && String(u.id) !== String(payload.id); });
   385→    if (dup) throw new Error('Username sudah digunakan pengguna lain.');
   386→    const patch = {
   387→      username: payload.username, role: payload.role, nama: payload.nama,
   388→      ref_id: payload.ref_id || '', aktif: payload.aktif !== undefined ? payload.aktif : true
   389→    };
   390→    if (payload.password) patch.password = hashPassword(payload.password);
   391→    const updated = updateById(SHEETS.USERS, payload.id, patch);
   392→    writeLog(s.username, s.role, 'admin_saveUser', 'update:' + payload.username);
   393→    const copy = Object.assign({}, updated); delete copy.password; return copy;
   394→  } else {
   395→    const dup = users.find(function (u) { return u.username === payload.username; });
   396→    if (dup) throw new Error('Username sudah digunakan.');
   397→    if (!payload.password) throw new Error('Password wajib diisi untuk pengguna baru.');
   398→    const created = insert(SHEETS.USERS, {
   399→      username: payload.username, password: hashPassword(payload.password), role: payload.role,
   400→      nama: payload.nama, ref_id: payload.ref_id || '',
   401→      aktif: payload.aktif !== undefined ? payload.aktif : true, dibuat_pada: new Date().toISOString()
   402→    });
   403→    writeLog(s.username, s.role, 'admin_saveUser', 'create:' + payload.username);
   404→    const copy = Object.assign({}, created); delete copy.password; return copy;
   405→  }
   406→}

   523→function actGuruGetAnalisis(payload, token) {
   524→  requireAuth(token, ['guru', 'admin']);
   525→  const nilai = getAll(SHEETS.NILAI).filter(function (n) {
   526→    return String(n.kelas_id) === String(payload.kelas_id) && String(n.mapel_id) === String(payload.mapel_id);
   527→  });
   528→  const siswaList = getAll(SHEETS.SISWA).filter(function (x) { return String(x.kelas_id) === String(payload.kelas_id); });
   529→  const settings = getSettings();
   530→  const kkm = Number(payload.kkm || settings.kkm_default || 75);
   531→  const perSiswa = siswaList.map(function (sw) {
   532→    const nl = nilai.filter(function (n) { return String(n.siswa_id) === String(sw.id); });
   533→    const rata = nl.length ? nl.reduce(function (a, b) { return a + Number(b.nilai); }, 0) / nl.length : null;
   534→    return { siswa_id: sw.id, nama: sw.nama, rata_rata: rata, jumlah_nilai: nl.length, tuntas: rata !== null ? rata >= kkm : null };
   535→  });
   536→  const semuaRata = perSiswa.filter(function (p) { return p.rata_rata !== null; }).map(function (p) { return p.rata_rata; });
   537→  const rataKelas = semuaRata.length ? semuaRata.reduce(function (a, b) { return a + b; }, 0) / semuaRata.length : 0;
   538→  const tuntas = perSiswa.filter(function (p) { return p.tuntas; }).length;
   539→  const distribusi = { sangat_baik: 0, baik: 0, cukup: 0, kurang: 0 };
   540→  semuaRata.forEach(function (v) {
   541→    if (v >= 90) distribusi.sangat_baik++;
   542→    else if (v >= kkm) distribusi.baik++;
   543→    else if (v >= kkm - 15) distribusi.cukup++;
   544→    else distribusi.kurang++;
   545→  });
   546→  const jenisSet = nilai.reduce(function (acc, n) { if (acc.indexOf(n.jenis) === -1) acc.push(n.jenis); return acc; }, []);
   547→  const trenJenis = jenisSet.map(function (j) {
   548→    const vals = nilai.filter(function (n) { return n.jenis === j; }).map(function (n) { return Number(n.nilai); });
   549→    return { jenis: j, rata_rata: vals.length ? vals.reduce(function (a, b) { return a + b; }, 0) / vals.length : 0, jumlah: vals.length };
   550→  });
   551→  return {
   552→    rata_kelas: rataKelas, kkm: kkm, jumlah_siswa: siswaList.length, jumlah_tuntas: tuntas,
   553→    distribusi: distribusi, per_siswa: perSiswa, tren_jenis: trenJenis
   554→  };
   555→}

   779→function hitungRekap(catatan) {
   780→  const r = { Hadir: 0, Sakit: 0, Izin: 0, Alpa: 0 };
   781→  catatan.forEach(function (k) { if (r[k.status] !== undefined) r[k.status]++; });
   782→  const total = r.Hadir + r.Sakit + r.Izin + r.Alpa;
   783→  return {
   784→    hadir: r.Hadir, sakit: r.Sakit, izin: r.Izin, alpa: r.Alpa,
   785→    total_pertemuan: total,
   786→    persen_hadir: total ? Math.round(r.Hadir / total * 1000) / 10 : null
   787→  };
   788→}

   790→function actGuruGetRekapKehadiran(payload, token) {
   791→  requireAuth(token, ['guru']);
   792→  const dari = payload.dari ? String(payload.dari).slice(0, 10) : '';
   793→  const sampai = payload.sampai ? String(payload.sampai).slice(0, 10) : '';
   794→  const catatan = getAll(SHEETS.KEHADIRAN).filter(function (k) {
   795→    if (String(k.kelas_id) !== String(payload.kelas_id)) return false;
   796→    if (String(k.mapel_id) !== String(payload.mapel_id)) return false;
   797→    const t = String(k.tanggal).slice(0, 10);
   798→    if (dari && t < dari) return false;
   799→    if (sampai && t > sampai) return false;
   800→    return true;
   801→  });
   802→  const siswa = getAll(SHEETS.SISWA).filter(function (x) {
   803→    return String(x.kelas_id) === String(payload.kelas_id) && String(x.aktif).toUpperCase() !== 'FALSE';
   804→  });
   805→  const tanggalUnik = [];
   806→  catatan.forEach(function (k) {
   807→    const t = String(k.tanggal).slice(0, 10);
   808→    if (tanggalUnik.indexOf(t) === -1) tanggalUnik.push(t);
   809→  });
   810→  tanggalUnik.sort();
   811→
   812→  const perSiswa = siswa.map(function (sw) {
   813→    const milik = catatan.filter(function (k) { return String(k.siswa_id) === String(sw.id); });
   814→    return Object.assign({ siswa_id: sw.id, nama: sw.nama, nisn: sw.nisn }, hitungRekap(milik));
   815→  });
   816→
   817→  const perTanggal = tanggalUnik.map(function (t) {
   818→    const hari = catatan.filter(function (k) { return String(k.tanggal).slice(0, 10) === t; });
   819→    return Object.assign({ tanggal: t }, hitungRekap(hari));
   820→  });
   821→
   822→  return {
   823→    per_siswa: perSiswa,
   824→    per_tanggal: perTanggal,
   825→    jumlah_pertemuan: tanggalUnik.length,
   826→    ringkasan: hitungRekap(catatan)
   827→  };
   828→}

   832→  const mapel = getAll(SHEETS.MAPEL);

   931→function rekapKebiasaan(catatan) {
   932→  const perKebiasaan = KEBIASAAN7.map(function (k) {
   933→    const nilai = catatan.map(function (r) { return Number(r[k.kunci]) || 0; }).filter(function (v) { return v > 0; });
   934→    const rata = nilai.length ? nilai.reduce(function (a, b) { return a + b; }, 0) / nilai.length : null;
   935→    const dist = { 1: 0, 2: 0, 3: 0, 4: 0 };
   936→    nilai.forEach(function (v) { dist[v]++; });
   937→    return {
   938→      kunci: k.kunci, nama: k.nama,
   939→      rata_rata: rata === null ? null : Math.round(rata * 100) / 100,
   940→      persen: rata === null ? null : Math.round(rata / 4 * 1000) / 10,
   941→      jumlah_laporan: nilai.length, distribusi: dist,
   942→      predikat: predikatKebiasaan(rata)
   943→    };
   944→  });
   945→  const semuaNilai = [];
   946→  catatan.forEach(function (r) {
   947→    KEBIASAAN7.forEach(function (k) { const v = Number(r[k.kunci]) || 0; if (v > 0) semuaNilai.push(v); });
   948→  });
   949→  const rataUmum = semuaNilai.length ? semuaNilai.reduce(function (a, b) { return a + b; }, 0) / semuaNilai.length : null;
   950→
   951→  const perTanggal = catatan.slice().sort(function (a, b) { return String(a.tanggal).localeCompare(String(b.tanggal)); }).map(function (r) {
   952→    const v = KEBIASAAN7.map(function (k) { return Number(r[k.kunci]) || 0; }).filter(function (x) { return x > 0; });
   953→    return {
   954→      tanggal: String(r.tanggal).slice(0, 10),
   955→      rata_rata: v.length ? Math.round(v.reduce(function (a, b) { return a + b; }, 0) / v.length * 100) / 100 : null,
   956→      jumlah_terisi: v.length, catatan: r.catatan || ''
   957→    };
   958→  });
   959→
   960→  return {
   961→    per_kebiasaan: perKebiasaan,
   962→    per_tanggal: perTanggal,
   963→    jumlah_hari_lapor: catatan.length,
   964→    rata_umum: rataUmum === null ? null : Math.round(rataUmum * 100) / 100,
   965→    persen_umum: rataUmum === null ? null : Math.round(rataUmum / 4 * 1000) / 10,
   966→    predikat: predikatKebiasaan(rataUmum)
   967→  };
   968→}

  1251→function initDatabase() {
  1252→  const ditambahkan = [];
  1253→  Object.keys(SCHEMA).forEach(function (name) {
  1254→    const sh = sheet(name);
  1255→    if (sh.getLastRow() === 0) {
  1256→      sh.getRange(1, 1, 1, SCHEMA[name].length).setValues([SCHEMA[name]]);
  1257→      sh.setFrozenRows(1);
  1258→      ditambahkan.push('sheet ' + name);
  1259→      return;
  1260→    }
  1261→    // Migrasi: tambahkan kolom baru yang belum ada pada instalasi lama.
  1262→    const lastCol = sh.getLastColumn();
  1263→    const header = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(function (h) { return String(h); });
  1264→    const kurang = SCHEMA[name].filter(function (kol) { return header.indexOf(kol) === -1; });
  1265→    if (kurang.length) {
  1266→      sh.getRange(1, lastCol + 1, 1, kurang.length).setValues([kurang]);
  1267→      ditambahkan.push(name + ': ' + kurang.join(', '));
  1268→    }
  1269→  });
  1270→
  1271→  const settings = getSettings();
  1272→  if (!settings.nama_sekolah) {
  1273→    saveSettings({
  1274→      nama_sekolah: 'Nama Sekolah Anda',
  1275→      tahun_ajaran: '2026/2027',
  1276→      semester: 'Ganjil',
  1277→      kkm_default: 75,
  1278→      nama_kepala_sekolah: ''
  1279→    });
  1280→  }
  1281→
  1282→  const users = getAll(SHEETS.USERS);
  1283→  if (!users.length) {
  1284→    insert(SHEETS.USERS, {
  1285→      username: 'admin', password: hashPassword('admin123'), role: 'admin',
  1286→      nama: 'Administrator', ref_id: '', aktif: true, dibuat_pada: new Date().toISOString()
  1287→    });
  1288→  }
  1289→
  1290→  return 'Database SIMANTAP siap.' + (ditambahkan.length ? ' Ditambahkan: ' + ditambahkan.join(' | ') : ' Tidak ada perubahan struktur.');
  1291→}
