   128→function ChildSelector({
   129→  children,
   130→  selected,
   131→  onSelect,
   132→}: {
   133→  children: ChildOption[];
   134→  selected: string;

   143→          <Button
   144→            key={child.id}
   145→            variant={selected === child.id ? 'default' : 'outline'}
   146→            size="sm"
   147→            className={
   148→              selected === child.id
   149→                ? 'bg-[#1F3864] hover:bg-[#152850]'
   150→                : ''
   151→            }
   152→            onClick={() => onSelect(child.id)}
   153→          >

   156→              ({child.className})
   157→            </span>
   158→          </Button>
   159→        ))}

   161→    </div>
