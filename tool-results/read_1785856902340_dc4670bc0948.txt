   128→function getCountdown(dueDate: string): { text: string; urgent: boolean } {
   129→  const now = new Date(); now.setHours(0, 0, 0, 0);
   130→  const due = new Date(dueDate); due.setHours(0, 0, 0, 0);
   131→  const diff = Math.ceil((due.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   132→  if (diff < 0) return { text: `${Math.abs(diff)} hari lalu`, urgent: true };
   133→  if (diff === 0) return { text: 'Hari ini', urgent: true };
   134→  if (diff <= 3) return { text: `${diff} hari lagi`, urgent: true };
   135→  if (diff <= 7) return { text: `${diff} hari lagi`, urgent: false };
   136→  return { text: `${diff} hari lagi`, urgent: false };
   137→}
