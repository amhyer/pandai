   174→function formatDuration(minutes: number): string {
   175→  const h = Math.floor(minutes / 60);
   176→  const m = minutes % 60;
   177→  return h > 0 ? `${h} jam ${m > 0 ? `${m} menit` : ''}` : `${m} menit`;
   178→}

   180→function getDifficultyVariant(d: string) {
   181→  switch (d) {
   182→    case 'Mudah': return 'bg-emerald-50 text-emerald-700 border-emerald-200';
   183→    case 'Sedang': return 'bg-amber-50 text-amber-700 border-amber-200';
   184→    case 'Sukar': return 'bg-red-50 text-red-700 border-red-200';
   185→    default: return '';
   186→  }
   187→}

   189→function getDiscriminationVariant(d: string) {
   190→  switch (d) {
   191→    case 'Tinggi': return 'bg-emerald-50 text-emerald-700 border-emerald-200';
   192→    case 'Sedang': return 'bg-blue-50 text-blue-700 border-blue-200';
   193→    case 'Rendah': return 'bg-red-50 text-red-700 border-red-200';
   194→    default: return '';

   248→function AddClassDialog({
   249→  open,
   250→  onOpenChange,
   251→  onSubmit,
   252→  isSubmitting,
   253→  editData,
   254→}: {
   255→  open: boolean;
   256→  onOpenChange: (v: boolean) => void;
   257→  onSubmit: (data: ClassFormData) => void;
   258→  isSubmitting: boolean;
   259→  editData?: ClassRow | null;
   260→}) {
   261→  const [form, setForm] = useState<ClassFormData>({
   262→    name: '',
   263→    grade: '10',
   264→    academicYear: '2024/2025',
   265→  });
   266→
   267→  // Reset form when dialog opens or editData changes
   268→  useEffect(() => {
   269→    if (open && editData) {
   270→      setForm({ name: editData.name, grade: editData.grade, academicYear: editData.academicYear });
   271→    } else if (open) {
   272→      setForm({ name: '', grade: '10', academicYear: '2024/2025' });
   273→    }
   274→  }, [open, editData]);
   275→
   276→  const handleSubmit = (e: React.FormEvent) => {
   277→    e.preventDefault();
   278→    if (!form.name.trim()) {
   279→      toast.error('Nama kelas wajib diisi');
   280→      return;
   281→    }
   282→    onSubmit(form);
   283→  };
   284→
   285→  return (
   286→    <Dialog open={open} onOpenChange={onOpenChange}>
   287→      <DialogContent className="sm:max-w-md">
   288→        <DialogHeader>
   289→          <DialogTitle>{editData ? 'Edit Kelas' : 'Tambah Kelas Baru'}</DialogTitle>
   290→        </DialogHeader>
   291→        <form onSubmit={handleSubmit} className="space-y-4">
   292→          <div className="space-y-2">
   293→            <Label htmlFor="cls-name">Nama Kelas *</Label>
   294→            <Input
   295→              id="cls-name"
   296→              placeholder="Contoh: X IPA 1"
   297→              value={form.name}
   298→              onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   299→            />
   300→          </div>
   301→          <div className="grid grid-cols-2 gap-4">
   302→            <div className="space-y-2">
   303→              <Label htmlFor="cls-grade">Tingkat</Label>
   304→              <Select value={form.grade} onValueChange={(v) => setForm((f) => ({ ...f, grade: v }))}>
   305→                <SelectTrigger id="cls-grade">
   306→                  <SelectValue />
   307→                </SelectTrigger>
   308→                <SelectContent>
   309→                  <SelectItem value="10">Kelas 10</SelectItem>
   310→                  <SelectItem value="11">Kelas 11</SelectItem>
   311→                  <SelectItem value="12">Kelas 12</SelectItem>
   312→                </SelectContent>
   313→              </Select>
   314→            </div>
   315→            <div className="space-y-2">
   316→              <Label htmlFor="cls-year">Tahun Ajaran</Label>
   317→              <Input
   318→                id="cls-year"
   319→                placeholder="2024/2025"
   320→                value={form.academicYear}
   321→                onChange={(e) => setForm((f) => ({ ...f, academicYear: e.target.value }))}
   322→              />
   323→            </div>
   324→          </div>
   325→          <div className="flex justify-end gap-2 pt-2">
   326→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   327→              Batal
   328→            </Button>
   329→            <Button
   330→              type="submit"
   331→              className="bg-[#1F3864] hover:bg-[#152850]"

   335→            </Button>

   339→    </Dialog>

   341→}

   343→export function ClassesView() {
   344→  const user = useAppStore((s) => s.user);
   345→  const [classes, setClasses] = useState<ClassRow[]>([]);
   346→  const [loading, setLoading] = useState(true);
   347→  const [search, setSearch] = useState('');
   348→  const [gradeFilter, setGradeFilter] = useState<string>('all');
   349→  const [dialogOpen, setDialogOpen] = useState(false);
   350→  const [submitting, setSubmitting] = useState(false);
   351→  const [editClass, setEditClass] = useState<ClassRow | null>(null);
   352→  const [deleteId, setDeleteId] = useState<string | null>(null);
   353→
   354→  const fetchData = useCallback(async () => {
   355→    if (!user?.schoolId) {
   356→      // Use placeholder data when no schoolId (demo mode)
   357→      setClasses(PLACEHOLDER_CLASSES);
   358→      setLoading(false);
   359→      return;
   360→    }
   361→    try {
   362→      setLoading(true);
   363→      const res = await fetch(`/api/classes?schoolId=${user.schoolId}`);
   364→      if (res.ok) {
   365→        const data = await res.json();
   366→        setClasses(Array.isArray(data) ? data : data.classes ?? []);
   367→      } else {
   368→        setClasses(PLACEHOLDER_CLASSES);
   369→      }
   370→    } catch {
   371→      setClasses(PLACEHOLDER_CLASSES);
   372→    } finally {
   373→      setLoading(false);
   374→    }
   375→  }, [user?.schoolId]);
   376→
   377→  useEffect(() => {
   378→    fetchData();
   379→  }, [fetchData]);
   380→

   382→    setSubmitting(true);
   383→    try {
   384→      // Simulate API call

   388→          prev.map((c) =>

   397→          grade: form.grade,
   398→          academicYear: form.academicYear,
   399→          studentCount: 0,
   400→        };
   401→        setClasses((prev) => [...prev, newClass]);
   402→        toast.success(`Kelas "${form.name}" berhasil ditambahkan`);
   403→      }
   404→      setDialogOpen(false);
   405→      setEditClass(null);
   406→    } catch {

   409→      setSubmitting(false);

   446→            Tambah, edit, dan hapus rombongan belajar di sekolah Anda.
   447→          </p>
