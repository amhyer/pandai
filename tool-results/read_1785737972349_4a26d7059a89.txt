   904→                  const isBest = bestStudent && d.studentId === bestStudent.studentId;
   905→                  const isWorst = worstStudent && d.studentId === worstStudent.studentId;
   906→                  return (
   907→                    <TableRow key={d.studentId} className={cn(isBest && 'bg-emerald-50/50', isWorst && 'bg-red-50/50')}>
   908→                      <TableCell className="text-center text-muted-foreground">{idx + 1}</TableCell>
   909→                      <TableCell className="font-medium">{d.studentName}
   910→                        {isBest && <Badge className="ml-1 bg-emerald-100 text-emerald-700 border-0 text-[10px]">Terbaik</Badge>}
   911→                        {isWorst && <Badge className="ml-1 bg-red-100 text-red-700 border-0 text-[10px]">Perlu Perhatian</Badge>}
   912→                      </TableCell>
   913→                      {d.ratings.map((r, hIdx) => (
   914→                        <TableCell key={hIdx} className="text-center">
   915→                          <span className={cn('font-medium', r >= 4 ? 'text-emerald-600' : r >= 3 ? 'text-amber-600' : 'text-red-600')}>{r > 0 ? r.toFixed(1) : '-'}</span>
   916→                        </TableCell>
   917→                      ))}
   918→                      <TableCell className="text-center">
   919→                        <span className={cn('font-bold', avg >= 4 ? 'text-emerald-600' : avg >= 3 ? 'text-amber-600' : 'text-red-600')}>{avg.toFixed(1)}</span>
   920→                      </TableCell>
   921→                    </TableRow>
   922→                  );
   923→                })}
   924→              </TableBody>
   925→            </Table>
   926→          </div>
   927→        </CardContent>
   928→      </Card>
   929→    </div>
   930→  );
   931→}

  1024→
  1025→      <Card>
  1026→        <CardContent className="p-0">
  1027→          <div className="overflow-x-auto max-h-[560px] overflow-y-auto">
