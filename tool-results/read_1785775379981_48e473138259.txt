  1124→/**
  1125→ * Lookup schools by NPSN (exact match) or name/city/province/district (partial, case-insensitive)
  1126→ * Supports word-based matching: "SMA 1 Makassar" will match "SMA Negeri 1 Makassar"
  1127→ */
  1128→export function lookupSchool(query: string): (NpsnSchool & { source: string })[] {
  1129→  if (!query || !query.trim()) return [];
  1130→  const q = query.trim();
  1131→
  1132→  // Exact NPSN match first
  1133→  const npsnMatch = NPSN_DATABASE.filter((s) => s.npsn === q);
  1134→  if (npsnMatch.length > 0) {
  1135→    return npsnMatch.map((s) => ({ ...s, source: 'dapodik' }));
  1136→  }
  1137→
  1138→  const lower = q.toLowerCase();
  1139→
  1140→  // Partial match (case-insensitive) on name, city, province, and district
  1141→  const substringMatches = NPSN_DATABASE.filter(
  1142→    (s) =>
  1143→      s.name.toLowerCase().includes(lower) ||
  1144→      s.city.toLowerCase().includes(lower) ||
  1145→      s.province.toLowerCase().includes(lower) ||
  1146→      s.district.toLowerCase().includes(lower) ||
  1147→      s.address.toLowerCase().includes(lower)
  1148→  );
  1149→  if (substringMatches.length > 0) {
  1150→    return substringMatches.map((s) => ({ ...s, source: 'dapodik' }));
  1151→  }
  1152→
  1153→  // Word-based matching: all significant words (>= 2 chars) must appear in the name/city/province
  1154→  const keywords = lower.split(/\s+/).filter((w) => w.length >= 2);
  1155→  if (keywords.length >= 2) {
  1156→    const wordMatches = NPSN_DATABASE.filter((s) => {
  1157→      const searchPool = `${s.name} ${s.city} ${s.province} ${s.district} ${s.schoolType}`.toLowerCase();
  1158→      return keywords.every((kw) => searchPool.includes(kw));
  1159→    });
  1160→    if (wordMatches.length > 0) {
  1161→      return wordMatches.map((s) => ({ ...s, source: 'dapodik' }));
  1162→    }
  1163→  }
  1164→
  1165→  // Single word / short query: match against name, city, province
  1166→  const shortMatches = NPSN_DATABASE.filter((s) => {
  1167→    const searchPool = `${s.name} ${s.city} ${s.province} ${s.district} ${s.schoolType}`.toLowerCase();
  1168→    return keywords.some((kw) => searchPool.includes(kw));
  1169→  });
  1170→
  1171→  return shortMatches.map((s) => ({ ...s, source: 'dapodik' }));
  1172→}

  1177→export function getSchoolByNpsn(npsn: string): (NpsnSchool & { source: string }) | null {
  1178→  const school = NPSN_DATABASE.find((s) => s.npsn === npsn);
  1179→  if (!school) return null;
  1180→  return { ...school, source: 'dapodik' };
