   418→// ═══════════════════════════════════════════════════════════════════════
   419→// Sidebar Navigation — grouped by section (SIMANTAP pattern)
   420→// ═══════════════════════════════════════════════════════════════════════
   421→
   422→interface SidebarNavProps {
   423→  sections: NavSection[];
   424→  currentView: ViewType;
   425→  onNavigate: (view: ViewType) => void;
   426→  onLogout: () => void;
   427→}
   428→
   429→function SidebarNav({ sections, currentView, onNavigate, onLogout }: SidebarNavProps) {
   430→  const [collapsed, setCollapsed] = React.useState<Record<string, boolean>>({});
   431→
   432→  const toggleSection = (section: string) => {
   433→    setCollapsed((prev) => ({ ...prev, [section]: !prev[section] }));
   434→  };
   435→
   436→  return (
   437→    <div className="flex h-full flex-col min-h-0">
   438→      {/* Scrollable nav area */}
   439→      <ScrollArea className="flex-1 px-3 py-2 min-h-0">
   440→        <nav className="flex flex-col gap-1 pb-2" role="navigation" aria-label="Main navigation">
   441→          {sections.map((group) => (
   442→            <React.Fragment key={group.section}>
   443→              {/* Section label — clickable to collapse */}
   444→              <button
   445→                onClick={() => toggleSection(group.section)}
   446→                className="mt-4 mb-1 flex w-full items-center justify-between px-3 first:mt-0 group/section"
   447→                aria-expanded={!collapsed[group.section]}
   448→              >
   449→                <span className="text-[11px] font-semibold uppercase tracking-wider text-white/40">
   450→                  {group.section}
   451→                </span>
   452→                <ChevronRight
   453→                  className={cn(
   454→                    'h-3 w-3 text-white/30 transition-transform duration-200',
   455→                    !collapsed[group.section] && 'rotate-90'
   456→                  )}
   457→                />
   458→              </button>
   459→
   460→              {/* Items — animated collapse */}
   461→              <div
   462→                className={cn(
   463→                  'overflow-hidden transition-all duration-200 ease-in-out',
   464→                  collapsed[group.section] ? 'max-h-0 opacity-0' : 'max-h-[600px] opacity-100'
   465→                )}
   466→              >
   467→                {group.items.map((item) => {
   468→                  const Icon = item.icon;
   469→                  const isActive = currentView === item.view;
   470→                  return (
   471→                    <button
   472→                      key={item.view}
   473→                      onClick={() => onNavigate(item.view)}
   474→                      className={cn(
   475→                        'flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium transition-all duration-150 text-left',
   476→                        'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/30',
   477→                        'hover:scale-[1.02] active:scale-[0.98]',
   478→                        isActive
   479→                          ? 'bg-white/10 text-amber-400 shadow-sm shadow-amber-400/10 border-l-[3px] border-amber-400'
   480→                          : 'text-white/70 hover:bg-white/5 hover:text-white border-l-[3px] border-transparent'
   481→                      )}
   482→                      aria-current={isActive ? 'page' : undefined}
   483→                    >
   484→                      <Icon className="h-4.5 w-4.5 shrink-0" />
   485→                      <span className="truncate">{item.label}</span>
   486→                    </button>
   487→                  );
   488→                })}
   489→              </div>
   490→            </React.Fragment>
   491→          ))}
   492→        </nav>
   493→      </ScrollArea>
   494→
   495→      {/* Logout at bottom — always visible */}
   496→      <div className="shrink-0 border-t border-white/10 p-3">
   497→        <button
   498→          onClick={onLogout}
   499→          className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium text-white/60 transition-all duration-150 hover:bg-red-500/10 hover:text-red-400 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/30 text-left hover:scale-[1.02] active:scale-[0.98]"
   500→        >
   501→          <LogOut className="h-4.5 w-4.5 shrink-0" />
   502→          <span>Keluar</span>
   503→        </button>
   504→      </div>
   505→    </div>
   506→  );
   507→}

   516→  const sidebarOpen = useAppStore((s) => s.sidebarOpen);
   517→  const navigateTo = useAppStore((s) => s.navigateTo);
   518→  const logout = useAppStore((s) => s.logout);

   525→    ?.split(' ')

   528→      .toUpperCase()
   529→      .slice(0, 2) ?? 'U';

   562→      currentView={currentView}

   564→      onLogout={handleLogout}
   565→    />

   567→
   568→  return (
   569→    <div className="flex h-screen overflow-hidden bg-gray-50">
   570→      {/* ── Desktop Sidebar ── */}
   571→      <aside
   572→        className={cn(
   573→          'hidden lg:flex lg:flex-col lg:w-64 bg-[#1F3864] shrink-0 transition-all duration-300 overflow-hidden',
   574→          !sidebarOpen && 'lg:w-0'

   577→        {/* Branding + User info — fixed at top */}
   578→        <div className="shrink-0">
   579→          {sidebarBranding}
   580→          {sidebarUser && (
   581→            <>
   582→              <div className="mx-4">
   583→                <Separator className="bg-white/10" />
   584→              </div>
   585→              {sidebarUser}
   586→            </>
   587→          )}
   588→        </div>
   589→        {/* Nav takes remaining space with scroll */}
   590→        <div className="flex-1 min-h-0">
   591→          {sidebarNav}
   592→        </div>
   593→      </aside>
   594→
   595→      {/* ── Mobile Sidebar (Sheet) ── */}
   596→      <Sheet>
   597→        <SheetTrigger asChild>
   598→          <Button
   599→            variant="ghost"
   600→            size="icon"
   601→            className="fixed left-4 top-3 z-40 lg:hidden"
   602→            aria-label="Toggle menu"
   603→          >
   604→            <Menu className="h-5 w-5" />
   605→          </Button>
   606→        </SheetTrigger>
   607→        <SheetContent side="left" className="w-72 p-0 bg-[#1F3864] border-r-white/10 flex flex-col">
   608→          <SheetTitle className="sr-only">Menu Navigasi</SheetTitle>
   609→          {/* Branding + User info — fixed at top on mobile */}
   610→          <div className="shrink-0">
   611→            {sidebarBranding}
   612→            {sidebarUser && (
   613→              <>
   614→                <div className="mx-4">
   615→                  <Separator className="bg-white/10" />
   616→                </div>
   617→                {sidebarUser}
   618→              </>
   619→            )}
   620→          </div>
   621→          {/* Nav takes remaining space with scroll on mobile */}
   622→          <div className="flex-1 min-h-0">
   623→            {sidebarNav}
   624→          </div>
   625→        </SheetContent>
   626→      </Sheet>
   627→
   628→      {/* ── Right column: Header + Content ── */}
   629→      <div className="flex flex-1 flex-col overflow-hidden">
   630→        {/* ── Header ── */}
   631→        <header className="sticky top-0 z-30 flex h-14 shrink-0 items-center gap-4 border-b bg-white px-4 shadow-sm sm:px-6">

   653→                    <BreadcrumbItem>
   654→                      {isLast ? (
   655→                        <BreadcrumbPage>{crumb.label}</BreadcrumbPage>
   656→                      ) : (

   660→                        >
   661→                          {crumb.label}
   662→                        </BreadcrumbLink>

   664→                    </BreadcrumbItem>
   665→                    {!isLast && <BreadcrumbSeparator />}
   666→                  </React.Fragment>
   667→                );

   670→          </Breadcrumb>
   671→
   672→          {/* Mobile breadcrumb - just show current page */}
   673→          <span className="sm:hidden text-sm font-medium text-foreground">
   674→            {breadcrumbs[breadcrumbs.length - 1]?.label}
   675→          </span>
   676→
   677→          {/* Right side */}
   678→          <div className="ml-auto flex items-center gap-2">
   679→            {/* Notification bell */}
   680→            <Button
   681→              variant="ghost"
   682→              size="icon"
   683→              className="relative hover:bg-muted transition-colors duration-200"
   684→              aria-label="Notifikasi"
   685→              onClick={() => navigateTo('notifications')}

   687→              <Bell className="h-5 w-5" />

   689→            </Button>
   690→
   691→            {/* User dropdown */}
   692→            <DropdownMenu>
   693→              <DropdownMenuTrigger asChild>
   694→                <Button variant="ghost" className="relative h-9 w-9 rounded-full">
   695→                  <Avatar className="h-9 w-9">
   696→                    <AvatarFallback className="bg-[#1F3864] text-xs font-semibold text-white">
   697→                      {initials}
   698→                    </AvatarFallback>
   699→                  </Avatar>
   700→                </Button>
   701→              </DropdownMenuTrigger>
   702→              <DropdownMenuContent className="w-48" align="end" forceMount>
   703→                {user && (
   704→                  <>
   705→                    <div className="flex flex-col gap-1 p-2">
   706→                      <p className="text-sm font-medium leading-none">{user.name}</p>
   707→                      <p className="text-xs text-muted-foreground">{user.email}</p>
   708→                    </div>
   709→                    <DropdownMenuSeparator />
   710→                  </>
   711→                )}
   712→                <DropdownMenuItem onClick={() => navigateTo('profile')}>
   713→                  Profil

   715→                <DropdownMenuItem onClick={() => navigateTo('settings')}>
   716→                  Pengaturan
   717→                </DropdownMenuItem>
   718→                <DropdownMenuSeparator />
   719→                <DropdownMenuItem onClick={handleLogout} className="text-red-600 focus:text-red-600">
   720→                  Keluar
   721→                </DropdownMenuItem>
   722→              </DropdownMenuContent>
   723→            </DropdownMenu>
   724→          </div>
   725→        </header>
   726→
   727→        {/* ── Main Content ── */}
   728→        <main className="flex-1 overflow-y-auto bg-gradient-to-br from-slate-50 via-gray-50 to-slate-100 p-4 sm:p-6">
   729→          <div className="mx-auto max-w-7xl">
   730→            {children}
   731→          </div>
   732→        </main>
   733→      </div>
   734→    </div>
   735→  );
