   167→   167→function formatDuration(seconds: number): string {
   168→   168→  if (seconds <= 0) return '0m';
   169→   169→  const h = Math.floor(seconds / 3600);
   170→   170→  const m = Math.floor((seconds % 3600) / 60);
   171→   171→  if (h > 0) return `${h}j ${m}m`;
   172→   172→  return `${m}m`;
   173→   173→}

   215→   215→
   216→   216→export function SiswaNilaiView() {
   217→   217→  const user = useAppStore((s) => s.user);
   218→   218→  const [loading, setLoading] = useState(true);
   219→   219→  const [attempts, setAttempts] = useState<Attempt[]>([]);
   220→   220→  const [exams, setExams] = useState<ExamPackage[]>([]);
   221→   221→  const [expandedSubject, setExpandedSubject] = useState<string | null>(null);
   222→   222→  const [searchQuery, setSearchQuery] = useState('');
   223→   223→  const [error, setError] = useState<string | null>(null);
   224→   224→
   225→   225→  const fetchData = useCallback(async () => {
   226→   226→    if (!user?.id) return;
   227→   227→    setLoading(true);
   228→   228→    setError(null);
   229→   229→    try {
   230→   230→      const [attemptsRes, examsRes] = await Promise.all([
   231→   231→        fetch(`/api/attempts?userId=${user.id}`),
   232→   232→        fetch(`/api/exams${user.schoolId ? `?schoolId=${user.schoolId}` : ''}`),
   233→   233→      ]);
   234→   234→      if (!attemptsRes.ok || !examsRes.ok) throw new Error('Gagal mengambil data');
   235→   235→      const attemptsData: Attempt[] = await attemptsRes.json();
   236→   236→      const examsData: ExamPackage[] = await examsRes.json();
   237→   237→      setAttempts(attemptsData);
   238→   238→      setExams(examsData);
   239→   239→    } catch (err) {
   240→   240→      console.error(err);
   241→   241→      setError('Gagal memuat data nilai. Coba refresh halaman.');
   242→   242→      toast.error('Gagal memuat data nilai');
   243→   243→    } finally {
   244→   244→      setLoading(false);
   245→   245→    }
   246→   246→  }, [user?.id, user?.schoolId]);
   247→   247→
   248→   248→  useEffect(() => {
   249→   249→    fetchData();
   250→   250→  }, [fetchData]);
   251→   251→
   252→   252→  // Build a map of examPackageId → title
   253→   253→  const examTitleMap = useMemo(() => {
   254→   254→    const map = new Map<string, string>();
   255→   255→    exams.forEach((e) => map.set(e.id, e.title));
   256→   256→    return map;
   257→   257→  }, [exams]);
   258→   258→
   259→   259→  // Only consider submitted attempts for grading
   260→   260→  const completedAttempts = useMemo(
   261→   261→    () => attempts.filter((a) => a.status === 'submitted' || a.status === 'graded'),
   262→   262→    [attempts]
   263→   263→  );
   264→   264→
   265→   265→  // Group by subject from answers
   266→   266→  const subjectScores = useMemo((): SubjectScore[] => {
   267→   267→    const subjectMap = new Map<string, {
   268→   268→      subjectId: string;
   269→   269→      subjectName: string;
   270→   270→      subjectCode: string;
   271→   271→      attempts: Attempt[];
   272→   272→    }>();
   273→   273→
   274→   274→    completedAttempts.forEach((attempt) => {
   275→   275→      if (!attempt.answers || attempt.answers.length === 0) return;
   276→   276→      // Use the first answer's subject as representative
   277→   277→      const firstSubject = attempt.answers[0]?.question?.subject;
   278→   278→      if (!firstSubject) return;
   279→   279→      const sid = firstSubject.id;
   280→   280→      if (!subjectMap.has(sid)) {
   281→   281→        subjectMap.set(sid, {
   282→   282→          subjectId: sid,
   283→   283→          subjectName: firstSubject.name,
   284→   284→          subjectCode: firstSubject.code,
   285→   285→          attempts: [],
   286→   286→        });
   287→   287→      }
   288→   288→      subjectMap.get(sid)!.attempts.push(attempt);
   289→   289→    });
   290→   290→
   291→   291→    return Array.from(subjectMap.values()).map((group) => {
   292→   292→      const pcts = group.attempts.map((a) => a.percentage).sort((a, b) => b - a);
   293→   293→      const avg = pcts.reduce((s, v) => s + v, 0) / pcts.length;
   294→   294→      const latest = group.attempts.sort(
   295→   295→        (a, b) => new Date(b.submittedAt || b.startedAt).getTime() - new Date(a.submittedAt || a.startedAt).getTime()
   296→   296→      )[0];
   297→   297→      const prev = group.attempts.length > 1
   298→   298→        ? group.attempts.sort(
   299→   299→            (a, b) => new Date(b.submittedAt || b.startedAt).getTime() - new Date(a.submittedAt || a.startedAt).getTime()
   300→   300→          )[1]
   301→   301→        : null;
   302→   302→
   303→   303→      let trend: 'up' | 'down' | 'stable' | 'new' = 'new';
   304→   304→      if (prev) {
   305→   305→        const diff = latest.percentage - prev.percentage;
   306→   306→        if (diff > 2) trend = 'up';
   307→   307→        else if (diff < -2) trend = 'down';
   308→   308→        else trend = 'stable';
   309→   309→      }
   310→   310→
   311→   311→      return {
   312→   312→        subjectId: group.subjectId,
   313→   313→        subjectName: group.subjectName,
   314→   314→        subjectCode: group.subjectCode,
   315→   315→        attempts: group.attempts,
   316→   316→        avgPercentage: Math.round(avg * 10) / 10,
   317→   317→        bestPercentage: pcts[0],
   318→   318→        latestPercentage: latest.percentage,
   319→   319→        trend,
   320→   320→        totalAttempts: group.attempts.length,
   321→   321→      };
   322→   322→    }).sort((a, b) => b.avgPercentage - a.avgPercentage);
   323→   323→  }, [completedAttempts]);
   324→   324→
   325→   325→  // Filter by search
   326→   326→  const filteredSubjects = useMemo(() => {
   327→   327→    if (!searchQuery.trim()) return subjectScores;
   328→   328→    const q = searchQuery.toLowerCase();
   329→   329→    return subjectScores.filter(
   330→   330→      (s) =>
   331→   331→        s.subjectName.toLowerCase().includes(q) ||
   332→   332→        s.subjectCode.toLowerCase().includes(q)
   333→   333→    );
   334→   334→  }, [subjectScores, searchQuery]);
   335→   335→
   336→   336→  // Overall stats
   337→   337→  const overallStats = useMemo(() => {
   338→   338→    const allPcts = completedAttempts.map((a) => a.percentage);
   339→   339→    const avg = allPcts.length > 0 ? Math.round(allPcts.reduce((s, v) => s + v, 0) / allPcts.length) : 0;
   340→   340→    const highest = allPcts.length > 0 ? Math.round(Math.max(...allPcts)) : 0;
   341→   341→    const lowest = allPcts.length > 0 ? Math.round(Math.min(...allPcts)) : 0;
   342→   342→    return { avg, highest, lowest, total: completedAttempts.length };
   343→   343→  }, [completedAttempts]);
   344→   344→
   345→   345→  return (
   346→   346→    <div className="space-y-6">
   347→   347→      {/* Header */}
   348→   348→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   349→   349→        <div className="flex items-center gap-3">
   350→   350→          <div className="h-10 w-10 rounded-xl bg-gradient-to-br from-[#1F3864] to-[#2d4a7a] flex items-center justify-center shadow-sm">
   351→   351→            <BarChart3 className="h-5 w-5 text-white" />
   352→   352→          </div>
   353→   353→          <div>
   354→   354→            <h1 className="text-2xl font-bold tracking-tight">Nilai Saya</h1>
   355→   355→            <p className="text-sm text-muted-foreground">
   356→   356→              Pantau perkembangan skor dan lihat kemajuanmu
   357→   357→            </p>
   358→   358→          </div>
   359→   359→        </div>
   360→   360→        <Button
   361→   361→          variant="outline"
   362→   362→          size="sm"
   363→   363→          className="transition-all duration-200 hover:shadow-sm active:scale-[0.98]"
   364→   364→          onClick={() => {
   365→   365→            fetchData();
   366→   366→            toast.success('Data berhasil diperbarui');
   367→   367→          }}
   368→   368→        >
   369→   369→          <Target className="mr-1.5 h-4 w-4" />
   370→   370→          Refresh Data
   371→   371→        </Button>
   372→   372→      </div>
   373→   373→
   374→   374→      {/* Error State */}
   375→   375→      {error && (
   376→   376→        <Card className={`${cardBase} border-red-200 bg-red-50`}>
   377→   377→          <CardContent className="p-4 flex items-center gap-3">
   378→   378→            <AlertCircle className="h-5 w-5 text-red-500 shrink-0" />
   379→   379→            <p className="text-sm text-red-700">{error}</p>
   380→   380→            <Button
   381→   381→              variant="outline"
   382→   382→              size="sm"
   383→   383→              className="ml-auto shrink-0 transition-all duration-200 hover:shadow-sm active:scale-[0.98]"
   384→   384→              onClick={fetchData}
   385→   385→            >
   386→   386→              Coba Lagi
   387→   387→            </Button>
   388→   388→          </CardContent>
   389→   389→        </Card>
   390→   390→      )}
   391→   391→
   392→   392→      {/* Overview Stats */}
   393→   393→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   394→   394→        {loading ? (
   395→   395→          Array.from({ length: 4 }).map((_, i) => (
   396→   396→            <Card key={i} className={cardBase}>
   397→   397→              <CardContent className="p-4 space-y-2">
   398→   398→                <Skeleton className="h-4 w-24" />
   399→   399→                <Skeleton className="h-8 w-16" />
   400→   400→                <Skeleton className="h-3 w-20" />
   401→   401→              </CardContent>
   402→   402→            </Card>
   403→   403→          ))
   404→   404→        ) : (
   405→   405→          <>
   406→   406→            <Card className={cardBase}>
   407→   407→              <CardContent className="p-4">
   408→   408→                <div className="flex items-center gap-2 mb-1">
   409→   409→                  <div className="h-7 w-7 rounded-lg bg-emerald-100 flex items-center justify-center">
   410→   410→                    <TrendingUp className="h-3.5 w-3.5 text-emerald-600" />
   411→   411→                  </div>
   412→   412→                  <p className="text-xs text-muted-foreground font-medium">Rata-rata</p>
   413→   413→                </div>
   414→   414→                <p className="text-3xl font-bold text-[#1F3864]">{overallStats.avg}</p>
   415→   415→                <p className="text-xs text-muted-foreground mt-0.5">dari 100</p>
   416→   416→              </CardContent>
   417→   417→            </Card>
   418→   418→            <Card className={cardBase}>
   419→   419→              <CardContent className="p-4">
   420→   420→                <div className="flex items-center gap-2 mb-1">
   421→   421→                  <div className="h-7 w-7 rounded-lg bg-amber-100 flex items-center justify-center">

   423→   423→                  </div>
   424→   424→                  <p className="text-xs text-muted-foreground font-medium">Tertinggi</p>
   425→   425→                </div>
   426→   426→                <p className="text-3xl font-bold text-amber-600">{overallStats.highest}</p>
   427→   427→                <p className="text-xs text-muted-foreground mt-0.5">skor terbaik</p>
   428→   428→              </CardContent>
   429→   429→            </Card>
   430→   430→            <Card className={cardBase}>
   431→   431→              <CardContent className="p-4">

   433→   433→                  <div className="h-7 w-7 rounded-lg bg-red-100 flex items-center justify-center">
   434→   434→                    <AlertCircle className="h-3.5 w-3.5 text-red-500" />
   435→   435→                  </div>
   436→   436→                  <p className="text-xs text-muted-foreground font-medium">Terendah</p>
