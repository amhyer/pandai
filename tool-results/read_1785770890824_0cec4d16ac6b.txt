   197→function generateMockAttendance(year: number, month: number): AttendanceDay[] {
   198→  const days: AttendanceDay[] = [];
   199→  const daysInMonth = new Date(year, month + 1, 0).getDate();
   200→  const statuses: Array<'hadir' | 'izin' | 'sakit' | 'alpa'> = ['hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'hadir', 'izin', 'sakit', 'alpa'];
   201→  const notes: Record<string, string> = { izin: 'Keperluan keluarga', sakit: 'Demam', alpa: 'Tanpa keterangan' };
   202→  let statusIdx = 0;
   203→  for (let d = 1; d <= daysInMonth; d++) {
   204→    const date = new Date(year, month, d);
   205→    const dow = date.getDay();
   206→    if (dow === 0 || dow === 6) {
   207→      days.push({ day: d, status: 'weekend', date: `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}` });
   208→    } else {
   209→      const s = statuses[statusIdx % statuses.length];
   210→      days.push({ day: d, status: s, date: `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`, note: notes[s] || '' });
   211→      statusIdx++;
   212→    }
   213→  }
   214→  return days;
   215→}

   661→function getCountdown(dueDate: string): string {
   662→  const now = new Date();
   663→  const due = new Date(dueDate);
   664→  const diffMs = due.getTime() - now.getTime();
   665→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   666→  if (diffDays < 0) return `Terlambat ${Math.abs(diffDays)} hari`;
   667→  if (diffDays === 0) return 'Hari ini';
   668→  if (diffDays === 1) return 'Besok';
   669→  if (diffDays <= 7) return `${diffDays} hari lagi`;
   670→  return `${Math.ceil(diffDays / 7)} minggu lagi`;
   671→}
   672→
   673→function getDueUrgency(dueDate: string): 'overdue' | 'today' | 'soon' | 'future' {
   674→  const now = new Date();
   675→  const due = new Date(dueDate);
   676→  const diffMs = due.getTime() - now.getTime();
   677→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   678→  if (diffDays < 0) return 'overdue';
   679→  if (diffDays === 0) return 'today';
   680→  if (diffDays <= 3) return 'soon';
   681→  return 'future';
   682→}

  1087→function PercentageRing({ percentage, size = 140, strokeWidth = 10, color = '#10B981' }: { percentage: number; size?: number; strokeWidth?: number; color?: string }) {
  1088→  const radius = (size - strokeWidth) / 2;
  1089→  const circumference = 2 * Math.PI * radius;
  1090→  const offset = circumference * (1 - percentage / 100);
  1091→
  1092→  return (
  1093→    <div className="relative" style={{ width: size, height: size }}>
  1094→      {/* Outer ring - background */}
  1095→      <svg width={size} height={size} className="transform -rotate-90">
  1096→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/20" />
  1097→        <circle
  1098→          cx={size / 2} cy={size / 2} r={radius} fill="none"
  1099→          stroke={color}
  1100→          strokeWidth={strokeWidth}
  1101→          strokeLinecap="round"
  1102→          strokeDasharray={circumference}
  1103→          strokeDashoffset={offset}
  1104→          className="transition-all duration-1000 ease-out"
  1105→        />
  1106→      </svg>
  1107→      {/* Inner decorative ring */}
  1108→      <div
  1109→        className="absolute rounded-full border-2 border-dashed border-muted/30"
  1110→        style={{
  1111→          top: strokeWidth + 8,
  1112→          left: strokeWidth + 8,
  1113→          right: strokeWidth + 8,
  1114→          bottom: strokeWidth + 8,
  1115→        }}
  1116→      />
  1117→      {/* Center text */}
  1118→      <div className="absolute inset-0 flex items-center justify-center">
  1119→        <div className="text-center">
  1120→          <span className="text-2xl font-bold" style={{ color: BRAND }}>{percentage}%</span>
