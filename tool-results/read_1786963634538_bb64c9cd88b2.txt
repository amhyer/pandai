   157→   157→function formatDate(d: string) {
   158→   158→  if (!d) return '-';
   159→   159→  const date = new Date(d);
   160→   160→  return date.toLocaleDateString('id-ID', {
   161→   161→    day: 'numeric',
   162→   162→    month: 'short',
   163→   163→    year: 'numeric',
   164→   164→    hour: '2-digit',
   165→   165→    minute: '2-digit',
   166→   166→  });
   167→   167→}
   168→   168→
   169→   169→function parseOptions(optionsJson: string | null): ParsedOption[] {
   170→   170→  if (!optionsJson) return [];
   171→   171→  try {
   172→   172→    const parsed = JSON.parse(optionsJson);
   173→   173→    if (Array.isArray(parsed)) return parsed;
   174→   174→    return [];
   175→   175→  } catch {
   176→   176→    return [];
   177→   177→  }
   178→   178→}
   179→   179→
   180→   180→function getCountdown(deadline: string): { text: string; urgent: boolean; past: boolean } {
   181→   181→  const now = new Date().getTime();
   182→   182→  const end = new Date(deadline).getTime();
   183→   183→  const diff = end - now;
   184→   184→
   185→   185→  if (diff <= 0) return { text: 'Waktu habis', urgent: true, past: true };
   186→   186→
   187→   187→  const days = Math.floor(diff / (1000 * 60 * 60 * 24));
   188→   188→  const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
   189→   189→  const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
   190→   190→
   191→   191→  if (days > 0) return { text: `${days} hari ${hours} jam lagi`, urgent: days <= 1, past: false };
   192→   192→  if (hours > 0) return { text: `${hours} jam ${minutes} menit lagi`, urgent: hours <= 3, past: false };
   193→   193→  return { text: `${minutes} menit lagi`, urgent: true, past: false };
   194→   194→}
