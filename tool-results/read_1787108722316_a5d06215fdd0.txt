   123→function getPredikat(grade: number | null): { label: string; color: string } {
   124→  if (grade === null) return { label: '-', color: 'bg-gray-100 text-gray-500' };
   125→  if (grade >= 90) return { label: 'A', color: 'bg-emerald-100 text-emerald-700' };
   126→  if (grade >= 80) return { label: 'B', color: 'bg-blue-100 text-blue-700' };
   127→  if (grade >= 70) return { label: 'C', color: 'bg-amber-100 text-amber-700' };
   128→  return { label: 'D', color: 'bg-red-100 text-red-700' };
   129→}
   130→
   131→function gradeCellColor(grade: number | null): string {
   132→  if (grade === null) return '';
   133→  return grade >= 70 ? 'text-emerald-700 font-bold' : 'text-red-600 font-bold';

   179→function GuruView() {
   180→  const user = useAppStore((s) => s.user)!;
   181→  const schoolId = user.schoolId;
   182→
   183→  // --- State ---
   184→  const [term, setTerm] = useState(DEFAULT_TERM);
   185→  const [selectedClassId, setSelectedClassId] = useState('');
   186→  const [selectedComponentId, setSelectedComponentId] = useState('');
   187→
   188→  const [classes, setClasses] = useState<ClassOption[]>([]);
   189→  const [components, setComponents] = useState<GradeComponent[]>([]);
   190→  const [students, setStudents] = useState<Student[]>([]);
   191→  const [existingGrades, setExistingGrades] = useState<StudentGrade[]>([]);
   192→  const [scoreInputs, setScoreInputs] = useState<Record<string, string>>({});
   193→
   194→  const [loading, setLoading] = useState(false);
   195→  const [saving, setSaving] = useState(false);
   196→  const [activeTab, setActiveTab] = useState('input');
   197→
   198→  // --- Rekap state ---
   199→  const [rekapClassId, setRekapClassId] = useState('');
   200→  const [rekapTerm, setRekapTerm] = useState(DEFAULT_TERM);
   201→  const [rekapData, setRekapData] = useState<FinalGradeResult[]>([]);
   202→  const [rekapLoading, setRekapLoading] = useState(false);
   203→  const [expandedStudent, setExpandedStudent] = useState<string | null>(null);
   204→
   205→  // --- Fetch classes ---
   206→  useEffect(() => {
   207→    if (!schoolId) return;
   208→    fetch(`/api/classes?schoolId=${schoolId}`)
   209→      .then((r) => r.json())
   210→      .then((data) => setClasses(Array.isArray(data) ? data : []))
   211→      .catch(() => {});
   212→  }, [schoolId]);
   213→
   214→  // --- Fetch components when term changes ---
   215→  useEffect(() => {
   216→    if (!term || !schoolId) return;
   217→    setLoading(true);
   218→    fetch(`/api/grade-components?term=${encodeURIComponent(term)}&schoolId=${schoolId}`)
   219→      .then((r) => r.json())
   220→      .then((data) => {
   221→        setComponents(Array.isArray(data) ? data : []);
   222→        setSelectedComponentId('');
   223→      })
   224→      .catch(() => setComponents([]))
   225→      .finally(() => setLoading(false));
   226→  }, [term, schoolId]);
   227→
   228→  // --- Fetch students when class selected ---
   229→  useEffect(() => {
   230→    if (!selectedClassId || !schoolId) return;
   231→    fetch(`/api/users?role=SISWA&schoolId=${schoolId}&classId=${selectedClassId}`)
   232→      .then((r) => r.json())
   233→      .then((data) => setStudents(Array.isArray(data) ? data : []))
   234→      .catch(() => setStudents([]));
   235→  }, [selectedClassId, schoolId]);
   236→
   237→  // --- Fetch existing grades when component + class selected ---
   238→  useEffect(() => {
   239→    if (!selectedComponentId || !selectedClassId) {
   240→      setExistingGrades([]);
   241→      setScoreInputs({});
   242→      return;
   243→    }
   244→    const params = new URLSearchParams({
   245→      componentId: selectedComponentId,
   246→      classId: selectedClassId,
   247→      term,
   248→    });
   249→    fetch(`/api/student-grades?${params}`)
   250→      .then((r) => r.json())
   251→      .then((data) => {
   252→        const grades: StudentGrade[] = Array.isArray(data) ? data : [];
   253→        setExistingGrades(grades);
   254→        const inputs: Record<string, string> = {};
   255→        grades.forEach((g) => {
   256→          inputs[g.studentId] = String(g.score);
   257→        });
   258→        setScoreInputs(inputs);
   259→      })
   260→      .catch(() => {
   261→        setExistingGrades([]);
   262→        setScoreInputs({});
   263→      });
   264→  }, [selectedComponentId, selectedClassId, term]);
   265→
   266→  // --- Save all scores ---
   267→  const handleSaveAll = async () => {
   268→    if (!selectedComponentId) return;
   269→    setSaving(true);
   270→    let successCount = 0;
   271→    let failCount = 0;
   272→
   273→    const selectedComponent = components.find((c) => c.id === selectedComponentId);
   274→
   275→    for (const student of students) {
   276→      const val = scoreInputs[student.id];
   277→      if (val === undefined || val === '') continue;
   278→      const score = parseFloat(val);
   279→      if (isNaN(score) || score < 0 || score > 100) continue;
   280→
   281→      try {
   282→        const res = await fetch('/api/student-grades', {
   283→          method: 'POST',
   284→          headers: { 'Content-Type': 'application/json' },
   285→          body: JSON.stringify({
   286→            studentId: student.id,
   287→            componentId: selectedComponentId,
   288→            score,
   289→            maxScore: 100,
   290→            source: 'MANUAL',
   291→            classId: selectedClassId,
   292→            subjectId: selectedComponent?.subjectId || null,
   293→            term,
   294→          }),
   295→        });
   296→        if (res.ok) successCount++;
   297→        else failCount++;
   298→      } catch {
   299→        failCount++;
   300→      }
   301→    }
   302→
   303→    setSaving(false);
   304→    if (successCount > 0) {
   305→      toast.success(`${successCount} nilai berhasil disimpan`);
   306→      // Re-fetch grades to reflect upsert
   307→      const params = new URLSearchParams({
   308→        componentId: selectedComponentId,
   309→        classId: selectedClassId,
   310→        term,
   311→      });
   312→      fetch(`/api/student-grades?${params}`)
   313→        .then((r) => r.json())
   314→        .then((data) => {
   315→          const grades: StudentGrade[] = Array.isArray(data) ? data : [];
   316→          setExistingGrades(grades);
   317→          const inputs: Record<string, string> = {};
   318→          grades.forEach((g) => { inputs[g.studentId] = String(g.score); });
   319→          setScoreInputs(inputs);
   320→        })
   321→        .catch(() => {});
   322→    }
   323→    if (failCount > 0) {
   324→      toast.error(`${failCount} nilai gagal disimpan`);
   325→    }
   326→  };
   327→
   328→  // --- Fetch rekap ---
   329→  const fetchRekap = useCallback(() => {
   330→    if (!rekapClassId || !rekapTerm) return;
   331→    setRekapLoading(true);
   332→    const params = new URLSearchParams({
   333→      mode: 'class',
   334→      classId: rekapClassId,
   335→      term: rekapTerm,
   336→    });
   337→    fetch(`/api/grades/final?${params}`)
   338→      .then((r) => r.json())
   339→      .then((data) => setRekapData(Array.isArray(data) ? data : []))
   340→      .catch(() => setRekapData([]))
   341→      .finally(() => setRekapLoading(false));
   342→  }, [rekapClassId, rekapTerm]);
   343→
   344→  useEffect(() => {
   345→    fetchRekap();
   346→  }, [fetchRekap]);
   347→
   348→  const selectedComponent = components.find((c) => c.id === selectedComponentId);
   349→  const totalWeight = components.reduce((s, c) => s + c.weight, 0);
   350→
   351→  return (
   352→    <div className="space-y-6">
   353→      <div>
   354→        <h1 className="text-2xl font-bold tracking-tight">Komponen Nilai & Bobot</h1>
   355→        <p className="text-muted-foreground">Kelola komponen penilaian dan input nilai siswa</p>
   356→      </div>
   357→
   358→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   359→        <TabsList>
   360→          <TabsTrigger value="input" className="gap-2">
   361→            <Pencil className="h-4 w-4" /> Input Nilai Manual
   362→          </TabsTrigger>
   363→          <TabsTrigger value="rekap" className="gap-2">
   364→            <BarChart3 className="h-4 w-4" /> Rekap Nilai Akhir
   365→          </TabsTrigger>
   366→        </TabsList>
   367→
   368→        {/* ─── Tab 1: Input Nilai Manual ─── */}
   369→        <TabsContent value="input">
   370→          <AnimatePresence mode="wait">
   371→            <motion.div key="input" {...fadeSlide}>
   372→              <Card>
   373→                <CardHeader>
   374→                  <CardTitle className="text-lg flex items-center gap-2">
   375→                    <FileSpreadsheet className="h-5 w-5" /> Input Nilai Siswa
   376→                  </CardTitle>
   377→                  <CardDescription>Pilih kelas dan komponen, lalu masukkan nilai 0–100</CardDescription>
   378→                </CardHeader>
   379→                <CardContent className="space-y-4">
   380→                  {/* Selectors row */}
   381→                  <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   382→                    <div className="space-y-2">
   383→                      <Label>Periode</Label>
   384→                      <Select value={term} onValueChange={setTerm}>
   385→                        <SelectTrigger><SelectValue /></SelectTrigger>
   386→                        <SelectContent>
   387→                          {TERMS.map((t) => (<SelectItem key={t} value={t}>{t}</SelectItem>))}
   388→                        </SelectContent>
   389→                      </Select>
   390→                    </div>
   391→                    <div className="space-y-2">
   392→                      <Label>Kelas</Label>
   393→                      <Select value={selectedClassId} onValueChange={setSelectedClassId}>
   394→                        <SelectTrigger><SelectValue placeholder="Pilih kelas" /></SelectTrigger>
   395→                        <SelectContent>
   396→                          {classes.map((c) => (<SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>))}
   397→                        </SelectContent>
   398→                      </Select>
   399→                    </div>
   400→                    <div className="space-y-2">
   401→                      <Label>Komponen</Label>
   402→                      <Select value={selectedComponentId} onValueChange={setSelectedComponentId} disabled={loading}>
   403→                        <SelectTrigger><SelectValue placeholder={loading ? 'Memuat...' : 'Pilih komponen'} /></SelectTrigger>
   404→                        <SelectContent>
   405→                          {components.length === 0 && !loading && (
   406→                            <SelectItem value="_none" disabled>Tidak ada komponen</SelectItem>
   407→                          )}
   408→                          {components.map((c) => (
   409→                            <SelectItem key={c.id} value={c.id}>
   410→                              {c.name} ({c.weight}%)
   411→                            </SelectItem>
   412→                          ))}
   413→                        </SelectContent>
   414→                      </Select>
   415→                    </div>
   416→                  </div>
   417→
   418→                  {/* Weight indicator */}
   419→                  {components.length > 0 && (
   420→                    <div className="flex items-center gap-3 p-3 rounded-lg bg-muted/50">
   421→                      <Scale className="h-4 w-4 text-muted-foreground shrink-0" />
   422→                      <div className="flex-1">
   423→                        <div className="flex justify-between text-sm mb-1">
   424→                          <span className="text-muted-foreground">Total bobot komponen</span>
   425→                          <span className={cn('font-medium', totalWeight === 100 ? 'text-emerald-600' : 'text-amber-600')}>
   426→                            {totalWeight}%
   427→                          </span>
   428→                        </div>
   429→                        <Progress
   430→                          value={totalWeight}
   431→                          className={cn('h-2',
   432→                            totalWeight === 100 ? '[&>[data-slot=progress-indicator]]:bg-emerald-500' : '[&>[data-slot=progress-indicator]]:bg-amber-400'
   433→                          )}
   434→                        />
   435→                      </div>
   436→                      {totalWeight === 100 ? (
   437→                        <CheckCircle2 className="h-4 w-4 text-emerald-500 shrink-0" />
   438→                      ) : (
   439→                        <AlertTriangle className="h-4 w-4 text-amber-500 shrink-0" />
   440→                      )}
   441→                    </div>
   442→                  )}
   443→
   444→                  {/* Students table */}
   445→                  {!selectedClassId ? (
   446→                    <div className="text-center py-12 text-muted-foreground">
   447→                      <Users className="h-10 w-10 mx-auto mb-3 opacity-40" />
   448→                      <p className="text-sm">Pilih kelas untuk menampilkan daftar siswa</p>
   449→                    </div>
   450→                  ) : students.length === 0 ? (
   451→                    <div className="text-center py-12 text-muted-foreground">
   452→                      <Users className="h-10 w-10 mx-auto mb-3 opacity-40" />
   453→                      <p className="text-sm">Tidak ada siswa di kelas ini</p>
   454→                    </div>
   455→                  ) : !selectedComponentId ? (
   456→                    <div className="text-center py-12 text-muted-foreground">
   457→                      <ListChecks className="h-10 w-10 mx-auto mb-3 opacity-40" />
   458→                      <p className="text-sm">Pilih komponen nilai untuk mulai menginput</p>
   459→                    </div>
   460→                  ) : (
   461→                    <>
   462→                      <div className="rounded-lg border overflow-x-auto">
   463→                        <Table>
   464→                          <TableHeader>
   465→                            <TableRow>
   466→                              <TableHead className="w-10">No</TableHead>
   467→                              <TableHead>Nama Siswa</TableHead>
   468→                              <TableHead className="w-40 text-center">Nilai (0–100)</TableHead>

   470→                          </TableHeader>
   471→                          <TableBody>
   472→                            {students.map((s, idx) => {
   473→                              const existing = existingGrades.find((g) => g.studentId === s.id);
   474→                              return (

   478→                                  <TableCell className="text-center">
   479→                                    <Input
   480→                                      type="number"
   481→                                      min={0}
   482→                                      max={100}
   483→                                      value={scoreInputs[s.id] ?? ''}
   484→                                      onChange={(e) =>
   485→                                        setScoreInputs((prev) => ({
   486→                                          ...prev,
   487→                                          [s.id]: e.target.value,
   488→                                        }))
   489→                                      }
   490→                                      className="w-28 mx-auto text-center"
   491→                                      placeholder="-"
   492→                                    />
   493→                                    {existing && (
   494→                                      <span className="text-[11px] text-muted-foreground ml-1">

   497→                                    )}
   498→                                  </TableCell>
   499→                                </TableRow>
   500→                              );
   501→                            })}
   502→                          </TableBody>
   503→                        </Table>
   504→                      </div>
   505→                      <div className="flex justify-end">
   506→                        <Button onClick={handleSaveAll} disabled={saving || students.length === 0} className="gap-2">
   507→                          {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Save className="h-4 w-4" />}

   509→                        </Button>
   510→                      </div>
   511→                    </>
   512→                  )}
   513→                </CardContent>
   514→              </Card>
   515→            </motion.div>
   516→          </AnimatePresence>
   517→        </TabsContent>
   518→
   519→        {/* ─── Tab 2: Rekap Nilai Akhir ─── */}
   520→        <TabsContent value="rekap">
   521→          <AnimatePresence mode="wait">
   522→            <motion.div key="rekap" {...fadeSlide}>
   523→              <Card>
   524→                <CardHeader>
   525→                  <CardTitle className="text-lg flex items-center gap-2">
   526→                    <BarChart3 className="h-5 w-5" /> Rekap Nilai Akhir Kelas
   527→                  </CardTitle>
   528→                  <CardDescription>Nilai akhir dihitung dengan normalisasi SIMANTAP</CardDescription>
   529→                </CardHeader>
   530→                <CardContent className="space-y-4">
   531→                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   532→                    <div className="space-y-2">
   533→                      <Label>Kelas</Label>
   534→                      <Select value={rekapClassId} onValueChange={setRekapClassId}>
   535→                        <SelectTrigger><SelectValue placeholder="Pilih kelas" /></SelectTrigger>
   536→                        <SelectContent>
   537→                          {classes.map((c) => (<SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>))}
   538→                        </SelectContent>
   539→                      </Select>
   540→                    </div>
   541→                    <div className="space-y-2">
   542→                      <Label>Periode</Label>
   543→                      <Select value={rekapTerm} onValueChange={setRekapTerm}>
   544→                        <SelectTrigger><SelectValue /></SelectTrigger>
   545→                        <SelectContent>
   546→                          {TERMS.map((t) => (<SelectItem key={t} value={t}>{t}</SelectItem>))}
   547→                        </SelectContent>
   548→                      </Select>
   549→                    </div>
   550→                  </div>
   551→
