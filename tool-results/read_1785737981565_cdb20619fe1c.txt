   945→  const [expandedRows, setExpandedRows] = useState<Set<string>>(new Set());
   946→  const [saving, setSaving] = useState(false);
   947→  const [form, setForm] = useState({ date: todayStr(), className: '', subject: '', topic: '', activities: '', notes: '' });
   948→  const fetchedRef = useRef(false);
   949→
   950→  useEffect(() => {
   951→    if (fetchedRef.current) return;
   952→    fetchedRef.current = true;
   953→    const controller = new AbortController();
   954→    (async () => {
   955→      try {
   956→        const res = await fetch(`/api/teaching-journals?schoolId=${user?.schoolId ?? ''}&teacherId=${user?.id ?? ''}&month=${currentMonth()}`, { signal: controller.signal });
   957→        if (res.ok) {
   958→          const data = await res.json();
   959→          const list = Array.isArray(data) ? data : (Array.isArray(data.journals) ? data.journals : []);
   960→          if (list.length > 0) {
   961→            setJournals(list.map((j: Record<string, string>) => ({ id: j.id, date: j.date || j.createdAt || '', className: j.className || j.kelas || '', subject: j.subject || j.mataPelajaran || '', topic: j.topic || '', activities: j.activities || '', notes: j.notes || '' })));
   962→            return;
   963→          }
   964→        }
   965→      } catch { /* use mock */ }
   966→    })();
   967→    return () => controller.abort();
   968→  }, [user?.schoolId, user?.id]);
   969→
   970→  const filtered = journals.filter((j) => {
   971→    if (dateFilter && !j.date.includes(dateFilter)) return false;
   972→    if (search) { const q = search.toLowerCase(); if (!j.topic.toLowerCase().includes(q) && !j.className.toLowerCase().includes(q) && !j.subject.toLowerCase().includes(q)) return false; }
   973→    return true;
   974→  });
   975→
   976→  const todayCount = journals.filter((j) => j.date === todayStr()).length;
   977→  const monthLabel = MONTHS.find((m) => m.value === currentMonth())?.label || '';
   978→
   979→  const toggleRow = (id: string) => setExpandedRows((prev) => { const next = new Set(prev); if (next.has(id)) next.delete(id); else next.add(id); return next; });
   980→
   981→  const handleSave = async () => {
   982→    if (!form.className || !form.subject || !form.topic) { toast.error('Kelas, mata pelajaran, dan topik wajib diisi'); return; }
   983→    setSaving(true);
   984→    try {
   985→      const res = await fetch('/api/teaching-journals', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ schoolId: user?.schoolId, teacherId: user?.id, ...form }) });
   986→      if (res.ok) { toast.success('Jurnal mengajar berhasil disimpan'); setDialogOpen(false); setSaving(false); return; }
   987→    } catch { /* fallback */ }
   988→    setJournals((prev) => [{ id: `j${Date.now()}`, ...form }, ...prev]);
   989→    toast.success('Jurnal mengajar berhasil disimpan (lokal)');
   990→    setDialogOpen(false);
   991→    setSaving(false);
   992→  };
   993→
   994→  const handleDelete = () => {
   995→    if (!selectedJournal) return;
   996→    setJournals((prev) => prev.filter((j) => j.id !== selectedJournal.id));
   997→    toast.success('Jurnal berhasil dihapus');
   998→    setDeleteOpen(false);
   999→    setSelectedJournal(null);
  1000→  };
  1001→
  1002→  return (
  1003→    <div className="space-y-6">
  1004→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
  1005→        <div><h1 className="text-2xl font-bold text-[#1F3864]">Jurnal Mengajar</h1><p className="text-sm text-muted-foreground">Catat aktivitas dan refleksi mengajar harian</p></div>
  1006→        <Button onClick={() => { setForm({ date: todayStr(), className: '', subject: '', topic: '', activities: '', notes: '' }); setDialogOpen(true); }} className="bg-[#1F3864] hover:bg-[#1F3864]/90 text-white">
  1007→          <Plus className="w-4 h-4 mr-2" />Tambah Jurnal
  1008→        </Button>
  1009→      </div>
  1010→
  1011→      <div className="grid grid-cols-2 lg:grid-cols-3 gap-4">
  1012→        <StatCard icon={<BookOpen className="w-5 h-5" />} label="Total Jurnal" value={journals.length} bg="bg-[#1F3864]/10" color="text-[#1F3864]" />
  1013→        <StatCard icon={<Calendar className="w-5 h-5" />} label={monthLabel} value={journals.length} bg="bg-emerald-50" color="text-emerald-600" />
  1014→        <StatCard icon={<Clock className="w-5 h-5" />} label="Hari Ini" value={todayCount} bg="bg-amber-50" color="text-amber-600" />
  1015→      </div>
  1016→
  1017→      <Card><CardContent className="p-4"><div className="flex flex-col sm:flex-row gap-4 items-start sm:items-center">
  1018→        <div className="relative flex-1 w-full sm:max-w-xs">
  1019→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
  1020→          <Input placeholder="Cari topik, kelas, mapel..." className="pl-9" value={search} onChange={(e) => setSearch(e.target.value)} />
  1021→        </div>
  1022→        <div className="space-y-1"><Label className="text-xs text-muted-foreground">Filter Tanggal</Label><Input type="date" value={dateFilter} onChange={(e) => setDateFilter(e.target.value)} className="w-48" /></div>
  1023→      </div></CardContent></Card>
  1024→
  1025→      <Card>
  1026→        <CardContent className="p-0">
  1027→          <div className="overflow-x-auto max-h-[560px] overflow-y-auto">
  1028→            <Table>
  1029→              <TableHeader>
  1030→                <TableRow className="bg-[#1F3864]/5 hover:bg-[#1F3864]/5">
  1031→                  <TableHead className="w-10"></TableHead>
  1032→                  <TableHead className="font-semibold">Tanggal</TableHead>
  1033→                  <TableHead className="font-semibold">Kelas</TableHead>
  1034→                  <TableHead className="font-semibold">Mata Pelajaran</TableHead>
  1035→                  <TableHead className="font-semibold">Topik</TableHead>
  1036→                  <TableHead className="text-right font-semibold">Aksi</TableHead>
  1037→                </TableRow>
  1038→              </TableHeader>
  1039→              <TableBody>
  1040→                {filtered.length === 0 && <TableRow><TableCell colSpan={6} className="text-center py-12 text-muted-foreground">Tidak ada jurnal</TableCell></TableRow>}
  1041→                {filtered.map((j) => {
  1042→                  const isExpanded = expandedRows.has(j.id);
  1043→                  return (
  1044→                    <React.Fragment key={j.id}>
  1045→                      <TableRow className="group cursor-pointer" onClick={() => toggleRow(j.id)}>
  1046→                        <TableCell className="w-10">{isExpanded ? <ChevronUp className="w-4 h-4 text-muted-foreground" /> : <ChevronDown className="w-4 h-4 text-muted-foreground" />}</TableCell>
  1047→                        <TableCell className="text-sm whitespace-nowrap">{formatDate(j.date)}</TableCell>
  1048→                        <TableCell><Badge variant="outline" className="font-normal">{j.className}</Badge></TableCell>
  1049→                        <TableCell className="font-medium">{j.subject}</TableCell>
  1050→                        <TableCell className="max-w-[200px]"><div className="truncate" title={j.topic}>{j.topic}</div></TableCell>
  1051→                        <TableCell className="text-right" onClick={(e) => e.stopPropagation()}>
  1052→                          <div className="flex gap-1 justify-end">
  1053→                            <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => { setForm({ date: j.date, className: j.className, subject: j.subject, topic: j.topic, activities: j.activities, notes: j.notes }); setDialogOpen(true); }}><Pencil className="w-4 h-4" /></Button>
  1054→                            <Button variant="ghost" size="icon" className="h-8 w-8 text-red-500 hover:text-red-700 hover:bg-red-50" onClick={() => { setSelectedJournal(j); setDeleteOpen(true); }}><Trash2 className="w-4 h-4" /></Button>
  1055→                          </div>
  1056→                        </TableCell>
  1057→                      </TableRow>
  1058→                      {isExpanded && (
  1059→                        <TableRow>
  1060→                          <TableCell colSpan={6} className="bg-muted/30 px-6 py-4">
  1061→                            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
  1062→                              <div><h4 className="text-xs font-semibold text-muted-foreground uppercase mb-1">Aktivitas</h4><p className="text-sm whitespace-pre-wrap">{j.activities || 'Tidak ada catatan'}</p></div>
  1063→                              <div><h4 className="text-xs font-semibold text-muted-foreground uppercase mb-1">Catatan</h4><p className="text-sm whitespace-pre-wrap">{j.notes || 'Tidak ada catatan'}</p></div>
  1064→                            </div>
  1065→                          </TableCell>
  1066→                        </TableRow>
  1067→                      )}
  1068→                    </React.Fragment>
  1069→                  );
  1070→                })}
  1071→              </TableBody>
  1072→            </Table>
  1073→          </div>
  1074→        </CardContent>
  1075→      </Card>
  1076→
  1077→      {/* Create/Edit Dialog */}
  1078→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
  1079→        <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
  1080→          <DialogHeader><DialogTitle className="text-[#1F3864]">Tambah Jurnal Mengajar</DialogTitle><DialogDescription>Catat kegiatan belajar mengajar hari ini</DialogDescription></DialogHeader>
  1081→          <div className="space-y-4">
  1082→            <div className="grid grid-cols-2 gap-4">
  1083→              <div className="space-y-2"><Label>Tanggal</Label><Input type="date" value={form.date} onChange={(e) => setForm((f) => ({ ...f, date: e.target.value }))} /></div>
  1084→              <div className="space-y-2"><Label>Kelas *</Label>
  1085→                <Select value={form.className} onValueChange={(v) => setForm((f) => ({ ...f, className: v }))}>
  1086→                  <SelectTrigger><SelectValue placeholder="Pilih kelas" /></SelectTrigger>
  1087→                  <SelectContent>{MOCK_CLASSES.map((c) => <SelectItem key={c.id} value={c.name}>{c.name}</SelectItem>)}</SelectContent>
  1088→                </Select>
  1089→              </div>
  1090→            </div>
  1091→            <div className="space-y-2"><Label>Mata Pelajaran *</Label>
  1092→              <Select value={form.subject} onValueChange={(v) => setForm((f) => ({ ...f, subject: v }))}>
  1093→                <SelectTrigger><SelectValue placeholder="Pilih mata pelajaran" /></SelectTrigger>
  1094→                <SelectContent>{SUBJECTS.map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}</SelectContent>
  1095→              </Select>
  1096→            </div>
  1097→            <div className="space-y-2"><Label>Topik *</Label><Input placeholder="Topik pembelajaran" value={form.topic} onChange={(e) => setForm((f) => ({ ...f, topic: e.target.value }))} /></div>
  1098→            <div className="space-y-2"><Label>Aktivitas</Label><Textarea placeholder="Deskripsikan aktivitas pembelajaran..." rows={4} value={form.activities} onChange={(e) => setForm((f) => ({ ...f, activities: e.target.value }))} /></div>
  1099→            <div className="space-y-2"><Label>Catatan</Label><Textarea placeholder="Catatan tambahan atau refleksi..." rows={3} value={form.notes} onChange={(e) => setForm((f) => ({ ...f, notes: e.target.value }))} /></div>
  1100→          </div>
  1101→          <DialogFooter>
  1102→            <Button variant="outline" onClick={() => setDialogOpen(false)}>Batal</Button>
  1103→            <Button onClick={handleSave} disabled={saving} className="bg-[#1F3864] hover:bg-[#1F3864]/90 text-white">{saving ? 'Menyimpan...' : 'Simpan Jurnal'}</Button>
  1104→          </DialogFooter>
  1105→        </DialogContent>
  1106→      </Dialog>
  1107→
  1108→      {/* Delete Dialog */}
  1109→      <AlertDialog open={deleteOpen} onOpenChange={setDeleteOpen}>
  1110→        <AlertDialogContent>
  1111→          <AlertDialogHeader><AlertDialogTitle>Hapus Jurnal</AlertDialogTitle><AlertDialogDescription>Apakah Anda yakin ingin menghapus jurnal &quot;{selectedJournal?.topic}&quot;?</AlertDialogDescription></AlertDialogHeader>
  1112→          <AlertDialogFooter><AlertDialogCancel>Batal</AlertDialogCancel><AlertDialogAction onClick={handleDelete} className="bg-red-600 hover:bg-red-700">Hapus</AlertDialogAction></AlertDialogFooter>
  1113→        </AlertDialogContent>
  1114→      </AlertDialog>
  1115→    </div>
  1116→  );
