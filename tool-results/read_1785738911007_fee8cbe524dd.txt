   294→   294→            name: name.trim(),
   295→   295→            email: email.trim(),
   296→   296→            password,
   297→   297→            role,
   298→   298→            schoolCode: schoolCode.trim(),
   299→   299→          }),

   302→   302→        const data = await res.json();

   325→   325→      setName('');

   335→   335→            <GraduationCap className="h-8 w-8 text-amber-500" />
   336→   336→            <span className="text-3xl font-bold tracking-tight bg-gradient-to-r from-amber-400 to-amber-600 bg-clip-text text-transparent">
   337→   337→              PANDAI
