   222→// 1. SUBJECTSVIEW — Mata Pelajaran
   223→// ═══════════════════════════════════════════════════════════════════════
   224→
   225→export function SubjectsView() {
   226→  const { user } = useAppStore();
   227→  const [subjects, setSubjects] = useState<Subject[]>([]);
   228→  const [loading, setLoading] = useState(true);
   229→  const [search, setSearch] = useState('');
   230→  const [dialogOpen, setDialogOpen] = useState(false);
   231→  const [deleteOpen, setDeleteOpen] = useState(false);
   232→  const [editingSubject, setEditingSubject] = useState<Subject | null>(null);
   233→  const [deletingSubject, setDeletingSubject] = useState<Subject | null>(null);
   234→  const [formCode, setFormCode] = useState('');
   235→  const [formName, setFormName] = useState('');
   236→  const [formType, setFormType] = useState<'wajib' | 'pilihan'>('wajib');
   237→  const [formOrder, setFormOrder] = useState(1);
   238→  const [saving, setSaving] = useState(false);
   239→
   240→  const fetchSubjects = useCallback(async () => {
   241→    try {
   242→      const res = await fetch('/api/subjects');
   243→      if (res.ok) {
   244→        const data = await res.json();
   245→        setSubjects(Array.isArray(data) ? data : data.data ?? []);
   246→      } else {
   247→        setSubjects(MOCK_SUBJECTS);
   248→      }
   249→    } catch {
   250→      setSubjects(MOCK_SUBJECTS);
   251→    } finally {
   252→      setLoading(false);
   253→    }
   254→  }, []);
   255→
   256→  useEffect(() => {
   257→    fetchSubjects();
   258→  }, [fetchSubjects]);
   259→
   260→  const filtered = useMemo(() => {
   261→    if (!search.trim()) return subjects;
   262→    const q = search.toLowerCase();
   263→    return subjects.filter(
   264→      (s) =>
   265→        s.code.toLowerCase().includes(q) ||
   266→        s.name.toLowerCase().includes(q) ||
   267→        s.type.toLowerCase().includes(q)
   268→    );
   269→  }, [subjects, search]);
   270→
   271→  const wajibCount = subjects.filter((s) => s.type === 'wajib').length;
   272→  const pilihanCount = subjects.filter((s) => s.type === 'pilihan').length;
   273→
   274→  function openAdd() {
   275→    setEditingSubject(null);
   276→    setFormCode('');
   277→    setFormName('');
   278→    setFormType('wajib');
   279→    setFormOrder(subjects.length + 1);
   280→    setDialogOpen(true);
   281→  }
   282→
   283→  function openEdit(s: Subject) {
   284→    setEditingSubject(s);
   285→    setFormCode(s.code);
   286→    setFormName(s.name);
   287→    setFormType(s.type);
   288→    setFormOrder(s.sortOrder);
   289→    setDialogOpen(true);
   290→  }
   291→
   292→  async function handleSave() {
   293→    if (!formCode.trim() || !formName.trim()) {
   294→      toast.error('Kode dan Nama mata pelajaran wajib diisi');
   295→      return;
   296→    }
   297→    setSaving(true);
   298→    try {
   299→      if (editingSubject) {
   300→        const res = await fetch(`/api/subjects?id=${editingSubject.id}`, {
   301→          method: 'PATCH',
   302→          headers: { 'Content-Type': 'application/json' },
   303→          body: JSON.stringify({
   304→            code: formCode.trim(),
   305→            name: formName.trim(),
   306→            type: formType,
   307→            sortOrder: formOrder,
   308→          }),
   309→        });
   310→        if (res.ok) {
   311→          toast.success('Mata pelajaran berhasil diperbarui');
   312→          setDialogOpen(false);
   313→          fetchSubjects();
   314→          return;
   315→        }
   316→      } else {
   317→        const res = await fetch('/api/subjects', {
   318→          method: 'POST',
   319→          headers: { 'Content-Type': 'application/json' },
   320→          body: JSON.stringify({
   321→            code: formCode.trim(),
   322→            name: formName.trim(),
   323→            type: formType,
   324→            sortOrder: formOrder,
   325→          }),
   326→        });
   327→        if (res.ok) {
   328→          toast.success('Mata pelajaran berhasil ditambahkan');
   329→          setDialogOpen(false);
   330→          fetchSubjects();
   331→          return;
   332→        }
   333→      }
   334→    } catch {
   335→      // fallback: update local state
   336→    }
   337→    // Mock fallback
   338→    if (editingSubject) {
   339→      setSubjects((prev) =>
   340→        prev.map((s) =>
   341→          s.id === editingSubject.id
   342→            ? { ...s, code: formCode.trim(), name: formName.trim(), type: formType, sortOrder: formOrder }
   343→            : s
   344→        )
   345→      );
   346→      toast.success('Mata pelajaran berhasil diperbarui');
   347→    } else {
   348→      const newSubject: Subject = {
   349→        id: `s${Date.now()}`,
   350→        code: formCode.trim(),
   351→        name: formName.trim(),
   352→        type: formType,
   353→        sortOrder: formOrder,
   354→      };
   355→      setSubjects((prev) => [...prev, newSubject]);
   356→      toast.success('Mata pelajaran berhasil ditambahkan');
   357→    }
   358→    setDialogOpen(false);
   359→    setSaving(false);
   360→  }
   361→
   362→  async function handleDelete() {
   363→    if (!deletingSubject) return;
   364→    try {
   365→      const res = await fetch(`/api/subjects?id=${deletingSubject.id}`, { method: 'DELETE' });
   366→      if (res.ok) {
   367→        toast.success('Mata pelajaran berhasil dihapus');
   368→        setDeleteOpen(false);
   369→        fetchSubjects();
   370→        return;
   371→      }
   372→    } catch {
   373→      // fallback
   374→    }
   375→    setSubjects((prev) => prev.filter((s) => s.id !== deletingSubject.id));
   376→    toast.success('Mata pelajaran berhasil dihapus');
   377→    setDeleteOpen(false);
   378→    setDeletingSubject(null);
   379→  }
   380→
   381→  if (loading) {
   382→    return (
   383→      <div className="space-y-6">
   384→        <div className="flex items-center justify-between">
   385→          <Skeleton className="h-8 w-64" />
   386→          <Skeleton className="h-10 w-36" />
   387→        </div>
   388→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   389→          {[1, 2, 3].map((i) => (
   390→            <Skeleton key={i} className="h-24" />
   391→          ))}
   392→        </div>
   393→        <Skeleton className="h-96" />
   394→      </div>
   395→    );
   396→  }
   397→
   398→  return (
   399→    <div className="space-y-6">
   400→      {/* Header */}
   401→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   402→        <div>
   403→          <h1 className="text-2xl font-bold tracking-tight" style={{ color: BRAND }}>
   404→            Mata Pelajaran
   405→          </h1>
   406→          <p className="text-sm text-muted-foreground">
   407→            Kelola daftar mata pelajaran untuk {user?.schoolName ?? 'sekolah Anda'}
   408→          </p>
   409→        </div>
   410→        <Button onClick={openAdd} className="gap-2" style={{ backgroundColor: BRAND }}>
   411→          <Plus className="h-4 w-4" />
   412→          Tambah Mapel
   413→        </Button>
   414→      </div>
   415→
   416→      {/* Stats */}
   417→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   418→        <Card className="border-l-4" style={{ borderLeftColor: BRAND }}>
   419→          <CardContent className="flex items-center gap-4 p-4">
   420→            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-lg" style={{ backgroundColor: `${BRAND}15` }}>
   421→              <BookOpen className="h-6 w-6" style={{ color: BRAND }} />
   422→            </div>
   423→            <div>
   424→              <p className="text-sm text-muted-foreground">Total Mata Pelajaran</p>
   425→              <p className="text-2xl font-bold" style={{ color: BRAND }}>{subjects.length}</p>
   426→            </div>
   427→          </CardContent>
   428→        </Card>
   429→        <Card className="border-l-4 border-l-emerald-500">
   430→          <CardContent className="flex items-center gap-4 p-4">
   431→            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-lg bg-emerald-50">
   432→              <Layers className="h-6 w-6 text-emerald-600" />
