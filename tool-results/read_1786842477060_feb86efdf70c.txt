   365→function PerformanceBar({ value, max = 100, height = 'h-2.5' }: { value: number; max?: number; height?: string }) {

   367→  const color = pct >= 75 ? 'bg-emerald-500' : pct >= 60 ? 'bg-amber-500' : 'bg-red-500';
   368→  return (
   369→    <div className={cn('w-full overflow-hidden rounded-full bg-muted', height)}>
   370→      <div
   371→        className={cn('h-full rounded-full transition-all duration-500', color)}
   372→        style={{ width: pct + '%' }}
   373→      />
   374→    </div>
   375→  );
   376→}
