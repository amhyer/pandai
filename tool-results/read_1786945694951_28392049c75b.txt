   157→function formatDate(d: string) {
   158→  if (!d) return '-';
   159→  const date = new Date(d);
   160→  return date.toLocaleDateString('id-ID', {
   161→    day: 'numeric',
   162→    month: 'short',
   163→    year: 'numeric',
   164→    hour: '2-digit',
   165→    minute: '2-digit',
   166→  });
   167→}
   168→
   169→function parseOptions(optionsJson: string | null): ParsedOption[] {
   170→  if (!optionsJson) return [];
   171→  try {
   172→    const parsed = JSON.parse(optionsJson);
   173→    if (Array.isArray(parsed)) return parsed;
   174→    return [];
   175→  } catch {
   176→    return [];
   177→  }
   178→}
   179→
   180→function getCountdown(deadline: string): { text: string; urgent: boolean; past: boolean } {
   181→  const now = new Date().getTime();
   182→  const end = new Date(deadline).getTime();
   183→  const diff = end - now;
   184→
   185→  if (diff <= 0) return { text: 'Waktu habis', urgent: true, past: true };
   186→
   187→  const days = Math.floor(diff / (1000 * 60 * 60 * 24));
   188→  const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
   189→  const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
   190→
   191→  if (days > 0) return { text: `${days} hari ${hours} jam lagi`, urgent: days <= 1, past: false };
   192→  if (hours > 0) return { text: `${hours} jam ${minutes} menit lagi`, urgent: hours <= 3, past: false };
   193→  return { text: `${minutes} menit lagi`, urgent: true, past: false };
   194→}

   250→  // ═══════════════════════════════════════════════════════════════════

   261→    } catch {

   264→      setListLoading(false);
   265→    }

   267→
   268→  useEffect(() => {
   269→    fetchAssignments();

   271→
   272→  // ═══════════════════════════════════════════════════════════════════
   273→  // WORK VIEW — OPEN & LOAD

   283→    hasUnsavedRef.current = false;

   289→      ]);
   290→
   291→      if (!detailRes.ok) throw new Error('Gagal memuat tugas');
   292→      const detail: AssignmentDetail = await detailRes.json();
   293→      setWorkDetail(detail);
   294→
   295→      // Load existing answers from previous submission
   296→      if (subRes.ok) {
   297→        const sub: SubmissionWithAnswers | null = await subRes.json();

   299→          const loaded: Record<string, LocalAnswer> = {};

   301→            // questionId in the answer is Question.id
   302→            loaded[a.questionId] = {
   303→              answer: a.answer || undefined,
   304→              essayAnswer: a.essayAnswer || undefined,
   305→            };
   306→          }
   307→          setAnswers(loaded);
   308→        }
   309→      }
   310→    } catch {
   311→      toast.error('Gagal memuat tugas');
   312→      setView('list');
   313→    } finally {
   314→      setWorkLoading(false);
   315→    }
   316→  }, [studentId]);

   319→  // WORK VIEW — SAVE & SUBMIT
   320→  // ═══════════════════════════════════════════════════════════════════

   322→  const buildAnswersPayload = useCallback(() => {
   323→    return Object.entries(answers).map(([questionId, ans]) => ({

   325→      answer: ans.answer || null,
   326→      essayAnswer: ans.essayAnswer || null,
   327→    }));
   328→  }, [answers]);
   329→
   330→  const handleSaveDraft = useCallback(async () => {
   331→    if (!workId || !studentId) return;
   332→    setSaving(true);
   333→    try {
   334→      const payload = {
   335→        studentId,
   336→        schoolId,
   337→        classId,
   338→        action: 'draft' as const,
   339→        answers: buildAnswersPayload(),
   340→      };

   342→        method: 'POST',
   343→        headers: { 'Content-Type': 'application/json' },
   344→        body: JSON.stringify(payload),
   345→      });
   346→      if (!res.ok) throw new Error();
   347→      setLastSavedAt(new Date());
   348→      hasUnsavedRef.current = false;
   349→      toast.success('Tersimpan');
   350→    } catch {
   351→      toast.error('Gagal menyimpan jawaban');
   352→    } finally {
   353→      setSaving(false);
   354→    }
   355→  }, [workId, studentId, schoolId, classId, buildAnswersPayload]);
   356→
   357→  const handleSubmit = useCallback(async () => {
   358→    if (!workId || !studentId) return;
   359→    setSubmitting(true);
   360→    try {
   361→      const payload = {
   362→        studentId,
   363→        schoolId,
   364→        classId,
   365→        action: 'submit' as const,
   366→        answers: buildAnswersPayload(),
   367→      };
   368→      const res = await fetch(`/api/assignments/${workId}/submissions`, {
   369→        method: 'POST',
   370→        headers: { 'Content-Type': 'application/json' },
   371→        body: JSON.stringify(payload),
   372→      });
   373→      if (!res.ok) {
   374→        const err = await res.json().catch(() => null);
   375→        throw new Error(err?.error || 'Gagal mengumpulkan');
   376→      }
   377→      setConfirmSubmitOpen(false);
   378→      toast.success('Tugas berhasil dikumpulkan!');
   379→      // Navigate to result view
   380→      openResult(workId);
   381→    } catch (err) {
   382→      toast.error(err instanceof Error ? err.message : 'Gagal mengumpulkan tugas');
   383→    } finally {
   384→      setSubmitting(false);
   385→    }
   386→  }, [workId, studentId, schoolId, classId, buildAnswersPayload]);
   387→
   388→  // ═══════════════════════════════════════════════════════════════════
