   135→function getCountdown(dueDate: string): { text: string; urgent: boolean } {
   136→  const now = new Date(); now.setHours(0, 0, 0, 0);
   137→  const due = new Date(dueDate); due.setHours(0, 0, 0, 0);
   138→  const diff = Math.ceil((due.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   139→  if (diff < 0) return { text: `${Math.abs(diff)} hari lalu`, urgent: true };
   140→  if (diff === 0) return { text: 'Hari ini', urgent: true };
   141→  if (diff <= 3) return { text: `${diff} hari lagi`, urgent: true };
   142→  if (diff <= 7) return { text: `${diff} hari lagi`, urgent: false };
   143→  return { text: `${diff} hari lagi`, urgent: false };
   144→}
