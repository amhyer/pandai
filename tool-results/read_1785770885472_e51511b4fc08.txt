   221→function getStatusBadge(status: string) {
   222→  const s = status?.toLowerCase();
   223→  if (s === 'active' || s === 'aktif') {
   224→    return (
   225→      <Badge className="bg-emerald-50 text-emerald-700 rounded-full border-0">
   226→        <CheckCircle2 className="mr-1 h-3 w-3" />
   227→        Aktif
   228→      </Badge>
   229→    );
   230→  }
   231→  if (s === 'inactive' || s === 'nonaktif' || s === 'tidak aktif') {
   232→    return (
   233→      <Badge className="bg-gray-100 text-gray-500 rounded-full border-0">
   234→        <XCircle className="mr-1 h-3 w-3" />
   235→        Nonaktif
   236→      </Badge>
   237→    );
   238→  }
   239→  return (
   240→    <Badge className="bg-amber-50 text-amber-600 rounded-full border-0">
   241→      <AlertCircle className="mr-1 h-3 w-3" />
   242→      {status}
   243→    </Badge>
   244→  );
   245→}

   247→function useDebounce<T>(value: T, delay: number): T {
   248→  const [debouncedValue, setDebouncedValue] = useState<T>(value);
   249→  useEffect(() => {
   250→    const timer = setTimeout(() => setDebouncedValue(value), delay);
   251→    return () => clearTimeout(timer);
   252→  }, [value, delay]);
   253→  return debouncedValue;
   254→}

   277→export function UsersGlobalView() {
   278→  const [rawSearch, setRawSearch] = useState('');
   279→  const [roleFilter, setRoleFilter] = useState('all');

   281→  const [users, setUsers] = useState<User[]>([]);
   282→  const [loading, setLoading] = useState(true);
   283→  const [deleteTarget, setDeleteTarget] = useState<User | null>(null);
   284→  const [deleting, setDeleting] = useState(false);
   285→
   286→  const debouncedSearch = useDebounce(rawSearch, 300);
   287→
   288→  useEffect(() => {
   289→    async function fetchUsers() {
   290→      try {
   291→        setLoading(true);
   292→        const res = await fetch('/api/users');
   293→        if (res.ok) {
   294→          const json = await res.json();
   295→          if (Array.isArray(json.data) && json.data.length > 0) {
   296→            setUsers(
   297→              json.data.map((u: Record<string, string>) => ({
   298→                id: String(u.id ?? ''),
   299→                name: u.name ?? u.fullName ?? '',
   300→                email: u.email ?? u.username ?? '',
   301→                role: u.role ?? '',
   302→                school: u.schoolName ?? u.school ?? '',
   303→                status: u.status ?? 'active',
   304→              }))
   305→            );
   306→          } else {
   307→            setUsers(FALLBACK_USERS);
   308→          }
   309→        } else {
   310→          setUsers(FALLBACK_USERS);
   311→        }
   312→      } catch {
   313→        setUsers(FALLBACK_USERS);
   314→      } finally {
   315→        setLoading(false);
   316→      }
   317→    }
   318→    fetchUsers();
   319→  }, []);
   320→
   321→  const filtered = useMemo(() => {
   322→    let list = users;
   323→    if (roleFilter !== 'all') {
   324→      list = list.filter((u) => u.role === roleFilter);
   325→    }
   326→    if (debouncedSearch.trim()) {
   327→      const q = debouncedSearch.toLowerCase();
   328→      list = list.filter(
   329→        (u) =>
   330→          u.name.toLowerCase().includes(q) ||
   331→          u.email.toLowerCase().includes(q) ||
   332→          u.school.toLowerCase().includes(q)
   333→      );
   334→    }
   335→    return list;
   336→  }, [users, debouncedSearch, roleFilter]);
   337→
   338→  const totalPages = Math.max(1, Math.ceil(filtered.length / ITEMS_PER_PAGE));
   339→  const paginated = filtered.slice(
   340→    (currentPage - 1) * ITEMS_PER_PAGE,
   341→    currentPage * ITEMS_PER_PAGE
   342→  );
   343→
   344→  useEffect(() => {

   371→      toast.error('Terjadi kesalahan saat menghapus pengguna.');

  1124→function AnimatedNumber({ value, duration = 800 }: { value: number | string; duration?: number }) {
  1125→  const [display, setDisplay] = useState('0');
  1126→  const targetNum = typeof value === 'number' ? value : parseFloat(value.replace(/,/g, ''));
  1127→  const isFormatted = typeof value === 'string' && value.includes(',');
  1128→  const decimalPlaces = typeof value === 'string' && value.includes('.') ? value.split('.')[1]?.length ?? 0 : 0;
  1129→
  1130→  useEffect(() => {
  1131→    const startTime = performance.now();
  1132→    function tick(now: number) {
  1133→      const elapsed = now - startTime;
  1134→      const progress = Math.min(elapsed / duration, 1);
  1135→      const eased = 1 - Math.pow(1 - progress, 3);
  1136→      const current = targetNum * eased;
  1137→      if (decimalPlaces > 0) {
  1138→        setDisplay(current.toFixed(decimalPlaces));
  1139→      } else if (isFormatted) {
  1140→        setDisplay(Math.round(current).toLocaleString());
  1141→      } else {
  1142→        setDisplay(Math.round(current).toString());
  1143→      }
  1144→      if (progress < 1) requestAnimationFrame(tick);
  1145→    }
  1146→    requestAnimationFrame(tick);
  1147→  }, [targetNum, duration, decimalPlaces, isFormatted]);
  1148→
  1149→  return <span>{display}</span>;
  1150→}
