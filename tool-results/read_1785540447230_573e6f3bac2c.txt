    70→// ─── Section Wrapper with scroll-triggered animation ────────────────────────
    71→
    72→function AnimatedSection({
    73→  children,
    74→  className = '',
    75→  delay = 0,
    76→}: {
    77→  children: React.ReactNode;
    78→  className?: string;
    79→  delay?: number;
    80→}) {
    81→  const [inView, setInView] = useState(false);
    82→
    83→  useEffect(() => {
    84→    const observer = new IntersectionObserver(
    85→      ([entry]) => {
    86→        if (entry.isIntersecting) setInView(true);
    87→      },
    88→      { threshold: 0.1, rootMargin: '0px 0px -50px 0px' }
    89→    );
    90→    const el = document.getElementById(`section-${className.replace(/\s/g, '-')}`);
    91→    if (el) observer.observe(el);
    92→    return () => observer.disconnect();
    93→  }, [className]);
    94→
    95→  return (
    96→    <motion.section
    97→      id={`section-${className.replace(/\s/g, '-')}`}
    98→      initial="hidden"
    99→      animate={inView ? 'visible' : 'hidden'}
   100→      variants={staggerContainer}
   101→      className={className}
   102→    >
   103→      {children}
   104→    </motion.section>
   105→  );
   106→}
