   229→function getInitials(name: string): string {
   230→  return name
   231→    .split(/[,\s.]+/)
   232→    .filter(Boolean)
   233→    .slice(0, 2)
   234→    .map((w) => w[0])
   235→    .join('')
   236→    .toUpperCase();
   237→}
   238→
   239→function getAvatarColor(name: string): string {
   240→  const colors = [
   241→    'bg-violet-500', 'bg-emerald-500', 'bg-amber-500',
   242→    'bg-sky-500', 'bg-rose-500', 'bg-teal-500',
   243→  ];
   244→  let hash = 0;
   245→  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
   246→  return colors[Math.abs(hash) % colors.length];
   247→}

   317→export function SubjectsView() {
   318→  const { user } = useAppStore();
   319→  const [subjects, setSubjects] = useState<Subject[]>([]);
   320→  const [loading, setLoading] = useState(true);
   321→  const [search, setSearch] = useState('');

   323→  const [dialogOpen, setDialogOpen] = useState(false);
   324→  const [deleteOpen, setDeleteOpen] = useState(false);
   325→  const [editingSubject, setEditingSubject] = useState<Subject | null>(null);
   326→  const [deletingSubject, setDeletingSubject] = useState<Subject | null>(null);
   327→  const [formCode, setFormCode] = useState('');
   328→  const [formName, setFormName] = useState('');
   329→  const [formType, setFormType] = useState<'wajib' | 'pilihan'>('wajib');
   330→  const [formOrder, setFormOrder] = useState(1);
   331→  const [saving, setSaving] = useState(false);
   332→
   333→  const fetchSubjects = useCallback(async () => {
   334→    try {
   335→      const res = await fetch('/api/subjects');
   336→      if (res.ok) {
   337→        const data = await res.json();
   338→        setSubjects(Array.isArray(data) ? data : data.data ?? []);
   339→      } else {
   340→        setSubjects(MOCK_SUBJECTS);
   341→      }
   342→    } catch {
   343→      setSubjects(MOCK_SUBJECTS);
   344→    } finally {
   345→      setLoading(false);
   346→    }
   347→  }, []);
   348→
   349→  useEffect(() => {
   350→    fetchSubjects();
   351→  }, [fetchSubjects]);
   352→
   353→  const filtered = useMemo(() => {
   354→    let result = subjects;
   355→    if (typeFilter !== 'semua') {
   356→      result = result.filter((s) => s.type === typeFilter);
   357→    }
   358→    if (search.trim()) {
   359→      const q = search.toLowerCase();
   360→      result = result.filter(
   361→        (s) =>
   362→          s.code.toLowerCase().includes(q) ||
   363→          s.name.toLowerCase().includes(q)
   364→      );
   365→    }
   366→    return result;
   367→  }, [subjects, search, typeFilter]);
   368→
   369→  const wajibCount = subjects.filter((s) => s.type === 'wajib').length;
   370→  const pilihanCount = subjects.filter((s) => s.type === 'pilihan').length;
   371→
   372→  function openAdd() {
   373→    setEditingSubject(null);
   374→    setFormCode('');
   375→    setFormName('');
   376→    setFormType('wajib');
   377→    setFormOrder(subjects.length + 1);
   378→    setDialogOpen(true);
   379→  }
   380→
   381→  function openEdit(s: Subject) {
   382→    setEditingSubject(s);
   383→    setFormCode(s.code);
   384→    setFormName(s.name);
   385→    setFormType(s.type);
   386→    setFormOrder(s.sortOrder);
   387→    setDialogOpen(true);
   388→  }
   389→
   390→  async function handleSave() {
   391→    if (!formCode.trim() || !formName.trim()) {
   392→      toast.error('Kode dan Nama mata pelajaran wajib diisi');
   393→      return;
   394→    }
   395→    setSaving(true);
   396→    try {
   397→      if (editingSubject) {
   398→        const res = await fetch(`/api/subjects?id=${editingSubject.id}`, {
   399→          method: 'PATCH',
   400→          headers: { 'Content-Type': 'application/json' },
   401→          body: JSON.stringify({
   402→            code: formCode.trim(),
   403→            name: formName.trim(),
   404→            type: formType,
   405→            sortOrder: formOrder,
   406→          }),
   407→        });
   408→        if (res.ok) {
   409→          toast.success('Mata pelajaran berhasil diperbarui');
   410→          setDialogOpen(false);
   411→          fetchSubjects();
   412→          return;
   413→        }
   414→      } else {
   415→        const res = await fetch('/api/subjects', {
   416→          method: 'POST',
   417→          headers: { 'Content-Type': 'application/json' },
   418→          body: JSON.stringify({
   419→            code: formCode.trim(),
   420→            name: formName.trim(),
   421→            type: formType,
   422→            sortOrder: formOrder,
   423→          }),
   424→        });
   425→        if (res.ok) {
   426→          toast.success('Mata pelajaran berhasil ditambahkan');
   427→          setDialogOpen(false);
   428→          fetchSubjects();
   429→          return;
   430→        }
   431→      }
   432→    } catch {
   433→      // fallback
   434→    }
   435→    if (editingSubject) {
   436→      setSubjects((prev) =>
   437→        prev.map((s) =>
   438→          s.id === editingSubject.id
   439→            ? { ...s, code: formCode.trim(), name: formName.trim(), type: formType, sortOrder: formOrder }
   440→            : s

  1887→function useDebounce(value: string, delay: number): string {
  1888→  const [debounced, setDebounced] = useState(value);
  1889→  useEffect(() => {
  1890→    const timer = setTimeout(() => setDebounced(value), delay);
  1891→    return () => clearTimeout(timer);
  1892→  }, [value, delay]);
  1893→  return debounced;
  1894→}
