   555→// ═══════════════════════════════════════════════════════════════════════
   556→// Sidebar Navigation — grouped by section (SIMANTAP pattern)
   557→// ═══════════════════════════════════════════════════════════════════════
   558→
   559→interface SidebarNavProps {
   560→  sections: NavSection[];
   561→  currentView: ViewType;
   562→  onNavigate: (view: ViewType) => void;
   563→  onLogout: () => void;
   564→}
   565→
   566→function SidebarNav({ sections, currentView, onNavigate, onLogout }: SidebarNavProps) {
   567→  const [collapsed, setCollapsed] = React.useState<Record<string, boolean>>({});
   568→
   569→  const toggleSection = (section: string) => {
   570→    setCollapsed((prev) => ({ ...prev, [section]: !prev[section] }));
   571→  };
   572→
   573→  return (
   574→    <div className="flex h-full flex-col min-h-0">
   575→      {/* Scrollable nav area */}
   576→      <ScrollArea className="flex-1 px-3 py-2 min-h-0">
   577→        <nav className="flex flex-col gap-1 pb-2" role="navigation" aria-label="Main navigation">
   578→          {sections.map((group) => (
   579→            <React.Fragment key={group.section}>
   580→              {/* Section label — clickable to collapse */}
   581→              <button
   582→                onClick={() => toggleSection(group.section)}
   583→                className="mt-4 mb-1 flex w-full items-center justify-between px-3 first:mt-0 group/section"
   584→                aria-expanded={!collapsed[group.section]}
   585→              >
   586→                <span className="text-[11px] font-semibold uppercase tracking-wider text-white/40">
   587→                  {group.section}
   588→                </span>
   589→                <ChevronRight
   590→                  className={cn(
   591→                    'h-3 w-3 text-white/30 transition-transform duration-200',
   592→                    !collapsed[group.section] && 'rotate-90'
   593→                  )}
   594→                />
   595→              </button>
   596→
   597→              {/* Items — animated collapse */}
   598→              <div
   599→                className={cn(
   600→                  'overflow-hidden transition-all duration-200 ease-in-out',
   601→                  collapsed[group.section] ? 'max-h-0 opacity-0' : 'max-h-[600px] opacity-100'
   602→                )}
   603→              >
   604→                {group.items.map((item) => {
   605→                  const Icon = item.icon;
   606→                  const isActive = currentView === item.view;
   607→                  return (
   608→                    <button
   609→                      key={item.view}
   610→                      onClick={() => onNavigate(item.view)}
   611→                      className={cn(
   612→                        'flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium transition-all duration-150 text-left',
   613→                        'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/30',
   614→                        'hover:scale-[1.02] active:scale-[0.98]',
   615→                        isActive
   616→                          ? 'bg-white/10 text-amber-400 shadow-sm shadow-amber-400/10 border-l-[3px] border-amber-400'
   617→                          : 'text-white/70 hover:bg-white/5 hover:text-white border-l-[3px] border-transparent'
   618→                      )}
   619→                      aria-current={isActive ? 'page' : undefined}
   620→                    >
   621→                      <Icon className="h-4.5 w-4.5 shrink-0" />
   622→                      <span className="truncate">{item.label}</span>
   623→                    </button>
   624→                  );
   625→                })}
   626→              </div>
   627→            </React.Fragment>
   628→          ))}
   629→        </nav>
   630→      </ScrollArea>
   631→
   632→      {/* Logout at bottom — always visible */}
   633→      <div className="shrink-0 border-t border-white/10 p-3">
   634→        <button
   635→          onClick={onLogout}
   636→          className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium text-white/60 transition-all duration-150 hover:bg-red-500/10 hover:text-red-400 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/30 text-left hover:scale-[1.02] active:scale-[0.98]"
   637→        >
   638→          <LogOut className="h-4.5 w-4.5 shrink-0" />
   639→          <span>Keluar</span>
   640→        </button>
   641→      </div>
   642→    </div>
   643→  );
   644→}
