    90→function getInitials(name: string) {
    91→  return name
    92→    .split(' ')
    93→    .map((w) => w[0])
    94→    .slice(0, 2)
    95→    .join('')
    96→    .toUpperCase();
    97→}

    99→function getAvatarColor(name: string) {
   100→  const colors = [
   101→    'bg-gradient-to-br from-rose-400 to-rose-600',
   102→    'bg-gradient-to-br from-sky-400 to-sky-600',
   103→    'bg-gradient-to-br from-emerald-400 to-emerald-600',
   104→    'bg-gradient-to-br from-amber-400 to-amber-600',
   105→    'bg-gradient-to-br from-violet-400 to-violet-600',
   106→  ];
   107→  let hash = 0;
   108→  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
   109→  return colors[Math.abs(hash) % colors.length];
   110→}

   125→function getScoreColor(score: number) {
   126→  if (score >= 4.5) return { text: 'text-emerald-600', bg: 'bg-emerald-50', label: 'Luar Biasa' };
   127→  if (score >= 3.5) return { text: 'text-green-600', bg: 'bg-green-50', label: 'Baik' };
   128→  if (score >= 2.5) return { text: 'text-amber-600', bg: 'bg-amber-50', label: 'Cukup' };
   129→  return { text: 'text-red-500', bg: 'bg-red-50', label: 'Perlu Perhatian' };
   130→}

   132→function getBarColor(score: number) {
   133→  if (score >= 4) return 'bg-emerald-400';
   134→  if (score >= 3) return 'bg-amber-400';
   135→  if (score >= 2) return 'bg-orange-400';
   136→  return 'bg-red-400';

   143→function StarRating({
   144→  rating,
   145→  onChange,
   146→  size = 'md',
   147→  showLabel = false,
   148→}: {
   149→  rating: number;
   150→  onChange?: (r: number) => void;
   151→  size?: 'sm' | 'md' | 'lg';
   152→  showLabel?: boolean;
   153→}) {
   154→  const [hoveredStar, setHoveredStar] = useState(0);
   155→
   156→  const sizeClasses = { sm: 'text-lg', md: 'text-2xl', lg: 'text-3xl' };
   157→  const displayRating = hoveredStar || rating;
   158→
   159→  const ratingLabels = ['', 'Sangat Kurang', 'Kurang', 'Cukup', 'Baik', 'Sangat Baik'];
   160→
   161→  return (
   162→    <div className="flex items-center gap-1.5">
   163→      <div className="flex gap-0.5">
   164→        {[1, 2, 3, 4, 5].map((star) => (
   165→          <button
   166→            key={star}
   167→            type="button"
   168→            onClick={() => onChange?.(star === rating ? 0 : star)}
   169→            onMouseEnter={() => onChange && setHoveredStar(star)}
   170→            onMouseLeave={() => setHoveredStar(0)}
   171→            className={cn(
   172→              sizeClasses[size],
   173→              'transition-all duration-200',
   174→              onChange
   175→                ? 'cursor-pointer hover:scale-125 active:scale-95'
   176→                : 'cursor-default',
   177→              star <= displayRating
   178→                ? 'text-amber-400 drop-shadow-[0_0_6px_rgba(251,191,36,0.4)]'
   179→                : 'text-gray-300'
   180→            )}
   181→          >
   182→            ★
   183→          </button>
   184→        ))}
   185→      </div>
   186→      {showLabel && displayRating > 0 && (
   187→        <span className="text-xs text-muted-foreground font-medium ml-1">
   188→          {ratingLabels[displayRating]}
   189→        </span>
   190→      )}
   191→    </div>
   192→  );
   193→}
