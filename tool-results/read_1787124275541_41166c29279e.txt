   145→function getPredikatColor(predikat: string) {
   146→  switch (predikat?.toLowerCase()) {
   147→    case 'a':
   148→    case 'sangat baik':
   149→      return 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300';
   150→    case 'b':
   151→    case 'baik':
   152→      return 'bg-blue-100 text-blue-800 dark:bg-blue-900/40 dark:text-blue-300';
   153→    case 'c':

   168→async function fetchPdfBlob(url: string): Promise<Blob | null> {
   169→  try {
   170→    const res = await fetch(url);
   171→    if (!res.ok) {
   172→      const errText = await res.text().catch(() => 'Gagal mengunduh PDF');
   173→      throw new Error(errText);
   174→    }
   175→    return await res.blob();
   176→  } catch (err: unknown) {
   177→    const message = err instanceof Error ? err.message : 'Terjadi kesalahan';
   178→    toast.error(message);
   179→    return null;
   180→  }
   181→}
   182→
   183→function downloadBlob(blob: Blob, filename: string) {
   184→  const url = URL.createObjectURL(blob);
   185→  const a = document.createElement('a');
   186→  a.href = url;
   187→  a.download = filename;
   188→  document.body.appendChild(a);
   189→  a.click();
   190→  document.body.removeChild(a);
   191→  URL.revokeObjectURL(url);
   192→}
   193→
   194→function openBlobInTab(blob: Blob) {
   195→  const url = URL.createObjectURL(blob);
   196→  window.open(url, '_blank');
   197→}
