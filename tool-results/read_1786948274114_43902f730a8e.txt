   178→function formatDate(d: string) {
   179→  if (!d) return '-';
   180→  const date = new Date(d);
   181→  return date.toLocaleDateString('id-ID', {
   182→    day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   183→  });
   184→}
   185→
   186→function toLocalDatetime(d: string) {
   187→  if (!d) return '';
   188→  const date = new Date(d);
   189→  const pad = (n: number) => String(n).padStart(2, '0');
   190→  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}T${pad(date.getHours())}:${pad(date.getMinutes())}`;
   191→}
